import {
  ChangeDetectionStrategy,
  Component,
  OnInit,
  computed,
  inject,
  signal
} from '@angular/core';
import { RouterLink } from '@angular/router';

import { CustomerVehicleApi } from '../../customer-vehicle/services/customer-vehicle-api';
import { AuthSession } from '../../../core/auth/auth-session';
import { AgentCustomerListItem } from '../../../core/models/agent-customer.models';

@Component({
  selector: 'app-agent-dashboard',
  standalone: true,
  imports: [RouterLink],
  templateUrl: './agent-dashboard.html',
  styleUrl: './agent-dashboard.css',
  changeDetection: ChangeDetectionStrategy.OnPush
})
export class AgentDashboard implements OnInit {
  private readonly customerVehicleApi = inject(CustomerVehicleApi);
  private readonly authSession = inject(AuthSession);

  readonly customers = signal<AgentCustomerListItem[]>([]);
  readonly isLoading = signal(true);
  readonly errorMessage = signal<string | null>(null);

  readonly customerCount = computed(() => this.customers().length);

  readonly customersRequiringAttention = computed(() =>
    this.customers().filter(customer =>
      customer.profileStatus !== 'Completed'
    )
  );

  readonly attentionCount = computed(
    () => this.customersRequiringAttention().length
  );

  readonly agentName = computed(() => {
    const user = this.authSession.currentUser();

    if (!user) {
      return 'Agent';
    }

    const possibleName =
      (user as { firstName?: string | null }).firstName ??
      (user as { userName?: string | null }).userName ??
      (user as { username?: string | null }).username ??
      '';

    return possibleName.trim() || 'Agent';
  });

  ngOnInit(): void {
    this.loadCustomers();
  }

  loadCustomers(): void {
    this.isLoading.set(true);
    this.errorMessage.set(null);

    this.customerVehicleApi.getAgentCustomers().subscribe({
      next: customers => {
        this.customers.set(customers ?? []);
        this.isLoading.set(false);
      },
      error: error => {
        this.isLoading.set(false);

        const status = error?.status;

        if (status === 403) {
          this.errorMessage.set(
            'You are not authorized to view your assigned customers.'
          );
        } else if (status === 503) {
          this.errorMessage.set(
            'The customer service is temporarily unavailable.'
          );
        } else {
          this.errorMessage.set(
            'We could not load your agent workspace right now.'
          );
        }
      }
    });
  }
}
