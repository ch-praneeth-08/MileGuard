    [HttpGet(
        "policies/{policyId:guid}/document")]
    public async Task<IActionResult> GetDocument(
        Guid policyId)
    {
        var agentIdentityUserId =
            GetCurrentUserId();

        if (string.IsNullOrWhiteSpace(
                agentIdentityUserId))
        {
            return Unauthorized();
        }

        if (policyId == Guid.Empty)
        {
            return NotFound(
                new
                {
                    message =
                        "Policy was not found."
                });
        }

        var policy =
            await _policyRepository
                .GetByIdAsync(
                    policyId);

        if (policy is null)
        {
            return NotFound(
                new
                {
                    message =
                        "Policy was not found."
                });
        }

        var assignmentResult =
            await _customerVehicleClient
                .CheckAgentAssignmentAsync(
                    policy.CustomerProfileId,
                    agentIdentityUserId);

        if (assignmentResult.DependencyUnavailable)
        {
            return StatusCode(
                StatusCodes.Status503ServiceUnavailable,
                new
                {
                    message =
                        assignmentResult.Message ??
                        "Customer assignment information is temporarily unavailable."
                });
        }

        if (assignmentResult.NotFound)
        {
            return NotFound(
                new
                {
                    message =
                        assignmentResult.Message ??
                        "Customer profile was not found."
                });
        }

        if (!assignmentResult.Succeeded ||
            !assignmentResult.IsAssigned)
        {
            return StatusCode(
                StatusCodes.Status403Forbidden,
                new
                {
                    message =
                        "Customer is not assigned to the current Agent."
                });
        }

        var document =
            await _policyDocumentRepository
                .GetByPolicyIdAsync(
                    policy.Id);

        if (document is null)
        {
            return NotFound(
                new
                {
                    message =
                        "Policy document was not found."
                });
        }

        var content =
            await _policyDocumentStorage
                .ReadAsync(
                    document.StoredFileName);

        if (content is null)
        {
            return NotFound(
                new
                {
                    message =
                        "Policy document file is unavailable."
                });
        }

        return File(
            content,
            document.ContentType,
            document.FileName);
    }

    private string? GetCurrentUserId()
    {
        return User.FindFirstValue(
            ClaimTypes.NameIdentifier);
    }
}

    public InstallmentsController(
        IInstallmentPaymentService installmentPaymentService)

    }

    [HttpPost("{installmentId:guid}/payments")]
    public async Task<IActionResult> PayInstallment(
        Guid installmentId,
        [FromBody] MakePaymentRequest request)
    {
        var customerIdentityUserId =
            User.FindFirstValue(
                ClaimTypes.NameIdentifier);

        if (string.IsNullOrWhiteSpace(
                customerIdentityUserId))
        {
            return Unauthorized(
                new
                {
                    message =
                        "Authenticated Customer identity could not be determined."
                });
        }

        var result =
            await _installmentPaymentService
                .PayAsync(
                    installmentId,
                    customerIdentityUserId,
                    request);

        if (result.Succeeded &&
            result.Payment is not null)
        {
            return Ok(
                result.Payment);
        }

        if (result.PaymentFailed &&
            result.Payment is not null)
        {
            return Ok(
                result.Payment);
        }

        if (result.DependencyUnavailable)
        {
            return StatusCode(
                StatusCodes.Status503ServiceUnavailable,
                new
                {
                    message =
                        result.Message ??
                        "A required service is temporarily unavailable."
                });
        }

        if (result.Forbidden)
        {
            return StatusCode(
                StatusCodes.Status403Forbidden,
                new
                {
                    message =
                        result.Message ??
                        "You are not authorized to pay this installment."
                });
        }

        if (result.NotFound)
        {
            return NotFound(
                new
                {
                    message =
                        result.Message ??
                        "Installment was not found."
                });
        }

        if (result.InvalidState)
        {
            return Conflict(
                new
                {
                    message =
                        result.Message ??
                        "Installment cannot be paid in its current state."
                });
        }

        return BadRequest(
            new
            {
                message =
                    result.Message ??
                    "Installment payment could not be processed."
            });
    }
}

    [HttpPost("{purchaseId:guid}/payments")]
    public async Task<IActionResult> MakeFirstPayment(
    Guid purchaseId,
    [FromBody] MakePaymentRequest request)
    {
        var customerIdentityUserId =
            User.FindFirstValue(
                ClaimTypes.NameIdentifier);

        if (string.IsNullOrWhiteSpace(
                customerIdentityUserId))
        {
            return Unauthorized();
        }

        var result =
            await _firstPaymentService.PayAsync(
                purchaseId,
                customerIdentityUserId,
                request);

        if (result.Succeeded &&
            result.Payment is not null)
        {
            return Ok(
                result.Payment);
        }

        if (result.PaymentFailed &&
            result.Payment is not null)
        {
            return Ok(
                result.Payment);
        }

        if (result.DependencyUnavailable)
        {
            return StatusCode(
                StatusCodes.Status503ServiceUnavailable,
                new
                {
                    message =
                        result.Message
                });
        }

        if (result.Forbidden)
        {
            return StatusCode(
                StatusCodes.Status403Forbidden,
                new
                {
                    message =
                        result.Message
                });
        }

        if (result.NotFound)
        {
            return NotFound(
                new
                {
                    message =
                        result.Message
                });
        }

        if (result.InvalidState)
        {
            return Conflict(
                new
                {
                    message =
                        result.Message
                });
        }

        return BadRequest(
            new
            {
                message =
                    result.Message ??
                    "Payment could not be processed."
            });
    }
}
