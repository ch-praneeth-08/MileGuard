using System.Net;
using System.Net.Http.Headers;
using System.Net.Http.Json;
using ClaimsService.Application.Dtos.External;
using ClaimsService.Application.Exceptions;
using ClaimsService.Application.Interfaces.Clients;
using Microsoft.AspNetCore.Http;

namespace ClaimsService.Infrastructure.Clients;

public class CustomerVehicleClient : ICustomerVehicleClient
{
    private readonly HttpClient _httpClient;

    private readonly IHttpContextAccessor
        _httpContextAccessor;

    public CustomerVehicleClient(
        HttpClient httpClient,
        IHttpContextAccessor httpContextAccessor)
    {
        _httpClient =
            httpClient;

        _httpContextAccessor =
            httpContextAccessor;
    }

    public async Task<CustomerReferenceDto?> GetCustomerAsync(
        string customerIdentityUserId,
        CancellationToken cancellationToken = default)
    {
        if (string.IsNullOrWhiteSpace(
                customerIdentityUserId))
        {
            throw new ArgumentException(
                "Customer Identity User ID is required.",
                nameof(customerIdentityUserId));
        }

        using var request =
            new HttpRequestMessage(
                HttpMethod.Get,
                "api/customers/me");

        AddAuthorizationHeader(
            request);

        using var response =
            await SendAsync(
                request,
                cancellationToken);

        if (response.StatusCode ==
            HttpStatusCode.NotFound)
        {
            return null;
        }

        if (response.StatusCode ==
            HttpStatusCode.Forbidden)
        {
            throw new ForbiddenException(
                "Customer profile access was denied.");
        }

        if (response.StatusCode ==
            HttpStatusCode.Unauthorized)
        {
            throw new ForbiddenException(
                "Customer authorization is invalid or has expired.");
        }

        if (!response.IsSuccessStatusCode)
        {
            throw new ServiceUnavailableException(
                "Customer information could not be loaded.");
        }

        var customer =
            await response.Content
                .ReadFromJsonAsync<CustomerReferenceDto>(
                    cancellationToken:
                        cancellationToken);

        if (customer is null)
        {
            throw new ServiceUnavailableException(
                "Customer Vehicle service returned an invalid Customer response.");
        }

        return customer;
    }

    public async Task<bool> IsCustomerAssignedToAgentAsync(
        Guid customerProfileId,
        string agentIdentityUserId,
        CancellationToken cancellationToken = default)
    {
        if (customerProfileId ==
            Guid.Empty)
        {
            return false;
        }

        if (string.IsNullOrWhiteSpace(
                agentIdentityUserId))
        {
            return false;
        }

        using var request =
            new HttpRequestMessage(
                HttpMethod.Get,
                $"api/agent/customers/{customerProfileId}");

        AddAuthorizationHeader(
            request);

        using var response =
            await SendAsync(
                request,
                cancellationToken);

        if (response.IsSuccessStatusCode)
        {
            return true;
        }

        if (response.StatusCode ==
            HttpStatusCode.Forbidden)
        {
            return false;
        }

        if (response.StatusCode ==
            HttpStatusCode.NotFound)
        {
            return false;
        }

        if (response.StatusCode ==
            HttpStatusCode.Unauthorized)
        {
            throw new ForbiddenException(
                "Agent authorization is invalid or has expired.");
        }

        throw new ServiceUnavailableException(
            "Customer assignment information is temporarily unavailable.");
    }

    private async Task<HttpResponseMessage> SendAsync(
        HttpRequestMessage request,
        CancellationToken cancellationToken)
    {
        try
        {
            return await _httpClient.SendAsync(
                request,
                HttpCompletionOption.ResponseHeadersRead,
                cancellationToken);
        }
        catch (HttpRequestException)
        {
            throw new ServiceUnavailableException(
                "Customer Vehicle service is temporarily unavailable.");
        }
        catch (TaskCanceledException)
            when (!cancellationToken.IsCancellationRequested)
        {
            throw new ServiceUnavailableException(
                "Customer Vehicle service request timed out.");
        }
    }

    private void AddAuthorizationHeader(
        HttpRequestMessage request)
    {
        var httpContext =
            _httpContextAccessor.HttpContext;

        if (httpContext is null)
        {
            throw new ForbiddenException(
                "The current request context is unavailable.");
        }

        var authorization =
            httpContext
                .Request
                .Headers
                .Authorization
                .FirstOrDefault();

        if (string.IsNullOrWhiteSpace(
                authorization))
        {
            throw new ForbiddenException(
                "Authorization information is missing.");
        }

        if (!AuthenticationHeaderValue.TryParse(
                authorization,
                out var authorizationHeader))
        {
            throw new ForbiddenException(
                "Authorization information is invalid.");
        }

        request.Headers.Authorization =
            authorizationHeader;
    }
}

        GetEligibleClaimsOfficersAsync(
            CancellationToken cancellationToken = default)
    {
        using var response =
            await SendAsync(
                new HttpRequestMessage(
                    HttpMethod.Get,
                    "internal/identity/claims-officers/eligible"),
                cancellationToken);

        if (!response.IsSuccessStatusCode)
        {
            throw new ServiceUnavailableException(
                "Eligible Claims Officer information could not be loaded.");
        }

        var claimsOfficers =
            await response.Content
                .ReadFromJsonAsync<
                    List<ClaimsOfficerIdentityDto>>(
                    cancellationToken:
                        cancellationToken);

        return claimsOfficers ??
            [];
    }

    public async Task<ClaimsOfficerIdentityDto?>
        GetEligibleClaimsOfficerAsync(
            string identityUserId,
            CancellationToken cancellationToken = default)
    {
        if (string.IsNullOrWhiteSpace(
                identityUserId))
        {
            return null;
        }

        var encodedIdentityUserId =
            Uri.EscapeDataString(
                identityUserId);

        using var response =
            await SendAsync(
                new HttpRequestMessage(
                    HttpMethod.Get,
                    $"internal/identity/claims-officers/{encodedIdentityUserId}/eligible"),
                cancellationToken);

        if (response.StatusCode ==
            HttpStatusCode.NotFound)
        {
            return null;
        }

        if (!response.IsSuccessStatusCode)
        {
            throw new ServiceUnavailableException(
                "Claims Officer eligibility could not be verified.");
        }

        return await response.Content
            .ReadFromJsonAsync<
                ClaimsOfficerIdentityDto>(
                cancellationToken:
                    cancellationToken);
    }

using ClaimsService.Application.Interfaces.Clients;
using ClaimsService.Application.Interfaces.Repositories;
using ClaimsService.Infrastructure.Clients;
using ClaimsService.Infrastructure.Persistence;
using ClaimsService.Infrastructure.Repositories;
using Microsoft.EntityFrameworkCore;
using Microsoft.Extensions.Configuration;
using Microsoft.Extensions.DependencyInjection;

namespace ClaimsService.Infrastructure;

public static class DependencyInjection
{
    public static IServiceCollection AddInfrastructure(
        this IServiceCollection services,
        IConfiguration configuration)
    {
        var connectionString =
            configuration.GetConnectionString(
                "ClaimsDb");

        if (string.IsNullOrWhiteSpace(
                connectionString))
        {
            throw new InvalidOperationException(
                "Connection string 'ClaimsDb' is not configured.");
        }

        services.AddDbContext<ClaimsDbContext>(
            options =>
                options.UseSqlServer(
                    connectionString));

        services.AddScoped<
            IClaimRepository,
            ClaimRepository>();

        services.AddHttpContextAccessor();

        var customerVehicleServiceUrl =
            configuration[
                "Services:CustomerVehicleService"];

        if (string.IsNullOrWhiteSpace(
                customerVehicleServiceUrl))
        {
            throw new InvalidOperationException(
                "Customer Vehicle service URL is not configured.");
        }

        var policyBillingServiceUrl =
            configuration[
                "Services:PolicyBillingService"];


        if (string.IsNullOrWhiteSpace(
                policyBillingServiceUrl))
        {
            throw new InvalidOperationException(
                "Policy Billing service URL is not configured.");
        }
        var identityServiceUrl =
    configuration[
        "Services:IdentityService"];

        if (string.IsNullOrWhiteSpace(
                identityServiceUrl))
        {
            throw new InvalidOperationException(
                "Identity service URL is not configured.");
        }
        services.AddHttpClient<
            ICustomerVehicleClient,
            CustomerVehicleClient>(
            client =>
            {
                client.BaseAddress =
                    new Uri(
                        customerVehicleServiceUrl);

                client.Timeout =
                    TimeSpan.FromSeconds(15);
            });

        services.AddHttpClient<
            IPolicyBillingClient,
            PolicyBillingClient>(
            client =>
            {
                client.BaseAddress =
                    new Uri(
                        policyBillingServiceUrl);

                client.Timeout =
                    TimeSpan.FromSeconds(15);
            });

        services.AddHttpClient<
            IIdentityClient,
            IdentityClient>(
            client =>
            {
                client.BaseAddress =
                    new Uri(
                        identityServiceUrl);

                client.Timeout =
                    TimeSpan.FromSeconds(15);
            });
                return services;
            }
}

using ClaimsService.Application.Interfaces.Repositories;
using ClaimsService.Domain.Entities;
using ClaimsService.Domain.Enums;
using ClaimsService.Infrastructure.Persistence;
using Microsoft.EntityFrameworkCore;

namespace ClaimsService.Infrastructure.Repositories;

public class ClaimRepository : IClaimRepository
{
    private readonly ClaimsDbContext _dbContext;

    public ClaimRepository(
        ClaimsDbContext dbContext)
    {
        _dbContext = dbContext;
    }

    public async Task AddAsync(
        Claim claim)
    {
        ArgumentNullException.ThrowIfNull(
            claim);

        await _dbContext.Claims.AddAsync(
            claim);
    }

    public async Task<Claim?> GetByIdAsync(
        Guid claimId)
    {
        return await _dbContext.Claims
            .FirstOrDefaultAsync(
                claim =>
                    claim.Id == claimId);
    }

    public async Task<IReadOnlyList<Claim>>
        GetByCustomerProfileIdAsync(
            Guid customerProfileId)
    {
        return await _dbContext.Claims
            .AsNoTracking()
            .Where(
                claim =>
                    claim.CustomerProfileId ==
                    customerProfileId)
            .OrderByDescending(
                claim =>
                    claim.SubmittedAtUtc)
            .ToListAsync();
    }

    public async Task<IReadOnlyList<Claim>>
        GetAllAsync()
    {
        return await _dbContext.Claims
            .AsNoTracking()
            .OrderByDescending(
                claim =>
                    claim.SubmittedAtUtc)
            .ToListAsync();
    }

    public async Task<IReadOnlyList<Claim>>
        GetByStatusesAsync(
            IReadOnlyCollection<ClaimStatus> statuses)
    {
        ArgumentNullException.ThrowIfNull(
            statuses);

        if (statuses.Count == 0)
        {
            return [];
        }

        return await _dbContext.Claims
            .AsNoTracking()
            .Where(
                claim =>
                    statuses.Contains(
                        claim.Status))
            .OrderBy(
                claim =>
                    claim.SubmittedAtUtc)
            .ToListAsync();
    }

    public async Task<IReadOnlyList<Claim>>
        GetUnassignedAsync()
    {
        return await _dbContext.Claims
            .AsNoTracking()
            .Where(
                claim =>
                    claim.Status ==
                        ClaimStatus.Submitted &&
                    claim.AssignedClaimsOfficerIdentityUserId ==
                        null)
            .OrderBy(
                claim =>
                    claim.SubmittedAtUtc)
            .ToListAsync();
    }

    public async Task<IReadOnlyList<Claim>>
        GetByAssignedClaimsOfficerAsync(
            string claimsOfficerIdentityUserId)
    {
        if (string.IsNullOrWhiteSpace(
                claimsOfficerIdentityUserId))
        {
            return [];
        }

        return await _dbContext.Claims
            .AsNoTracking()
            .Where(
                claim =>
                    claim.AssignedClaimsOfficerIdentityUserId ==
                    claimsOfficerIdentityUserId)
            .OrderByDescending(
                claim =>
                    claim.AssignedAtUtc)
            .ToListAsync();
    }

    public async Task<int>
        GetActionableCountByClaimsOfficerAsync(
            string claimsOfficerIdentityUserId)
    {
        if (string.IsNullOrWhiteSpace(
                claimsOfficerIdentityUserId))
        {
            return 0;
        }

        return await _dbContext.Claims
            .AsNoTracking()
            .CountAsync(
                claim =>
                    claim.AssignedClaimsOfficerIdentityUserId ==
                        claimsOfficerIdentityUserId &&
                    (
                        claim.Status ==
                            ClaimStatus.Submitted ||
                        claim.Status ==
                            ClaimStatus.InReview ||
                        claim.Status ==
                            ClaimStatus.Approved ||
                        claim.Status ==
                            ClaimStatus.Settled
                    ));
    }

    public async Task<bool> ClaimNumberExistsAsync(
        string claimNumber)
    {
        if (string.IsNullOrWhiteSpace(
                claimNumber))
        {
            return false;
        }

        return await _dbContext.Claims
            .AsNoTracking()
            .AnyAsync(
                claim =>
                    claim.ClaimNumber ==
                    claimNumber);
    }

    public async Task SaveChangesAsync()
    {
        await _dbContext.SaveChangesAsync();
    }
}
