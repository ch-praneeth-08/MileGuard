using PolicyBillingService.Application.Interfaces.Documents;

namespace PolicyBillingService.Infrastructure.Documents;

public class LocalPolicyDocumentStorage
    : IPolicyDocumentStorage
{
    private readonly string _storagePath;

    public LocalPolicyDocumentStorage(
        string storagePath)
    {
        _storagePath =
            storagePath;

        Directory.CreateDirectory(
            _storagePath);
    }

    public async Task<string> SaveAsync(
        string fileName,
        byte[] content)
    {
        ArgumentException.ThrowIfNullOrWhiteSpace(
            fileName);

        ArgumentNullException.ThrowIfNull(
            content);

        var storedFileName =
            $"{Guid.NewGuid():N}.pdf";

        var fullPath =
            Path.Combine(
                _storagePath,
                storedFileName);

        await File.WriteAllBytesAsync(
            fullPath,
            content);

        return storedFileName;
    }

    public async Task<byte[]?> ReadAsync(
        string storedFileName)
    {
        if (string.IsNullOrWhiteSpace(
                storedFileName))
        {
            return null;
        }

        var safeFileName =
            Path.GetFileName(
                storedFileName);

        var fullPath =
            Path.Combine(
                _storagePath,
                safeFileName);

        if (!File.Exists(fullPath))
        {
            return null;
        }

        return await File.ReadAllBytesAsync(
            fullPath);
    }
}

    public async Task<PolicyListResult>
        GetCustomerPoliciesAsync(
            Guid customerProfileId,
            string agentIdentityUserId)
    {
        if (customerProfileId == Guid.Empty)
        {
            return PolicyListResult.MissingCustomer(
                "Customer profile was not found.");
        }

        if (string.IsNullOrWhiteSpace(
                agentIdentityUserId))
        {
            throw new ArgumentException(
                "Agent Identity User ID is required.",
                nameof(agentIdentityUserId));
        }

        var assignmentResult =
            await _customerVehicleClient
                .CheckAgentAssignmentAsync(
                    customerProfileId,
                    agentIdentityUserId);

        if (assignmentResult.DependencyUnavailable)
        {
            return PolicyListResult.Unavailable(
                assignmentResult.Message ??
                "Customer assignment information is temporarily unavailable.");
        }

        if (assignmentResult.NotFound)
        {
            return PolicyListResult.MissingCustomer(
                assignmentResult.Message ??
                "Customer profile was not found.");
        }

        if (!assignmentResult.Succeeded ||
            !assignmentResult.IsAssigned)
        {
            return PolicyListResult.ForbiddenResult(
                "Customer is not assigned to the current Agent.");
        }

        var policies =
            await _policyRepository
                .GetByCustomerProfileIdAsync(
                    customerProfileId);

        var policyDtos =
            new List<PolicyListItemDto>();

        foreach (var policy in policies)
        {
            if (policy.Snapshot is null)
            {
                throw new InvalidOperationException(
                    "Policy does not contain its immutable snapshot.");
            }

            policyDtos.Add(
                new PolicyListItemDto
                {
                    PolicyId =
                        policy.Id,

                    PolicyNumber =
                        policy.PolicyNumber,

                    VehicleId =
                        policy.VehicleId,

                    VehicleDescription =
                        $"{policy.Snapshot.VehicleMake} {policy.Snapshot.VehicleModel}",

                    PlanName =
                        policy.Snapshot.PlanName,

                    AnnualPremium =
                        policy.Snapshot.AnnualPremium,

                    EffectiveDate =
                        policy.EffectiveDate,

                    ExpirationDate =
                        policy.ExpirationDate,

                    EffectiveStatus =
                        GetPolicyStatus(
                            policy),

                    IssuedAtUtc =
                        policy.IssuedAtUtc,

                    CancellationEffectiveAtUtc =
                        policy.CancellationEffectiveAtUtc
                });
        }

        return PolicyListResult.Success(
            policyDtos);
    }
    public async Task<AgentPolicyRenewalResult> GetRenewalAsync(
    Guid policyId,
    string agentIdentityUserId)
    {
        if (policyId == Guid.Empty)
        {
            return AgentPolicyRenewalResult.Missing(
                "Policy was not found.");
        }

        if (string.IsNullOrWhiteSpace(
                agentIdentityUserId))
        {
            throw new ArgumentException(
                "Agent Identity User ID is required.",
                nameof(agentIdentityUserId));
        }

        var policy =
            await _policyRepository.GetByIdAsync(
                policyId);

        if (policy is null)
        {
            return AgentPolicyRenewalResult.Missing(
                "Policy was not found.");
        }

        var assignmentResult =
            await _customerVehicleClient
                .CheckAgentAssignmentAsync(
                    policy.CustomerProfileId,
                    agentIdentityUserId);

        if (assignmentResult.DependencyUnavailable)
        {
            return AgentPolicyRenewalResult.Unavailable(
                assignmentResult.Message ??
                "Customer assignment information is temporarily unavailable.");
        }

        if (assignmentResult.NotFound)
        {
            return AgentPolicyRenewalResult.Missing(
                assignmentResult.Message ??
                "Customer profile was not found.");
        }

        if (!assignmentResult.Succeeded ||
            !assignmentResult.IsAssigned)
        {
            return AgentPolicyRenewalResult.AccessDenied(
                "Customer is not assigned to the current Agent.");
        }

        var renewal =
            await _renewalRepository
                .GetBySourcePolicyIdAsync(
                    policy.Id);

        var today =
            DateOnly.FromDateTime(
                DateTime.UtcNow);

        var renewalWindowStarts =
            policy.ExpirationDate.AddDays(-30);

        var renewalEffectiveDate =
            policy.ExpirationDate;

        var renewalEligible =
            !policy.CancellationEffectiveAtUtc.HasValue &&
            !policy.SuccessorPolicyId.HasValue &&
            today >= policy.EffectiveDate &&
            today < policy.ExpirationDate &&
            today >= renewalWindowStarts;

        string renewalStatus;

        if (policy.CancellationEffectiveAtUtc.HasValue)
        {
            renewalStatus =
                "Cancelled";
        }
        else if (policy.SuccessorPolicyId.HasValue)
        {
            renewalStatus =
                "PolicyIssued";
        }
        else if (renewal?.RenewedPolicyId is not null)
        {
            renewalStatus =
                "PolicyIssued";
        }
        else if (renewal?.RenewalQuoteId is not null)
        {
            renewalStatus =
                "QuoteInProgress";
        }
        else if (renewal is not null)
        {
            renewalStatus =
                "Started";
        }
        else if (today < policy.EffectiveDate)
        {
            renewalStatus =
                "NotRenewable";
        }
        else if (today >= policy.ExpirationDate)
        {
            renewalStatus =
                "NotRenewable";
        }
        else if (today < renewalWindowStarts)
        {
            renewalStatus =
                "TooEarly";
        }
        else
        {
            renewalStatus =
                "Eligible";
        }

        return AgentPolicyRenewalResult.Success(
            new AgentPolicyRenewalDto
            {
                PolicyId =
                    policy.Id,

                PolicyNumber =
                    policy.PolicyNumber,

                RenewalEligible =
                    renewalEligible,

                RenewalWindowStarts =
                    renewalWindowStarts,

                RenewalEffectiveDate =
                    renewalEffectiveDate,

                RenewalStatus =
                    renewalStatus,

                RenewalId =
                    renewal?.Id,

                RenewalQuoteId =
                    renewal?.RenewalQuoteId,

                RenewedPolicyId =
                    renewal?.RenewedPolicyId
            });
    }

    public async Task<PolicyPaymentsResult>
        GetPaymentsAsync(
            Guid policyId,
            string agentIdentityUserId)
    {
        if (policyId == Guid.Empty)
        {
            return PolicyPaymentsResult.Missing(
                "Policy was not found.");
        }

        if (string.IsNullOrWhiteSpace(
                agentIdentityUserId))
        {
            throw new ArgumentException(
                "Agent Identity User ID is required.",
                nameof(agentIdentityUserId));
        }

        var policy =
            await _policyRepository
                .GetByIdAsync(
                    policyId);

        if (policy is null)
        {
            return PolicyPaymentsResult.Missing(
                "Policy was not found.");
        }

        var assignmentResult =
            await _customerVehicleClient
                .CheckAgentAssignmentAsync(
                    policy.CustomerProfileId,
                    agentIdentityUserId);

        if (assignmentResult.DependencyUnavailable)
        {
            return PolicyPaymentsResult.Unavailable(
                assignmentResult.Message ??
                "Customer assignment information is temporarily unavailable.");
        }

        if (assignmentResult.NotFound)
        {
            return PolicyPaymentsResult.Missing(
                assignmentResult.Message ??
                "Customer profile was not found.");
        }

        if (!assignmentResult.Succeeded ||
            !assignmentResult.IsAssigned)
        {
            return PolicyPaymentsResult.AccessDenied(
                "Customer is not assigned to the current Agent.");
        }

        if (policy.PaymentSchedule is null)
        {
            throw new InvalidOperationException(
                "Policy does not contain a payment schedule.");
        }

        var installments =
            policy.PaymentSchedule
                .Installments
                .OrderBy(
                    installment =>
                        installment.InstallmentNumber)
                .ToList();

        var installmentDtos =
            new List<PolicyInstallmentDto>();

        foreach (var installment in installments)
        {
            installmentDtos.Add(
                new PolicyInstallmentDto
                {
                    InstallmentId =
                        installment.Id,

                    InstallmentNumber =
                        installment.InstallmentNumber,

                    Amount =
                        installment.Amount,

                    DueDate =
                        installment.DueDate,

                    Status =
                        GetInstallmentStatus(
                            installment),

                    PaidAtUtc =
                        installment.PaidAtUtc,

                    CanPayNow =
                        false
                });
        }

        var totalPaid =
            installments
                .Where(
                    installment =>
                        installment.PaidAtUtc.HasValue)
                .Sum(
                    installment =>
                        installment.Amount);

        var remainingAmount =
            installments
                .Where(
                    installment =>
                        installment.PaidAtUtc is null &&
                    installment.ContractStatus ==
                        InstallmentContractStatus.Payable)
                .Sum(
                    installment =>
                        installment.Amount);

        var paymentsDto =
            new PolicyPaymentsDto
            {
                PolicyId =
                    policy.Id,

                PolicyNumber =
                    policy.PolicyNumber,

                AnnualPremium =
                    policy.PaymentSchedule
                        .AnnualPremium,

                PaymentFrequency =
                    policy.PaymentSchedule
                        .Frequency
                        .ToString(),

                TotalPaid =
                    totalPaid,

                RemainingAmount =
                    remainingAmount,

                Installments =
                    installmentDtos
            };

        return PolicyPaymentsResult.Success(
            paymentsDto);
    }

    private static string GetPolicyStatus(
        Policy policy)
    {
        var nowUtc =
            DateTime.UtcNow;

        if (policy.CancellationEffectiveAtUtc.HasValue &&
            nowUtc >=
                policy.CancellationEffectiveAtUtc.Value)
        {
            return "Cancelled";
        }

        var today =
            DateOnly.FromDateTime(
                nowUtc);

        if (today < policy.EffectiveDate)
        {
            return "Issued";
        }

        if (today >= policy.ExpirationDate)
        {
            return "Expired";
        }

        return "Active";
    }

    private static string GetInstallmentStatus(
        Installment installment)
    {
        if (installment.PaidAtUtc.HasValue)
        {
            return "Paid";
        }

        if (installment.ContractStatus ==
            InstallmentContractStatus.NoLongerDue)
        {
            return "NoLongerDue";
        }

        var today =
            DateOnly.FromDateTime(
                DateTime.UtcNow);

        if (today > installment.DueDate)
        {
            return "Overdue";
        }

        if (today == installment.DueDate)
        {
            return "Due";
        }

        return "Upcoming";
    }
}

{
    private readonly IPurchaseRepository _purchaseRepository;
    private readonly IPolicyIssuanceRepository _issuanceRepository;

    public async Task<FirstPaymentResult> PayAsync(
        Guid purchaseId,
        string customerIdentityUserId,
        MakePaymentRequest request)
    {
        ArgumentNullException.ThrowIfNull(request);

        if (purchaseId == Guid.Empty)
        {
            return FirstPaymentResult.Missing(
                "Purchase was not found.");
        }

        if (string.IsNullOrWhiteSpace(customerIdentityUserId))
        {
            throw new ArgumentException(
                "Customer Identity User ID is required.",
                nameof(customerIdentityUserId));
        }

        if (!Enum.IsDefined(
                typeof(PaymentMethod),
                request.Method))
        {
            return FirstPaymentResult.Invalid(
                "Payment method is invalid.");
        }

        var purchase =
            await _purchaseRepository.GetByIdForUpdateAsync(
                purchaseId);

        if (purchase is null)
        {
            return FirstPaymentResult.Missing(
                "Purchase was not found.");
        }

        if (purchase.IssuedPolicyId.HasValue)
        {
            return FirstPaymentResult.Invalid(
                "A Policy has already been issued for this Purchase.");
        }

        var existingPolicy =
            await _policyRepository
                .GetByUnderwritingApplicationIdAsync(
                    purchase.UnderwritingApplicationId);

        if (existingPolicy is not null)
        {
            return FirstPaymentResult.Invalid(
                "A Policy has already been issued for this Underwriting application.");
        }

        var customerResult =
            await _customerVehicleClient.GetCustomerProfileAsync(
                customerIdentityUserId);

        if (customerResult.DependencyUnavailable)
        {
            return FirstPaymentResult.Unavailable(
                customerResult.Message ??
                "Customer information is temporarily unavailable.");
        }

        if (!customerResult.Succeeded ||
            customerResult.Customer is null)
        {
            return FirstPaymentResult.Missing(
                "Customer profile was not found.");
        }

        if (purchase.CustomerProfileId !=
            customerResult.Customer.CustomerProfileId)
        {
            return FirstPaymentResult.AccessDenied(
                "Purchase does not belong to the current Customer.");
        }

        var offerResult =
            await _underwritingClient.GetApprovedOfferAsync(
                purchase.UnderwritingApplicationId);

        if (offerResult.DependencyUnavailable)
        {
            return FirstPaymentResult.Unavailable(
                offerResult.Message ??
                "Underwriting information is temporarily unavailable.");
        }

        if (offerResult.NotFound)
        {
            return FirstPaymentResult.Missing(
                offerResult.Message ??
                "Underwriting application was not found.");
        }

        if (!offerResult.Succeeded ||
            offerResult.Offer is null)
        {
            return FirstPaymentResult.Invalid(
                offerResult.Message ??
                "Underwriting offer is no longer available for purchase.");
        }

        var offer = offerResult.Offer;

        if (offer.CustomerProfileId !=
                purchase.CustomerProfileId ||
            offer.VehicleId !=
                purchase.VehicleId)
        {
            return FirstPaymentResult.Invalid(
                "Purchase no longer matches the approved Underwriting offer.");
        }

        if (offer.AnnualPremium !=
            purchase.AnnualPremium)
        {
            return FirstPaymentResult.Invalid(
                "Purchase premium no longer matches the approved Underwriting offer.");
        }

        /*
         * =====================================================
         * RENEWAL CONTEXT
         * =====================================================
         */

        var purchaseHasAnyRenewalContext =
            purchase.RenewalId.HasValue ||
            purchase.SourcePolicyId.HasValue;

        var purchaseHasCompleteRenewalContext =
            purchase.RenewalId.HasValue &&
            purchase.SourcePolicyId.HasValue;

        if (purchaseHasAnyRenewalContext &&
            !purchaseHasCompleteRenewalContext)
        {
            return FirstPaymentResult.Invalid(
                "Purchase contains incomplete Renewal information.");
        }

        var offerHasAnyRenewalContext =
            offer.RenewalId.HasValue ||
            offer.SourcePolicyId.HasValue ||
            offer.RenewalEffectiveDate.HasValue;

        var offerHasCompleteRenewalContext =
            offer.RenewalId.HasValue &&
            offer.SourcePolicyId.HasValue &&
            offer.RenewalEffectiveDate.HasValue;

        if (offerHasAnyRenewalContext &&
            !offerHasCompleteRenewalContext)
        {
            return FirstPaymentResult.Invalid(
                "Approved Underwriting offer contains incomplete Renewal information.");
        }

        var isRenewal =
            purchaseHasCompleteRenewalContext;

        if (isRenewal != offerHasCompleteRenewalContext)
        {
            return FirstPaymentResult.Invalid(
                "Purchase Renewal information does not match the approved Underwriting offer.");
        }

        Policy? sourcePolicy = null;
        Renewal? renewal = null;

        if (isRenewal)
        {
            var renewalId =
                purchase.RenewalId!.Value;

            var sourcePolicyId =
                purchase.SourcePolicyId!.Value;

            if (renewalId == Guid.Empty)
            {
                return FirstPaymentResult.Invalid(
                    "Purchase contains an invalid Renewal ID.");
            }

            if (sourcePolicyId == Guid.Empty)
            {
                return FirstPaymentResult.Invalid(
                    "Purchase contains an invalid Source Policy ID.");
            }

            if (offer.RenewalId != purchase.RenewalId)
            {
                return FirstPaymentResult.Invalid(
                    "Purchase Renewal ID does not match the approved Underwriting offer.");
            }

            if (offer.SourcePolicyId !=
                purchase.SourcePolicyId)
            {
                return FirstPaymentResult.Invalid(
                    "Purchase Source Policy does not match the approved Underwriting offer.");
            }

            if (offer.RenewalEffectiveDate!.Value !=
                purchase.EffectiveDate)
            {
                return FirstPaymentResult.Invalid(
                    "Purchase Effective Date does not match the approved Renewal Effective Date.");
            }

            sourcePolicy =
                await _issuanceRepository.GetPolicyForUpdateAsync(
                    sourcePolicyId);

            if (sourcePolicy is null)
            {
                return FirstPaymentResult.Missing(
                    "Source Policy for this Renewal was not found.");
            }

            if (sourcePolicy.CustomerProfileId !=
                purchase.CustomerProfileId)
            {
                return FirstPaymentResult.Invalid(
                    "Renewal Customer does not match the Source Policy.");
            }

            if (sourcePolicy.VehicleId !=
                purchase.VehicleId)
            {
                return FirstPaymentResult.Invalid(
                    "Renewal Vehicle does not match the Source Policy.");
            }

            if (sourcePolicy.CancellationEffectiveAtUtc.HasValue)
            {
                return FirstPaymentResult.Invalid(
                    "A cancelled Source Policy cannot be renewed.");
            }

            if (sourcePolicy.SuccessorPolicyId.HasValue)
            {
                return FirstPaymentResult.Invalid(
                    "The Source Policy already has a successor Policy.");
            }

            if (sourcePolicy.ExpirationDate !=
                purchase.EffectiveDate)
            {
                return FirstPaymentResult.Invalid(
                    "Renewal Effective Date must match the Source Policy Expiration Date.");
            }

            renewal =
                await _issuanceRepository.GetRenewalForUpdateAsync(
                    renewalId);

            if (renewal is null)
            {
                return FirstPaymentResult.Missing(
                    "Renewal record was not found.");
            }

            if (renewal.SourcePolicyId !=
                sourcePolicy.Id)
            {
                return FirstPaymentResult.Invalid(
                    "Renewal does not match the Source Policy.");
            }

            if (renewal.CustomerProfileId !=
                purchase.CustomerProfileId)
            {
                return FirstPaymentResult.Invalid(
                    "Renewal Customer does not match the Purchase.");
            }

            if (renewal.VehicleId !=
                purchase.VehicleId)
            {
                return FirstPaymentResult.Invalid(
                    "Renewal Vehicle does not match the Purchase.");
            }

            if (!renewal.RenewalQuoteId.HasValue)
            {
                return FirstPaymentResult.Invalid(
                    "Renewal does not contain a Renewal Quote.");
            }

            if (renewal.RenewedPolicyId.HasValue)
            {
                return FirstPaymentResult.Invalid(
                    "A successor Policy has already been issued for this Renewal.");
            }
        }

        /*
         * =====================================================
         * POLICY TERM
         * =====================================================
         */

        var expirationDate =
            purchase.EffectiveDate.AddYears(1);

        var hasOverlap =
            await _policyRepository.HasOverlappingPolicyAsync(
                purchase.VehicleId,
                purchase.EffectiveDate,
                expirationDate);

        if (hasOverlap)
        {
            return FirstPaymentResult.Invalid(
                "The Policy term overlaps an existing Policy for this Vehicle.");
        }

        /*
         * =====================================================
         * PAYMENT SCHEDULE
         * =====================================================
         */

        var schedule =
            _scheduleGenerator.Generate(
                purchase.AnnualPremium,
                purchase.EffectiveDate,
                purchase.PaymentFrequency);

        if (schedule.Installments.Count == 0)
        {
            throw new InvalidOperationException(
                "Payment schedule did not generate any installments.");
        }

        var amountDueNow =
            schedule.Installments[0].Amount;

        var attemptedAtUtc =
            DateTime.UtcNow;

        var paymentReference =
            $"PAY-{Guid.NewGuid():N}"
                .ToUpperInvariant();

        /*
         * =====================================================
         * PAYMENT
         * =====================================================
         */

        var simulation =
            _paymentProcessor.Process(
                request.Method,
                request.TestScenario);

        if (!simulation.Succeeded)
        {
            var failedTransaction =
                new PaymentTransaction
                {
                    Id = Guid.NewGuid(),
                    PaymentReference = paymentReference,
                    PurchaseId = purchase.Id,
                    InstallmentId = null,
                    Amount = amountDueNow,
                    Method = request.Method,

    public InstallmentPaymentService(

        _paymentRepository =

        _customerVehicleClient =

    }

    public async Task<InstallmentPaymentResult> PayAsync(
        Guid installmentId,
        string customerIdentityUserId,
        MakePaymentRequest request)
    {
        ArgumentNullException.ThrowIfNull(
            request);

        if (installmentId == Guid.Empty)
        {
            return InstallmentPaymentResult.Missing(
                "Installment was not found.");
        }

        if (string.IsNullOrWhiteSpace(
                customerIdentityUserId))
        {
            throw new ArgumentException(
                "Customer Identity User ID is required.",
                nameof(customerIdentityUserId));
        }

        if (!Enum.IsDefined(
                typeof(PaymentMethod),
                request.Method))
        {
            return InstallmentPaymentResult.Invalid(
                "Payment method is invalid.");
        }

        var customerResult =
            await _customerVehicleClient
                .GetCustomerProfileAsync(
                    customerIdentityUserId);

        if (customerResult.DependencyUnavailable)
        {
            return InstallmentPaymentResult.Unavailable(
                customerResult.Message ??
                "Customer information is temporarily unavailable.");
        }

        if (!customerResult.Succeeded ||
            customerResult.Customer is null)
        {
            return InstallmentPaymentResult.Missing(
                "Customer profile was not found.");
        }

        var installment =
            await _paymentRepository
                .GetInstallmentForUpdateAsync(
                    installmentId);

        if (installment is null)
        {
            return InstallmentPaymentResult.Missing(
                "Installment was not found.");
        }

        var schedule =
            installment.PaymentSchedule;

        var policy =
            schedule.Policy;

        if (policy.CustomerProfileId !=
            customerResult.Customer.CustomerProfileId)
        {
            return InstallmentPaymentResult.AccessDenied(
                "Installment does not belong to the current Customer.");
        }

        if (policy.CancellationEffectiveAtUtc.HasValue &&
            DateTime.UtcNow >=
                policy.CancellationEffectiveAtUtc.Value)
        {
            return InstallmentPaymentResult.Invalid(
                "Payments cannot be made for a cancelled Policy.");
        }

        var today =
            DateOnly.FromDateTime(
                DateTime.UtcNow);

        if (today >= policy.ExpirationDate)
        {
            return InstallmentPaymentResult.Invalid(
                "Payments cannot be made for an expired Policy.");
        }

        if (installment.PaidAtUtc.HasValue)
        {
            return InstallmentPaymentResult.Invalid(
                "Installment has already been paid.");
        }

        if (installment.ContractStatus !=
            InstallmentContractStatus.Payable)
        {
            return InstallmentPaymentResult.Invalid(
                "Installment is no longer due.");
        }

        var installments =
            await _paymentRepository
                .GetScheduleInstallmentsAsync(
                    schedule.Id);

        var earliestOutstanding =
            installments
                .FirstOrDefault(
                    candidate =>
                        candidate.PaidAtUtc is null &&
                        candidate.ContractStatus ==
                            InstallmentContractStatus.Payable);

        if (earliestOutstanding is null)
        {
            return InstallmentPaymentResult.Invalid(
                "There are no outstanding installments.");
        }

        if (earliestOutstanding.Id !=
            installment.Id)
        {
            return InstallmentPaymentResult.Invalid(
                "Only the earliest outstanding installment can be paid.");
        }

        if (today < installment.DueDate)
        {
            return InstallmentPaymentResult.Invalid(
                "Future installments cannot be prepaid.");
        }

        var attemptedAtUtc =
            DateTime.UtcNow;

        var paymentReference =
            $"PAY-{Guid.NewGuid():N}"
                .ToUpperInvariant();

        var simulation =
            _paymentProcessor.Process(
                request.Method,
                request.TestScenario);

        if (!simulation.Succeeded)
        {
            var failedTransaction =
                new PaymentTransaction
                {
                    Id =
                        Guid.NewGuid(),

                    PaymentReference =
                        paymentReference,

                    PurchaseId =
                        null,

                    InstallmentId =
                        installment.Id,

                    Amount =
                        installment.Amount,

                    Method =
                        request.Method,

                    Result =
                        PaymentResult.Failed,

                    AttemptedAtUtc =
                        attemptedAtUtc,

                    FailureCode =
                        simulation.FailureCode,

                    FailureMessage =
                        simulation.FailureMessage
                };

            await _paymentRepository
                .AddTransactionAsync(
                    failedTransaction);

            await _paymentRepository
                .SaveChangesAsync();

            return InstallmentPaymentResult.Failed(
                new InstallmentPaymentDto
                {
                    InstallmentId =
                        installment.Id,

                    InstallmentNumber =
                        installment.InstallmentNumber,

                    PaymentReference =
                        failedTransaction.PaymentReference,

                    PaymentResult =
                        failedTransaction.Result.ToString(),

                    Amount =
                        failedTransaction.Amount,

                    PaymentMethod =
                        failedTransaction.Method.ToString(),

                    AttemptedAtUtc =
                        failedTransaction.AttemptedAtUtc,

                    PaidAtUtc =
                        null,

                    FailureCode =
                        failedTransaction.FailureCode,

                    FailureMessage =
                        failedTransaction.FailureMessage
                });
        }

        var successfulTransaction =
            new PaymentTransaction
            {
                Id =
                    Guid.NewGuid(),

                PaymentReference =
                    paymentReference,

                PurchaseId =
                    null,

                InstallmentId =
                    installment.Id,

                Amount =
                    installment.Amount,

                Method =
                    request.Method,

                Result =
                    PaymentResult.Succeeded,

                AttemptedAtUtc =
                    attemptedAtUtc,

                FailureCode =
                    null,

                FailureMessage =
                    null
            };

        try
        {
            await _paymentRepository.ExecuteInTransactionAsync(
                async () =>
                {
                    installment.PaidAtUtc =
                        attemptedAtUtc;

                    await _paymentRepository
                        .AddTransactionAsync(
                            successfulTransaction);

                    await _paymentRepository
                        .SaveChangesAsync();
                });
        }
        catch (Exception)
        {
            return InstallmentPaymentResult.Invalid(
                "Installment payment state changed. Reload the Policy and try again.");
        }

        return InstallmentPaymentResult.Success(
            new InstallmentPaymentDto
            {
                InstallmentId =
                    installment.Id,

                InstallmentNumber =
                    installment.InstallmentNumber,

                PaymentReference =
                    successfulTransaction.PaymentReference,

                PaymentResult =
                    successfulTransaction.Result.ToString(),

                Amount =
                    successfulTransaction.Amount,

                PaymentMethod =
                    successfulTransaction.Method.ToString(),

                AttemptedAtUtc =
                    successfulTransaction.AttemptedAtUtc,

                PaidAtUtc =
                    installment.PaidAtUtc,

                FailureCode =
                    null,

                FailureMessage =
                    null
            });
    }

using PolicyBillingService.Application.Dtos.Purchases;
using PolicyBillingService.Application.Interfaces;
using PolicyBillingService.Domain.Enums;

namespace PolicyBillingService.Application.Services;

public class PaymentScheduleGenerator
    : IPaymentScheduleGenerator
{
    public PaymentSchedulePreviewDto Generate(
        decimal annualPremium,
        DateOnly effectiveDate,
        PaymentFrequency frequency)
    {
        if (annualPremium <= 0)
        {
            throw new ArgumentOutOfRangeException(
                nameof(annualPremium),
                "Annual Premium must be greater than zero.");
        }

        if (!Enum.IsDefined(
                typeof(PaymentFrequency),
                frequency))
        {
            throw new ArgumentOutOfRangeException(
                nameof(frequency),
                "Payment Frequency is invalid.");
        }

        var installmentCount =
            GetInstallmentCount(
                frequency);

        var monthInterval =
            GetMonthInterval(
                frequency);

        var normalInstallmentAmount =
            decimal.Round(
                annualPremium / installmentCount,
                2,
                MidpointRounding.AwayFromZero);

        var installments =
            new List<ScheduleInstallmentDto>();

        decimal allocatedAmount = 0m;

        for (var index = 0;
             index < installmentCount;
             index++)
        {
            var installmentNumber =
                index + 1;

            var isFinalInstallment =
                installmentNumber ==
                installmentCount;

            var amount =
                isFinalInstallment
                    ? annualPremium -
                      allocatedAmount
                    : normalInstallmentAmount;

            var dueDate =
                installmentNumber == 1
                    ? effectiveDate
                    : effectiveDate.AddMonths(
                        monthInterval * index);

            installments.Add(
                new ScheduleInstallmentDto
                {
                    InstallmentNumber =
                        installmentNumber,

                    Amount =
                        amount,

                    DueDate =
                        dueDate
                });

            allocatedAmount +=
                amount;
        }

        if (installments.Sum(
                installment =>
                    installment.Amount) !=
            annualPremium)
        {
            throw new InvalidOperationException(
                "Generated payment schedule does not equal the Annual Premium.");
        }

        return new PaymentSchedulePreviewDto
        {
            AnnualPremium =
                annualPremium,

            PaymentFrequency =
                frequency.ToString(),

            InstallmentCount =
                installmentCount,

            AmountDueNow =
                installments[0].Amount,

            Installments =
                installments
        };
    }

    private static int GetInstallmentCount(
        PaymentFrequency frequency)
    {
        return frequency switch
        {
            PaymentFrequency.Annual =>
                1,

            PaymentFrequency.SemiAnnual =>
                2,

            PaymentFrequency.Quarterly =>
                4,

            PaymentFrequency.Monthly =>
                12,

            _ =>
                throw new ArgumentOutOfRangeException(
                    nameof(frequency))
        };
    }

    private static int GetMonthInterval(
        PaymentFrequency frequency)
    {
        return frequency switch
        {
            PaymentFrequency.Annual =>
                12,

            PaymentFrequency.SemiAnnual =>
                6,

            PaymentFrequency.Quarterly =>
                3,

            PaymentFrequency.Monthly =>
                1,

            _ =>
                throw new ArgumentOutOfRangeException(
                    nameof(frequency))
        };
    }
}

    Policy policy)
    {

            nowUtc >=

            return "Cancelled";
        }

        var today =
            DateOnly.FromDateTime(
                nowUtc);

        if (today < policy.EffectiveDate)
        {

        if (today < policy.ExpirationDate)

    public async Task<PolicyPaymentsResult> GetPaymentsAsync(
    Guid policyId,
    string customerIdentityUserId)
    {
        if (policyId == Guid.Empty)
        {
            return PolicyPaymentsResult.Missing(
                "Policy was not found.");
        }

        if (string.IsNullOrWhiteSpace(
                customerIdentityUserId))
        {
            throw new ArgumentException(
                "Customer Identity User ID is required.",
                nameof(customerIdentityUserId));
        }

        var customerResult =
            await _customerVehicleClient
                .GetCustomerProfileAsync(
                    customerIdentityUserId);

        if (customerResult.DependencyUnavailable)
        {
            return PolicyPaymentsResult.Unavailable(
                customerResult.Message ??
                "Customer information is temporarily unavailable.");
        }

        if (customerResult.NotFound ||
            !customerResult.Succeeded ||
            customerResult.Customer is null)
        {
            return PolicyPaymentsResult.Missing(
                "Customer profile was not found.");
        }

        var policy =
            await _policyRepository
                .GetByIdAsync(
                    policyId);

        if (policy is null)
        {
            return PolicyPaymentsResult.Missing(
                "Policy was not found.");
        }

        if (policy.CustomerProfileId !=
            customerResult.Customer.CustomerProfileId)
        {
            return PolicyPaymentsResult.AccessDenied(
                "Policy does not belong to the current Customer.");
        }

        if (policy.PaymentSchedule is null)
        {
            throw new InvalidOperationException(
                "Policy does not contain a payment schedule.");
        }

        var installments =
            policy.PaymentSchedule
                .Installments
                .OrderBy(
                    installment =>
                        installment.InstallmentNumber)
                .ToList();

        var earliestOutstandingInstallment =
            installments
                .FirstOrDefault(
                    installment =>
                        installment.PaidAtUtc is null &&
                        installment.ContractStatus ==
                            InstallmentContractStatus.Payable);

        var installmentDtos =
            new List<PolicyInstallmentDto>();

        foreach (var installment in installments)
        {
            string status;

            if (installment.PaidAtUtc.HasValue)
            {
                status =
                    "Paid";
            }
            else if (installment.ContractStatus ==
                     InstallmentContractStatus.NoLongerDue)
            {
                status =
                    "NoLongerDue";
            }
            else
            {
                var today =
                    DateOnly.FromDateTime(
                        DateTime.UtcNow);

                if (today > installment.DueDate)
                {
                    status =
                        "Overdue";
                }
                else if (today == installment.DueDate)
                {
                    status =
                        "Due";
                }
                else
                {
                    status =
                        "Upcoming";
                }
            }

            var canPayNow =
                false;

            if (earliestOutstandingInstallment is not null &&
                earliestOutstandingInstallment.Id ==
                    installment.Id &&
                installment.PaidAtUtc is null &&
                installment.ContractStatus ==
                    InstallmentContractStatus.Payable)
            {
                var today =
                    DateOnly.FromDateTime(
                        DateTime.UtcNow);

                if (today >= installment.DueDate)
                {
                    canPayNow =
                        true;
                }
            }

            installmentDtos.Add(
                new PolicyInstallmentDto
                {
                    InstallmentId =
                        installment.Id,

                    InstallmentNumber =
                        installment.InstallmentNumber,

                    Amount =
                        installment.Amount,

                    DueDate =
                        installment.DueDate,

                    Status =
                        status,

                    PaidAtUtc =
                        installment.PaidAtUtc,

                    CanPayNow =
                        canPayNow
                });
        }

        var totalPaid =
            installments
                .Where(
                    installment =>
                        installment.PaidAtUtc.HasValue)
                .Sum(
                    installment =>
                        installment.Amount);

        var remainingAmount =
            installments
                .Where(
                    installment =>
                        installment.PaidAtUtc is null &&
                        installment.ContractStatus ==
                            InstallmentContractStatus.Payable)
                .Sum(
                    installment =>
                        installment.Amount);

        var payments =
            new PolicyPaymentsDto
            {
                PolicyId =
                    policy.Id,

                PolicyNumber =
                    policy.PolicyNumber,

                AnnualPremium =
                    policy.PaymentSchedule.AnnualPremium,

                PaymentFrequency =
                    policy.PaymentSchedule
                        .Frequency
                        .ToString(),

                TotalPaid =
                    totalPaid,

                RemainingAmount =
                    remainingAmount,

                Installments =
                    installmentDtos
            };

        return PolicyPaymentsResult.Success(
            payments);
    }


    private static string GetInstallmentStatus(
    Installment installment)
    {
        if (installment.PaidAtUtc.HasValue)
        {
            return "Paid";
        }

        if (installment.ContractStatus ==
            InstallmentContractStatus.NoLongerDue)
        {
            return "NoLongerDue";
        }

        var today =
            DateOnly.FromDateTime(
                DateTime.UtcNow);

        if (today > installment.DueDate)
        {
            return "Overdue";
        }

        if (today == installment.DueDate)
        {
            return "Due";
        }

        return "Upcoming";
    }

    public async Task<CancelPolicyResult> CancelAsync(
        Guid policyId,
        string customerIdentityUserId)
    {
        if (policyId == Guid.Empty)
        {
            return CancelPolicyResult.Missing(
                "Policy was not found.");
        }

        if (string.IsNullOrWhiteSpace(
                customerIdentityUserId))
        {
            throw new ArgumentException(
                "Customer Identity User ID is required.",
                nameof(customerIdentityUserId));
        }

        var customerResult =
            await _customerVehicleClient
                .GetCustomerProfileAsync(
                    customerIdentityUserId);

        if (customerResult.DependencyUnavailable)
        {
            return CancelPolicyResult.Unavailable(
                customerResult.Message ??
                "Customer information is temporarily unavailable.");
        }

        if (!customerResult.Succeeded ||
            customerResult.Customer is null)
        {
            return CancelPolicyResult.Missing(
                "Customer profile was not found.");
        }

        var policy =
            await _servicingRepository
                .GetForUpdateAsync(
                    policyId);

        if (policy is null)
        {
            return CancelPolicyResult.Missing(
                "Policy was not found.");
        }

        if (policy.CustomerProfileId !=
            customerResult.Customer.CustomerProfileId)
        {
            return CancelPolicyResult.AccessDenied(
                "Policy does not belong to the current Customer.");
        }

        if (policy.CancellationEffectiveAtUtc.HasValue)
        {
            return CancelPolicyResult.Invalid(
                "Policy has already been cancelled.");
        }

        var nowUtc =
            DateTime.UtcNow;

        var today =
            DateOnly.FromDateTime(
                nowUtc);

        if (today >= policy.ExpirationDate)
        {
            return CancelPolicyResult.Invalid(
                "An expired Policy cannot be cancelled.");
        }

        var cancellationEffectiveAtUtc =
            nowUtc;

        policy.CancellationEffectiveAtUtc =
            cancellationEffectiveAtUtc;

        var noLongerDueCount =
            0;

        foreach (var installment
                 in policy.PaymentSchedule.Installments)
        {
            if (installment.PaidAtUtc.HasValue)
            {
                continue;
            }

            if (installment.ContractStatus !=
                InstallmentContractStatus.Payable)
            {
                continue;
            }

            installment.ContractStatus =
                InstallmentContractStatus.NoLongerDue;

            noLongerDueCount++;
        }

        var history =
            new PolicyStatusHistory
            {
                Id =
                    Guid.NewGuid(),

                PolicyId =
                    policy.Id,

                Action =
                    PolicyHistoryAction.Cancelled,

                OccurredAtUtc =
                    cancellationEffectiveAtUtc,

                PerformedByIdentityUserId =
                    customerIdentityUserId
            };

        await _servicingRepository
            .ExecuteInTransactionAsync(
                async () =>
                {
                    await _servicingRepository
                        .AddStatusHistoryAsync(
                            history);

                    await _servicingRepository
                        .SaveChangesAsync();
                });

        return CancelPolicyResult.Success(
            new CancelPolicyDto
            {
                PolicyId =
                    policy.Id,

                PolicyNumber =
                    policy.PolicyNumber,

                EffectiveStatus =
                    "Cancelled",

                CancellationEffectiveAtUtc =
                    cancellationEffectiveAtUtc,

                NoLongerDueInstallmentCount =
                    noLongerDueCount
            });
    }
}

    public async Task<ConfigurePurchaseResult> ConfigureAsync(
        Guid underwritingApplicationId,
        string customerIdentityUserId,
        ConfigurePurchaseRequest request)
    {
        ArgumentNullException.ThrowIfNull(
            request);

        if (underwritingApplicationId == Guid.Empty)
        {
            return ConfigurePurchaseResult.Missing(
                "Underwriting application was not found.");
        }

        if (string.IsNullOrWhiteSpace(
                customerIdentityUserId))
        {
            throw new ArgumentException(
                "Customer Identity User ID is required.",
                nameof(customerIdentityUserId));
        }

        if (!Enum.IsDefined(
                typeof(PaymentFrequency),
                request.PaymentFrequency))
        {
            return ConfigurePurchaseResult.Invalid(
                "Payment Frequency is invalid.");
        }

        /*
         * =====================================================
         * CUSTOMER
         * =====================================================
         */

        var customerResult =
            await _customerVehicleClient
                .GetCustomerProfileAsync(
                    customerIdentityUserId);

        if (customerResult.DependencyUnavailable)
        {
            return ConfigurePurchaseResult.Unavailable(
                customerResult.Message ??
                "Customer information is temporarily unavailable.");
        }

        if (!customerResult.Succeeded ||
            customerResult.Customer is null)
        {
            return ConfigurePurchaseResult.Missing(
                "Customer profile was not found.");
        }

        /*
         * =====================================================
         * APPROVED UNDERWRITING OFFER
         * =====================================================
         */

        var offerResult =
            await _underwritingClient
                .GetApprovedOfferAsync(
                    underwritingApplicationId);

        if (offerResult.DependencyUnavailable)
        {
            return ConfigurePurchaseResult.Unavailable(
                offerResult.Message ??
                "Underwriting information is temporarily unavailable.");
        }

        if (offerResult.NotFound)
        {
            return ConfigurePurchaseResult.Missing(
                offerResult.Message ??
                "Underwriting application was not found.");
        }

        if (!offerResult.Succeeded ||
            offerResult.Offer is null)
        {
            return ConfigurePurchaseResult.Invalid(
                offerResult.Message ??
                "Underwriting offer is not available for purchase.");
        }

        var offer =
            offerResult.Offer;

        if (offer.CustomerProfileId !=
            customerResult.Customer.CustomerProfileId)
        {
            return ConfigurePurchaseResult.AccessDenied(
                "Underwriting offer does not belong to the current Customer.");
        }

        /*
         * =====================================================
         * VALIDATE RENEWAL CONTEXT
         * =====================================================
         */

        var renewalValidationMessage =
            ValidateRenewalContext(
                offer);

        if (renewalValidationMessage is not null)
        {
            return ConfigurePurchaseResult.Invalid(
                renewalValidationMessage);
        }

        var isRenewal =
            IsRenewalOffer(
                offer);

        DateOnly effectiveDate;

        if (isRenewal)
        {
            /*
             * For a Renewal, PolicyBilling controls the date.
             *
             * The successor Policy must begin exactly when
             * the predecessor Policy expires.
             */

            effectiveDate =
                offer.RenewalEffectiveDate!.Value;

            if (request.EffectiveDate !=
                effectiveDate)
            {
                return ConfigurePurchaseResult.Invalid(
                    $"Renewal Effective Date must be {effectiveDate:yyyy-MM-dd}.");
            }

            var sourcePolicyValidation =
                await ValidateSourcePolicyAsync(
                    offer);

            if (sourcePolicyValidation is not null)
            {
                return ConfigurePurchaseResult.Invalid(
                    sourcePolicyValidation);
            }
        }
        else
        {
            /*
             * Normal new-business purchase.
             */

            var today =
                DateOnly.FromDateTime(
                    DateTime.UtcNow);

            var latestEffectiveDate =
                today.AddDays(30);

            if (request.EffectiveDate < today ||
                request.EffectiveDate >
                    latestEffectiveDate)
            {
                return ConfigurePurchaseResult.Invalid(
                    "Effective Date must be between today and 30 calendar days from today.");
            }

            effectiveDate =
                request.EffectiveDate;
        }

        /*
         * =====================================================
         * POLICY TERM
         * =====================================================
         */

        var expirationDate =
            effectiveDate.AddYears(1);

        /*
         * =====================================================
         * OVERLAP VALIDATION
         * =====================================================
         *
         * HasOverlappingPolicyAsync uses the half-open rule:
         *
         * existing.EffectiveDate < newExpirationDate
         * &&
         * newEffectiveDate < existing.ExpirationDate
         *
         * Therefore:
         *
         * old expires 2027-10-10
         * new starts  2027-10-10
         *
         * is NOT considered an overlap.
         */

        var hasOverlap =
            await _policyRepository
                .HasOverlappingPolicyAsync(
                    offer.VehicleId,
                    effectiveDate,
                    expirationDate);

        if (hasOverlap)
        {
            return ConfigurePurchaseResult.Overlap(
                "The selected Effective Date overlaps an existing Policy term for this Vehicle.");
        }

        /*
         * =====================================================
         * PAYMENT SCHEDULE PREVIEW
         * =====================================================
         */

        var schedule =
            _scheduleGenerator.Generate(
                offer.AnnualPremium,
                effectiveDate,
                request.PaymentFrequency);

        return ConfigurePurchaseResult.Success(
            new ConfigurePurchaseDto
            {
                UnderwritingApplicationId =
                    offer.UnderwritingApplicationId,

                CustomerProfileId =
                    offer.CustomerProfileId,

                VehicleId =
                    offer.VehicleId,

                PlanName =
                    offer.PlanName,

                AnnualPremium =
                    offer.AnnualPremium,

                EffectiveDate =
                    effectiveDate,

                ExpirationDate =
                    expirationDate,

                OfferValidUntilUtc =
                    offer.OfferValidUntilUtc,

                PaymentSchedule =
                    schedule
            });
    }

         * =====================================================

         * =====================================================

        var renewalValidationMessage =

        {
            return AcceptPurchaseResult.Duplicate(
                "The selected Effective Date overlaps an existing Policy term for this Vehicle.");
        }

        /*
         * =====================================================
         * PAYMENT SCHEDULE
         * =====================================================
         */

        var schedule =
            _scheduleGenerator.Generate(
                offer.AnnualPremium,
                effectiveDate,
                request.PaymentFrequency);

        var acceptedAtUtc =
            DateTime.UtcNow;

        /*
         * =====================================================
         * CREATE PURCHASE
         * =====================================================
         */

        var purchase =
            new Purchase
            {
                Id =
                    Guid.NewGuid(),

                UnderwritingApplicationId =
                    offer.UnderwritingApplicationId,

                RenewalId =
                    offer.RenewalId,

                SourcePolicyId =
                    offer.SourcePolicyId,

                CustomerProfileId =
                    offer.CustomerProfileId,

                VehicleId =
                    offer.VehicleId,

                EffectiveDate =
                    effectiveDate,

                PaymentFrequency =
                    request.PaymentFrequency,

                AnnualPremium =
                    offer.AnnualPremium,

                AcceptedAtUtc =
                    acceptedAtUtc,

                IssuedPolicyId =
                    null
            };

        await _purchaseRepository.AddAsync(
            purchase);

        try
        {
            await _purchaseRepository
                .SaveChangesAsync();
        }
        catch
        {
            /*
             * Existing UnderwritingApplicationId uniqueness
             * protects normal and Renewal purchases.
             */

            var concurrentPurchase =
                await _purchaseRepository
                    .GetByUnderwritingApplicationIdAsync(
                        underwritingApplicationId);

            if (concurrentPurchase is not null)
            {
                return AcceptPurchaseResult.Duplicate(
                    "This Underwriting offer has already been accepted.");
            }

            throw;
        }

        return AcceptPurchaseResult.Success(
            new AcceptPurchaseDto
            {
                PurchaseId =
                    purchase.Id,

                UnderwritingApplicationId =
                    purchase.UnderwritingApplicationId,

                CustomerProfileId =
                    purchase.CustomerProfileId,

                VehicleId =
                    purchase.VehicleId,

                PlanName =
                    offer.PlanName,

                AnnualPremium =
                    purchase.AnnualPremium,

                EffectiveDate =
                    purchase.EffectiveDate,

                ExpirationDate =
                    expirationDate,

                PaymentFrequency =
                    purchase.PaymentFrequency
                        .ToString(),

                AcceptedAtUtc =
                    purchase.AcceptedAtUtc,

                AmountDueNow =
                    schedule.AmountDueNow,

                InstallmentCount =
                    schedule.InstallmentCount,

                IssuedPolicyId =
                    purchase.IssuedPolicyId
            });
    }

    /*
     * =========================================================
     * RENEWAL HELPERS
     * =========================================================
     */

    private static bool IsRenewalOffer(
        ApprovedOfferDto offer)
    {
        return offer.RenewalId.HasValue &&
               offer.SourcePolicyId.HasValue &&
               offer.RenewalEffectiveDate.HasValue;
    }

    private static string? ValidateRenewalContext(
        ApprovedOfferDto offer)
    {
        var hasAnyRenewalContext =
            offer.RenewalId.HasValue ||
            offer.SourcePolicyId.HasValue ||
            offer.RenewalEffectiveDate.HasValue;

        if (!hasAnyRenewalContext)
        {
            return null;
        }

        var hasCompleteRenewalContext =
            offer.RenewalId.HasValue &&
            offer.SourcePolicyId.HasValue &&
            offer.RenewalEffectiveDate.HasValue;

        if (!hasCompleteRenewalContext)
        {
            return
                "The approved Underwriting offer contains incomplete Renewal information.";
        }

        if (offer.RenewalId!.Value ==
            Guid.Empty)
        {
            return
                "The approved Underwriting offer contains an invalid Renewal ID.";
        }

        if (offer.SourcePolicyId!.Value ==
            Guid.Empty)
        {
            return
                "The approved Underwriting offer contains an invalid Source Policy ID.";
        }

        if (offer.RenewalEffectiveDate!.Value ==
            default)
        {
            return
                "The approved Underwriting offer contains an invalid Renewal Effective Date.";
        }

        return null;
    }

    private async Task<string?>
        ValidateSourcePolicyAsync(
            ApprovedOfferDto offer)
    {
        if (!IsRenewalOffer(
                offer))
        {
            return null;
        }

        var sourcePolicy =
            await _policyRepository
                .GetByIdAsync(
                    offer.SourcePolicyId!.Value);

        if (sourcePolicy is null)
        {
            return
                "The Source Policy for this Renewal was not found.";
        }

        if (sourcePolicy.CustomerProfileId !=
            offer.CustomerProfileId)
        {
            return
                "The Renewal Customer does not match the Source Policy.";
        }

        if (sourcePolicy.VehicleId !=
            offer.VehicleId)
        {
            return
                "The Renewal Vehicle does not match the Source Policy.";
        }

        if (sourcePolicy.CancellationEffectiveAtUtc.HasValue)
        {
            return
                "A cancelled Source Policy cannot be renewed.";
        }

        if (sourcePolicy.SuccessorPolicyId.HasValue)
        {
            return
                "The Source Policy has already been renewed.";
        }

        if (sourcePolicy.ExpirationDate !=
            offer.RenewalEffectiveDate!.Value)
        {
            return
                "The Renewal Effective Date must match the Source Policy Expiration Date.";
        }

        return null;
    }
}

using PolicyBillingService.Application.Dtos.QuoteRating;
using PolicyBillingService.Application.Dtos.Renewals;
using PolicyBillingService.Application.Interfaces;
using PolicyBillingService.Application.Interfaces.Clients;
using PolicyBillingService.Application.Interfaces.Repositories;
using PolicyBillingService.Domain.Entities;

namespace PolicyBillingService.Application.Services;

public class RenewalService : IRenewalService
{
    private readonly IPolicyRepository
        _policyRepository;

    private readonly IRenewalRepository
        _renewalRepository;

    private readonly ICustomerVehicleClient
        _customerVehicleClient;

    private readonly IQuoteRatingClient
        _quoteRatingClient;

    public RenewalService(
        IPolicyRepository policyRepository,
        IRenewalRepository renewalRepository,
        ICustomerVehicleClient customerVehicleClient,
        IQuoteRatingClient quoteRatingClient)
    {
        _policyRepository =
            policyRepository;

        _renewalRepository =
            renewalRepository;

        _customerVehicleClient =
            customerVehicleClient;

        _quoteRatingClient =
            quoteRatingClient;
    }

    public async Task<StartRenewalResult> StartAsync(
        Guid policyId,
        string agentIdentityUserId)
    {
        if (policyId == Guid.Empty)
        {
            return StartRenewalResult.Missing(
                "Policy was not found.");
        }

        if (string.IsNullOrWhiteSpace(
                agentIdentityUserId))
        {
            throw new ArgumentException(
                "Agent Identity User ID is required.",
                nameof(agentIdentityUserId));
        }

        /*
         * =====================================================
         * LOAD SOURCE POLICY
         * =====================================================
         */

        var policy =
            await _policyRepository
                .GetByIdAsync(
                    policyId);

        if (policy is null)
        {
            return StartRenewalResult.Missing(
                "Policy was not found.");
        }

        /*
         * =====================================================
         * VERIFY AGENT ASSIGNMENT
         * =====================================================
         */

        var assignmentResult =
            await _customerVehicleClient
                .CheckAgentAssignmentAsync(
                    policy.CustomerProfileId,
                    agentIdentityUserId);

        if (assignmentResult.DependencyUnavailable)
        {
            return StartRenewalResult.Unavailable(
                assignmentResult.Message ??
                "Customer assignment information is temporarily unavailable.");
        }

        if (assignmentResult.NotFound)
        {
            return StartRenewalResult.Missing(
                assignmentResult.Message ??
                "Customer profile was not found.");
        }

        if (!assignmentResult.Succeeded ||
            !assignmentResult.IsAssigned)
        {
            return StartRenewalResult.AccessDenied(
                "Customer is not assigned to the current Agent.");
        }

        /*
         * =====================================================
         * SOURCE POLICY ELIGIBILITY
         * =====================================================
         */

        if (policy.CancellationEffectiveAtUtc.HasValue)
        {
            return StartRenewalResult.Invalid(
                "A cancelled Policy cannot be renewed.");
        }

        var today =
            DateOnly.FromDateTime(
                DateTime.UtcNow);

        if (today < policy.EffectiveDate)
        {
            return StartRenewalResult.Invalid(
                "Renewal cannot be started for a Policy that is not yet active.");
        }

        if (today >= policy.ExpirationDate)
        {
            return StartRenewalResult.Invalid(
                "An expired Policy cannot start this renewal workflow.");
        }

        var renewalWindowStarts =
            policy.ExpirationDate
                .AddDays(-30);

        if (today < renewalWindowStarts)
        {
            return StartRenewalResult.Invalid(
                $"Renewal becomes available on {renewalWindowStarts:yyyy-MM-dd}.");
        }

        /*
         * The successor Policy begins exactly when the
         * predecessor Policy expires.
         */
        var renewalEffectiveDate =
            policy.ExpirationDate;

        /*
         * =====================================================
         * FIND OR CREATE RENEWAL
         * =====================================================
         */

        var renewal =
            await _renewalRepository
                .GetBySourcePolicyIdAsync(
                    policy.Id);

        if (renewal is null)
        {
            var newRenewal =
                new Renewal
                {
                    Id =
                        Guid.NewGuid(),

                    SourcePolicyId =
                        policy.Id,

                    CustomerProfileId =
                        policy.CustomerProfileId,

                    VehicleId =
                        policy.VehicleId,

                    StartedAtUtc =
                        DateTime.UtcNow,

                    RenewalQuoteId =
                        null,

                    RenewedPolicyId =
                        null
                };

            try
            {
                await _renewalRepository
                    .AddAsync(
                        newRenewal);

                await _renewalRepository
                    .SaveChangesAsync();

                renewal =
                    newRenewal;
            }
            catch
            {
                /*
                 * Another request may have created the
                 * Renewal concurrently.
                 */

                renewal =
                    await _renewalRepository
                        .GetBySourcePolicyIdAsync(
                            policy.Id);

                if (renewal is null)
                {
                    throw;
                }
            }
        }

        /*
         * Defensive validation.
         *
         * The Renewal found for this Source Policy must
         * describe the same contractual context.
         */

        if (renewal.SourcePolicyId !=
            policy.Id)
        {
            return StartRenewalResult.Invalid(
                "Renewal does not match the Source Policy.");
        }

        if (renewal.CustomerProfileId !=
            policy.CustomerProfileId)
        {
            return StartRenewalResult.Invalid(
                "Renewal Customer does not match the Source Policy.");
        }

        if (renewal.VehicleId !=
            policy.VehicleId)
        {
            return StartRenewalResult.Invalid(
                "Renewal Vehicle does not match the Source Policy.");
        }

        /*
         * =====================================================
         * RENEWAL ALREADY HAS A QUOTE
         * =====================================================
         *
         * If PolicyBilling already persisted the Quote ID,
         * this is an idempotent retry.
         */

        if (renewal.RenewalQuoteId.HasValue)
        {
            return StartRenewalResult.Success(
                MapResult(
                    renewal,
                    policy.PolicyNumber,
                    policy.ExpirationDate));
        }

        /*
         * =====================================================
         * START / RESUME QUOTE IN QUOTERATING
         * =====================================================
         */

        var quoteResult =
            await _quoteRatingClient
                .StartRenewalQuoteAsync(
                    new StartRenewalQuoteRequest
                    {
                        RenewalId =
                            renewal.Id,

                        SourcePolicyId =
                            policy.Id,

                        CustomerProfileId =
                            policy.CustomerProfileId,

                        VehicleId =
                            policy.VehicleId,

                        AgentIdentityUserId =
                            agentIdentityUserId,

                        RenewalEffectiveDate =
                            renewalEffectiveDate
                    });

        if (quoteResult.DependencyUnavailable)
        {
            return StartRenewalResult.Unavailable(
                quoteResult.Message ??
                "QuoteRatingService is temporarily unavailable.");
        }

        if (quoteResult.Conflict)
        {
            return StartRenewalResult.Duplicate(
                quoteResult.Message ??
                "Renewal Quote could not be started.");
        }

        if (!quoteResult.Succeeded ||
            quoteResult.Quote is null)
        {
            return StartRenewalResult.Unavailable(
                quoteResult.Message ??
                "Renewal Quote could not be started.");
        }

        var quote =
            quoteResult.Quote;

        /*
         * =====================================================
         * VALIDATE QUOTERATING RESPONSE
         * =====================================================
         */

        if (quote.QuoteId == Guid.Empty)
        {
            return StartRenewalResult.Unavailable(
                "QuoteRatingService returned an invalid Quote ID.");
        }

        if (quote.RenewalId !=
            renewal.Id)
        {
            return StartRenewalResult.Unavailable(
                "QuoteRatingService returned a Quote for a different Renewal.");
        }

        if (quote.SourcePolicyId !=
            policy.Id)
        {
            return StartRenewalResult.Unavailable(
                "QuoteRatingService returned a Quote for a different Source Policy.");
        }

        if (quote.CustomerProfileId !=
            policy.CustomerProfileId)
        {
            return StartRenewalResult.Unavailable(
                "QuoteRatingService returned a Quote for a different Customer.");
        }

        if (quote.VehicleId !=
            policy.VehicleId)
        {
            return StartRenewalResult.Unavailable(
                "QuoteRatingService returned a Quote for a different Vehicle.");
        }

        if (quote.RenewalEffectiveDate !=
            renewalEffectiveDate)
        {
            return StartRenewalResult.Unavailable(
                "QuoteRatingService returned an unexpected Renewal Effective Date.");
        }

        /*
         * =====================================================
         * PERSIST RENEWAL QUOTE ID
         * =====================================================
         *
         * Get a tracked Renewal because the earlier lookup
         * may be AsNoTracking().
         */

        var trackedRenewal =
            await _renewalRepository
                .GetByIdForUpdateAsync(
                    renewal.Id);

        if (trackedRenewal is null)
        {
            throw new InvalidOperationException(
                "Renewal could not be reloaded after Quote creation.");
        }

        /*
         * Another request may have completed this update while
         * QuoteRating was processing the request.
         */

        if (trackedRenewal.RenewalQuoteId.HasValue)
        {
            if (trackedRenewal.RenewalQuoteId.Value !=
                quote.QuoteId)
            {
                return StartRenewalResult.Invalid(
                    "Renewal is already associated with a different Quote.");
            }

            return StartRenewalResult.Success(
                MapResult(
                    trackedRenewal,
                    policy.PolicyNumber,
                    policy.ExpirationDate));
        }

        trackedRenewal.RenewalQuoteId =
            quote.QuoteId;

        await _renewalRepository
            .SaveChangesAsync();

        /*
         * =====================================================
         * SUCCESS
         * =====================================================
         */

        return StartRenewalResult.Success(
            MapResult(
                trackedRenewal,
                policy.PolicyNumber,
                policy.ExpirationDate));
    }

    private static StartRenewalDto MapResult(
        Renewal renewal,
        string sourcePolicyNumber,
        DateOnly sourcePolicyExpirationDate)
    {
        return new StartRenewalDto
        {
            RenewalId =
                renewal.Id,

            SourcePolicyId =
                renewal.SourcePolicyId,

            SourcePolicyNumber =
                sourcePolicyNumber,

            CustomerProfileId =
                renewal.CustomerProfileId,

            VehicleId =
                renewal.VehicleId,

            StartedAtUtc =
                renewal.StartedAtUtc,

            CurrentPolicyExpirationDate =
                sourcePolicyExpirationDate,

            RenewalEffectiveDate =
                sourcePolicyExpirationDate,

            RenewalQuoteId =
                renewal.RenewalQuoteId
        };
    }
}

using PolicyBillingService.Application.Dtos.Payments;
using PolicyBillingService.Application.Interfaces;
using PolicyBillingService.Domain.Enums;

namespace PolicyBillingService.Application.Services;

public class SimulatedPaymentProcessor
    : IPaymentProcessor
{
    public PaymentSimulationResult Process(
        PaymentMethod method,
        string testScenario)
    {
        if (!Enum.IsDefined(
                typeof(PaymentMethod),
                method))
        {
            return PaymentSimulationResult.Failure(
                "INVALID_PAYMENT_METHOD",
                "Payment method is invalid.");
        }

        var scenario =
            testScenario
                .Trim()
                .ToUpperInvariant();

        return scenario switch
        {
            "SUCCESS" =>
                PaymentSimulationResult.Success(),

            "FAIL" =>
                PaymentSimulationResult.Failure(
                    "SIMULATED_FAILURE",
                    "The simulated payment was declined."),

            _ =>
                PaymentSimulationResult.Failure(
                    "INVALID_TEST_SCENARIO",
                    "Test scenario must be SUCCESS or FAIL.")
        };
    }
}
