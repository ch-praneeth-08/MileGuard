import { Routes } from '@angular/router';

import { authGuard } from '../guards/auth-guard';
import { roleGuard } from '../guards/role-guard';

import { AgentShell } from '../../layout/agent-shell/agent-shell';
import { AgentDashboard } from '../../features/agent/agent-dashboard/agent-dashboard';

export const AGENT_ROUTES: Routes = [
  {
    path: 'agent',
    canActivate: [authGuard, roleGuard],
    data: {
      roles: ['Agent']
    },
    component: AgentShell,

    children: [
      {
        path: '',
        pathMatch: 'full',
        component: AgentDashboard
      },

      {
        path: 'customers',
        loadComponent: () =>
          import(
            '../../features/customer-vehicle/pages/agent-my-customers/agent-my-customers'
          ).then(component => component.AgentMyCustomers)
      },

      {
        path: 'customers/:customerId',
        loadComponent: () =>
          import(
            '../../features/customer-vehicle/pages/agent-customer-workspace/agent-customer-workspace'
          ).then(component => component.AgentCustomerWorkspace)
      },

      {
        path: 'customers/:customerId/vehicles/new',
        loadComponent: () =>
          import(
            '../../features/customer-vehicle/pages/agent-add-vehicle/agent-add-vehicle'
          ).then(component => component.AgentAddVehicle)
      },

      {
        path: 'customers/:customerId/vehicles/:vehicleId',
        loadComponent: () =>
          import(
            '../../features/customer-vehicle/pages/agent-vehicle-detail/agent-vehicle-detail'
          ).then(component => component.AgentVehicleDetail)
      },

      {
        path: 'customers/:customerId/quotes',
        loadComponent: () =>
          import(
            '../../features/quote-rating/pages/agent-quotes/agent-quotes'
          ).then(component => component.AgentQuotes)
      },

      {
        path: 'customers/:customerId/quotes/start',
        loadComponent: () =>
          import(
            '../../features/quote-rating/pages/quote-vehicle-selection/quote-vehicle-selection'
          ).then(component => component.QuoteVehicleSelection)
      },

      {
        path: 'customers/:customerId/quotes/:quoteId',
        loadComponent: () =>
          import(
            '../../features/quote-rating/pages/quote-workspace/quote-workspace'
          ).then(component => component.QuoteWorkspace)
      },

      {
        path: 'customers/:customerId/policies',
        loadComponent: () =>
          import(
            '../../features/policy-billing/pages/agent-customer-policies/agent-customer-policies'
          ).then(component => component.AgentCustomerPolicies)
      },

      {
        path: 'customers/:customerId/policies/:policyId',
        loadComponent: () =>
          import(
            '../../features/policy-billing/pages/agent-policy-detail/agent-policy-detail'
          ).then(component => component.AgentPolicyDetail)
      },

      {
        path: 'customers/:customerId/claims',
        loadComponent: () =>
          import(
            '../../features/claims/pages/agent-customer-claims/agent-customer-claims'
          ).then(component => component.AgentCustomerClaims)
      },

      {
        path: 'customers/:customerId/claims/new',
        loadComponent: () =>
          import(
            '../../features/claims/pages/agent-new-claim/agent-new-claim'
          ).then(component => component.AgentNewClaim)
      },

      {
        path: 'customers/:customerId/claims/:claimId',
        loadComponent: () =>
          import(
            '../../features/claims/pages/agent-claim-detail/agent-claim-detail'
          ).then(component => component.AgentClaimDetailPage)
      },
      {
        path: 'underwriting',
        loadComponent: () =>
          import(
            '../../features/underwriting/pages/agent-underwriting/agent-underwriting'
          ).then(
            component =>
              component.AgentUnderwriting
          )
      },
      {
        path: 'underwriting/:applicationId',
        loadComponent: () =>
          import(
            '../../features/underwriting/pages/agent-underwriting-detail/agent-underwriting-detail'

            component =>
              component.AgentUnderwritingDetailPage

    ]
  }
