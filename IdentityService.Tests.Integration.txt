using IdentityService.Infrastructure.Identity;
using IdentityService.Infrastructure.Persistence;
using Microsoft.EntityFrameworkCore;
using Microsoft.Extensions.DependencyInjection;

namespace IdentityService.Tests.Integration.Infrastructure;

public abstract class IntegrationTestBase
    : IAsyncLifetime
{
    protected readonly IdentityApiFactory Factory;

    protected IntegrationTestBase()
    {
        Factory = new IdentityApiFactory();
    }

    public async Task InitializeAsync()
    {
        using var scope =
            Factory.Services.CreateScope();

        var dbContext =
            scope.ServiceProvider
                .GetRequiredService<IdentityDbContext>();

        await dbContext.Database.EnsureDeletedAsync();

        await dbContext.Database.MigrateAsync();

        var seeder =
            scope.ServiceProvider
                .GetRequiredService<IdentitySeeder>();

        await seeder.SeedAsync();
    }

    public async Task DisposeAsync()
    {
        using var scope =
            Factory.Services.CreateScope();

        var dbContext =
            scope.ServiceProvider
                .GetRequiredService<IdentityDbContext>();

        await dbContext.Database.EnsureDeletedAsync();

        await Factory.DisposeAsync();
    }
}

    [Fact]
    public async Task ApprovePendingInternalUser_PersistsActiveStatusRoleAndApprovalMetadata()
    {
        using var scope =
            Factory.Services.CreateScope();

        var registrationService =
            scope.ServiceProvider.GetRequiredService<
                IRegistrationService>();

        var adminIdentityService =
            scope.ServiceProvider.GetRequiredService<
                IAdminIdentityService>();

        var dbContext =
            scope.ServiceProvider.GetRequiredService<
                IdentityDbContext>();

        var userManager =
            scope.ServiceProvider.GetRequiredService<
                UserManager<
                    IdentityService.Domain.Entities.ApplicationUser>>();

        var registrationRequest =
            new InternalRegisterRequest
            {
                FirstName = "Integration",
                LastName = "Agent",
                Email = "approval.agent@example.com",
                Password = "Agent@123",
                ConfirmPassword = "Agent@123"
            };

        var registrationResult =
            await registrationService.RegisterInternalAsync(
                registrationRequest);

        Assert.True(registrationResult.Succeeded);

        var pendingUser =
            await dbContext.Users
                .SingleAsync(user =>
                    user.Email ==
                        registrationRequest.Email);

        Assert.Equal(
            AccountStatus.Pending,
            pendingUser.AccountStatus);

        var approvalRequest =
            new ApproveInternalRegistrationRequest
            {
                Role = ApplicationRoles.Agent
            };

        var approvalResult =
            await adminIdentityService.ApproveAsync(
                pendingUser.Id,
                approvalRequest,
                "test-admin-id");

        Assert.True(approvalResult.Succeeded);

        dbContext.ChangeTracker.Clear();

        var approvedUser =
            await dbContext.Users
                .SingleAsync(user =>
                    user.Id == pendingUser.Id);

        Assert.Equal(
            AccountType.Internal,
            approvedUser.AccountType);

        Assert.Equal(
            AccountStatus.Active,
            approvedUser.AccountStatus);

        Assert.NotNull(
            approvedUser.ApprovedAtUtc);

        Assert.Equal(
            "test-admin-id",
            approvedUser.ApprovedByUserId);

        var roles =
            await userManager.GetRolesAsync(
                approvedUser);

        Assert.Single(roles);

        Assert.Equal(
            ApplicationRoles.Agent,
            roles[0]);
    }

using IdentityService.Application.Dtos.Authentication;
using IdentityService.Application.Interfaces;
using IdentityService.Application.Interfaces.Repositories;
using IdentityService.Domain.Constants;
using IdentityService.Domain.Entities;

using Microsoft.EntityFrameworkCore;
using Microsoft.Extensions.DependencyInjection;

namespace IdentityService.Tests.Integration.Persistence;

public class RefreshSessionPersistenceTests
    : IntegrationTestBase
{
    [Fact]
    public async Task CreateSession_PersistsHashedRefreshToken()
    {
        using var scope =
            Factory.Services.CreateScope();

        var userManager =
            scope.ServiceProvider.GetRequiredService<
                UserManager<ApplicationUser>>();

        var authSessionService =
            scope.ServiceProvider.GetRequiredService<
                IAuthSessionService>();

        var tokenHashService =
            scope.ServiceProvider.GetRequiredService<
                ITokenHashService>();

        var dbContext =
            scope.ServiceProvider.GetRequiredService<
                IdentityDbContext>();

        var user =
            await CreateCustomerAsync(
                userManager,
                "session.customer@example.com");

        var tokens =
            await authSessionService.CreateSessionAsync(
                user,
                ApplicationRoles.Customer);

        dbContext.ChangeTracker.Clear();

        var session =
            await dbContext.RefreshSessions
                .SingleAsync(item =>
                    item.UserId == user.Id);

        Assert.Equal(
            tokenHashService.Hash(tokens.RefreshToken),
            session.TokenHash);

        Assert.NotEqual(
            tokens.RefreshToken,
            session.TokenHash);

        Assert.Null(
            session.RevokedAtUtc);

        Assert.Null(
            session.ReplacementTokenHash);
    }

    [Fact]
    public async Task Refresh_RotatesSessionAndPreservesAbsoluteExpiry()
    {
        using var scope =
            Factory.Services.CreateScope();

        var userManager =
            scope.ServiceProvider.GetRequiredService<
                UserManager<ApplicationUser>>();

        var authSessionService =
            scope.ServiceProvider.GetRequiredService<
                IAuthSessionService>();

        var refreshTokenService =
            scope.ServiceProvider.GetRequiredService<
                IRefreshTokenService>();

        var dbContext =
            scope.ServiceProvider.GetRequiredService<
                IdentityDbContext>();

        var user =
            await CreateCustomerAsync(
                userManager,
                "rotation.customer@example.com");

        var loginTokens =
            await authSessionService.CreateSessionAsync(
                user,
                ApplicationRoles.Customer);

        var originalSession =
            await dbContext.RefreshSessions
                .SingleAsync(item =>
                    item.UserId == user.Id &&
                    item.RevokedAtUtc == null);

        var originalId =
            originalSession.Id;

        var originalStartedAt =
            originalSession.SessionStartedAtUtc;

        var originalAbsoluteExpiry =
            originalSession.AbsoluteExpiresAtUtc;

        var originalHash =
            originalSession.TokenHash;

        var refreshResult =
            await refreshTokenService.RefreshAsync(
                loginTokens.RefreshToken);

        Assert.True(
            refreshResult.Succeeded);

        Assert.NotNull(
            refreshResult.RefreshToken);

        dbContext.ChangeTracker.Clear();

        var sessions =
            await dbContext.RefreshSessions
                .Where(item =>
                    item.UserId == user.Id)
                .OrderBy(item =>
                    item.SessionStartedAtUtc)
                .ToListAsync();

        Assert.Equal(
            2,
            sessions.Count);

        var revokedSession =
            sessions.Single(item =>
                item.Id == originalId);

        var replacementSession =
            sessions.Single(item =>
                item.Id != originalId);

        Assert.NotNull(
            revokedSession.RevokedAtUtc);

        Assert.NotNull(
            revokedSession.ReplacementTokenHash);

        Assert.Equal(
            replacementSession.TokenHash,
            revokedSession.ReplacementTokenHash);

        Assert.NotEqual(
            originalHash,
            replacementSession.TokenHash);

        Assert.Null(
            replacementSession.RevokedAtUtc);

        Assert.Equal(
            originalStartedAt,
            replacementSession.SessionStartedAtUtc);

        Assert.Equal(
            originalAbsoluteExpiry,
            replacementSession.AbsoluteExpiresAtUtc);

        Assert.Equal(
            originalAbsoluteExpiry,
            refreshResult.RefreshSessionExpiresAtUtc);
    }

    [Fact]
    public async Task NewCompletedLogin_RevokesPreviousActiveSession()
    {
        using var scope =
            Factory.Services.CreateScope();

        var userManager =
            scope.ServiceProvider.GetRequiredService<
                UserManager<ApplicationUser>>();

        var authSessionService =
            scope.ServiceProvider.GetRequiredService<
                IAuthSessionService>();

        var dbContext =
            scope.ServiceProvider.GetRequiredService<
                IdentityDbContext>();

        var user =
            await CreateCustomerAsync(
                userManager,
                "replacement.customer@example.com");

        await authSessionService.CreateSessionAsync(
            user,
            ApplicationRoles.Customer);

        dbContext.ChangeTracker.Clear();

        var firstSession =
            await dbContext.RefreshSessions
                .SingleAsync(item =>
                    item.UserId == user.Id);

        var firstSessionId =
            firstSession.Id;

        await authSessionService.CreateSessionAsync(
            user,
            ApplicationRoles.Customer);

        dbContext.ChangeTracker.Clear();

        var sessions =
            await dbContext.RefreshSessions
                .Where(item =>
                    item.UserId == user.Id)
                .ToListAsync();

        Assert.Equal(
            2,
            sessions.Count);

        var previousSession =
            sessions.Single(item =>
                item.Id == firstSessionId);

        Assert.NotNull(
            previousSession.RevokedAtUtc);

        var activeSessions =
            sessions.Where(item =>
                item.RevokedAtUtc == null)
                .ToList();

        Assert.Single(
            activeSessions);
    }

    private static async Task<ApplicationUser>
        CreateCustomerAsync(
            UserManager<ApplicationUser> userManager,
            string email)
    {
        var user = new ApplicationUser
        {
            FirstName = "Integration",
            LastName = "Customer",
            Email = email,
            UserName = email,
            AccountType = AccountType.Customer,
            AccountStatus = AccountStatus.Active,
            CreatedAtUtc = DateTime.UtcNow
        };

        var createResult =
            await userManager.CreateAsync(
                user,
                "Customer@123");

        Assert.True(
            createResult.Succeeded);

        var roleResult =
            await userManager.AddToRoleAsync(
                user,
                ApplicationRoles.Customer);

        Assert.True(
            roleResult.Succeeded);

        return user;
    }
}
