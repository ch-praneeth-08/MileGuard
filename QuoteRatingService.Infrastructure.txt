using System.Net;
using System.Net.Http.Json;
using QuoteRatingService.Application.Dtos.Clients.CustomerVehicle;
using QuoteRatingService.Application.Interfaces.Clients;

namespace QuoteRatingService.Infrastructure.Clients;

public class CustomerVehicleClient
    : ICustomerVehicleClient
{
    private readonly HttpClient
        _httpClient;

    public CustomerVehicleClient(
        HttpClient httpClient)
    {
        _httpClient =
            httpClient;
    }

    public async Task<CustomerVehicleRiskResult>
        GetCustomerRiskAsync(
            Guid customerProfileId)
    {
        try
        {
            using var response =
                await _httpClient.GetAsync(
                    $"internal/customers/{customerProfileId}/risk");

            if (
                response.StatusCode ==
                HttpStatusCode.NotFound)
            {
                return CustomerVehicleRiskResult
                    .CustomerNotFound();
            }

            if (!response.IsSuccessStatusCode)
            {
                return CustomerVehicleRiskResult
                    .Unavailable();
            }

            var risk =
                await response.Content
                    .ReadFromJsonAsync<
                        CustomerVehicleRiskDto>();

            if (risk is null)
            {
                return CustomerVehicleRiskResult
                    .Unavailable();
            }

            return CustomerVehicleRiskResult
                .Success(risk);
        }
        catch (HttpRequestException)
        {
            return CustomerVehicleRiskResult
                .Unavailable();
        }
        catch (TaskCanceledException)
        {
            return CustomerVehicleRiskResult
                .Unavailable();
        }
    }
}

using Microsoft.Extensions.Configuration;
using QuoteRatingService.Application.Dtos.Clients.Claims;
using QuoteRatingService.Application.Interfaces.Clients;
using Microsoft.Extensions.Configuration;
namespace QuoteRatingService.Infrastructure.Clients;

public class DevelopmentClaimsHistoryClient
    : IClaimsHistoryClient
{
    private readonly IConfiguration
        _configuration;

    public DevelopmentClaimsHistoryClient(
        IConfiguration configuration)
    {
        _configuration =
            configuration;
    }

    public Task<ClaimsHistoryResult>
        GetQualifyingClaimsAsync(
            Guid customerProfileId,
            DateTime ratingAsOfUtc)
    {
        var claimsCount =
            _configuration.GetValue<int>(
                "DevelopmentClaims:QualifyingClaimsCount");

        if (claimsCount < 0)
        {
            throw new InvalidOperationException(
                "Development Claims count cannot be negative.");
        }

        var result =
            ClaimsHistoryResult.Success(
                claimsCount,
                ratingAsOfUtc);

        return Task.FromResult(
            result);
    }
}

using System.Net;
using System.Net.Http.Json;
using QuoteRatingService.Application.Dtos.Underwriting;
using QuoteRatingService.Application.Interfaces.Clients;

namespace QuoteRatingService.Infrastructure.Clients;

public class UnderwritingClient
    : IUnderwritingClient
{
    private readonly HttpClient _httpClient;

    public UnderwritingClient(
        HttpClient httpClient)
    {
        _httpClient = httpClient;
    }

    public async Task<UnderwritingApplicationCreateResult>
        CreateApplicationAsync(
            CreateUnderwritingApplicationRequest request)
    {
        ArgumentNullException.ThrowIfNull(
            request);

        try
        {
            using var response =
                await _httpClient.PostAsJsonAsync(
                    "internal/underwriting/applications",
                    request);

            if (response.StatusCode ==
                    HttpStatusCode.OK ||
                response.StatusCode ==
                    HttpStatusCode.Created)
            {
                var application =
                    await response.Content
                        .ReadFromJsonAsync<
                            CreateUnderwritingApplicationResponse>();

                if (application is null ||
                    application.ApplicationId ==
                    Guid.Empty)
                {
                    return UnderwritingApplicationCreateResult
                        .Unavailable(
                            "UnderwritingService returned an invalid application response.");
                }

                return UnderwritingApplicationCreateResult
                    .Success(
                        application);
            }

            if (response.StatusCode ==
                HttpStatusCode.BadRequest)
            {
                var error =
                    await TryReadErrorAsync(
                        response);

                return UnderwritingApplicationCreateResult
                    .BadRequest(
                        error ??
                        "UnderwritingService rejected the Quote submission.");
            }

            return UnderwritingApplicationCreateResult
                .Unavailable(
                    "UnderwritingService is temporarily unavailable.");
        }
        catch (HttpRequestException)
        {
            return UnderwritingApplicationCreateResult
                .Unavailable(
                    "UnderwritingService is temporarily unavailable.");
        }
        catch (TaskCanceledException)
        {
            return UnderwritingApplicationCreateResult
                .Unavailable(
                    "UnderwritingService request timed out.");
        }
    }

    private static async Task<string?> TryReadErrorAsync(
        HttpResponseMessage response)
    {
        try
        {
            var error =
                await response.Content
                    .ReadFromJsonAsync<ApiErrorResponse>();

            return error?.Message;
        }
        catch
        {
            return null;
        }
    }

    private sealed class ApiErrorResponse
    {
        public string? Message { get; set; }
    }
}

    {
        await SeedPlansAsync();
        await SeedCoveragesAsync();
        await SeedPlanCoveragesAsync();
        await SeedCoverageLimitOptionsAsync();
        await SeedDeductibleOptionsAsync();
        await SeedAddOnsAsync();
        await SeedPlanAddOnsAsync();
        await SeedRatingRulesAsync();
    }

     * =====================================================
     * PLANS
     * =====================================================
     */

    {
        await EnsurePlanAsync(
            QuoteRatingSeedIds.Plans.Basic,
            PlanCode.Basic,
            "Basic",
            8000m);

        await EnsurePlanAsync(
            QuoteRatingSeedIds.Plans.Standard,
            PlanCode.Standard,
            "Standard",
            14000m);

        await EnsurePlanAsync(
            QuoteRatingSeedIds.Plans.Premium,
            PlanCode.Premium,
            "Premium",
            20000m);

        await _dbContext.SaveChangesAsync();

        PlanCode code,

            return;
