using System.Net;
using System.Net.Http.Json;
using UnderwritingService.Application.Dtos.Identity;
using UnderwritingService.Application.Interfaces.Clients;

namespace UnderwritingService.Infrastructure.Clients;

public class IdentityClient : IIdentityClient
{
    private readonly HttpClient _httpClient;

    public IdentityClient(
        HttpClient httpClient)
    {
        _httpClient = httpClient;
    }

    public async Task<EligibleUnderwritersResult>
        GetEligibleUnderwritersAsync()
    {
        try
        {
            using var response =
                await _httpClient.GetAsync(
                    "internal/underwriters/eligible");

            if (response.StatusCode ==
                HttpStatusCode.OK)
            {
                var underwriters =
                    await response.Content
                        .ReadFromJsonAsync<
                            List<EligibleUnderwriterDto>>();

                if (underwriters is null)
                {
                    return EligibleUnderwritersResult.Unavailable(
                        "IdentityService returned an invalid response.");
                }

                return EligibleUnderwritersResult.Success(
                    underwriters);
            }

            return EligibleUnderwritersResult.Unavailable(
                "Eligible Underwriters are temporarily unavailable.");
        }
        catch (HttpRequestException)
        {
            return EligibleUnderwritersResult.Unavailable(
                "IdentityService is temporarily unavailable.");
        }
        catch (TaskCanceledException)
        {
            return EligibleUnderwritersResult.Unavailable(
                "IdentityService request timed out.");
        }
    }

    public async Task<EligibleUnderwriterResult>
        GetEligibleUnderwriterAsync(
            string identityUserId)
    {
        if (string.IsNullOrWhiteSpace(
                identityUserId))
        {
            return EligibleUnderwriterResult.Ineligible(
                "Underwriter Identity User ID is required.");
        }

        try
        {
            var encodedIdentityUserId =
                Uri.EscapeDataString(
                    identityUserId);

            using var response =
                await _httpClient.GetAsync(
                    $"internal/underwriters/{encodedIdentityUserId}/eligible");

            if (response.StatusCode ==
                HttpStatusCode.OK)
            {
                var underwriter =
                    await response.Content
                        .ReadFromJsonAsync<
                            EligibleUnderwriterDto>();

                if (underwriter is null ||
                    string.IsNullOrWhiteSpace(
                        underwriter.Id))
                {
                    return EligibleUnderwriterResult.Unavailable(
                        "IdentityService returned an invalid Underwriter response.");
                }

                return EligibleUnderwriterResult.Success(
                    underwriter);
            }

            if (response.StatusCode ==
                HttpStatusCode.NotFound)
            {
                return EligibleUnderwriterResult.Ineligible(
                    "Selected Underwriter is not currently eligible.");
            }

            return EligibleUnderwriterResult.Unavailable(
                "Underwriter eligibility could not be verified.");
        }
        catch (HttpRequestException)
        {
            return EligibleUnderwriterResult.Unavailable(
                "IdentityService is temporarily unavailable.");
        }
        catch (TaskCanceledException)
        {
            return EligibleUnderwriterResult.Unavailable(
                "IdentityService request timed out.");
        }
    }
}

using Microsoft.EntityFrameworkCore;
using Microsoft.EntityFrameworkCore.Metadata.Builders;
using UnderwritingService.Domain.Entities;

namespace UnderwritingService.Infrastructure
    .Persistence.Configurations;

public class ApplicationAddOnSnapshotConfiguration
    : IEntityTypeConfiguration<ApplicationAddOnSnapshot>
{
    public void Configure(
        EntityTypeBuilder<ApplicationAddOnSnapshot> builder)
    {
        builder.ToTable(
            "ApplicationAddOnSnapshots",
            table =>
            {
                table.HasCheckConstraint(
                    "CK_ApplicationAddOnSnapshots_Price_NonNegative",
                    "[AnnualPrice] >= 0");
            });

        builder.HasKey(
            addOn => addOn.Id);

        builder.Property(
                addOn => addOn.AddOnName)
            .IsRequired()
            .HasMaxLength(100);

        builder.Property(
                addOn => addOn.AnnualPrice)
            .HasPrecision(18, 2)
            .IsRequired();

        builder.HasOne(
                addOn =>
                    addOn.ApplicationSnapshot)
            .WithMany(
                snapshot =>
                    snapshot.AddOns)
            .HasForeignKey(
                addOn =>
                    addOn.ApplicationSnapshotId)
            .OnDelete(
                DeleteBehavior.Cascade);
    }
}

using Microsoft.EntityFrameworkCore;
using UnderwritingService.Application.Interfaces.Repositories;
using UnderwritingService.Domain.Entities;
using UnderwritingService.Domain.Enums;
using UnderwritingService.Infrastructure.Persistence;

namespace UnderwritingService.Infrastructure.Repositories;

public class UnderwritingApplicationRepository
    : IUnderwritingApplicationRepository
{
    private readonly UnderwritingDbContext _dbContext;

    public UnderwritingApplicationRepository(
        UnderwritingDbContext dbContext)
    {
        _dbContext = dbContext;
    }

    public async Task<UnderwritingApplication?>
        GetByOriginatingQuoteIdAsync(
            Guid originatingQuoteId)
    {
        return await _dbContext
            .UnderwritingApplications
            .AsNoTracking()
            .SingleOrDefaultAsync(
                application =>
                    application.OriginatingQuoteId ==
                    originatingQuoteId);
    }

    public async Task<IReadOnlyList<ApplicationReviewSection>>
    GetReviewSectionsAsync(
        Guid applicationId)
    {
        return await _dbContext
            .ApplicationReviewSections
            .AsNoTracking()
            .Where(
                section =>
                    section.UnderwritingApplicationId ==
                    applicationId)
            .ToListAsync();
    }
    public async Task<bool> DecisionExistsAsync(
    Guid applicationId)
    {
        return await _dbContext
            .UnderwritingDecisions
            .AsNoTracking()
            .AnyAsync(
                decision =>
                    decision.UnderwritingApplicationId ==
                    applicationId);
    }
    public async Task AddDecisionAsync(
    UnderwritingDecision decision)
    {
        await _dbContext
            .UnderwritingDecisions
            .AddAsync(
                decision);
    }
    public async Task<ApplicationReviewSection?>
    GetReviewSectionForUpdateAsync(
        Guid applicationId,
        ReviewSectionType sectionType)
    {
        return await _dbContext
            .ApplicationReviewSections
            .SingleOrDefaultAsync(
                section =>
                    section.UnderwritingApplicationId ==
                        applicationId &&
                    section.SectionType ==
                        sectionType);
    }
    public async Task AddStatusHistoryAsync(
    ApplicationStatusHistory history)
    {
        await _dbContext
            .ApplicationStatusHistories
            .AddAsync(history);
    }
    public async Task<UnderwritingApplication?>
    GetByIdForUpdateAsync(
        Guid applicationId)
    {
        return await _dbContext
            .UnderwritingApplications
            .SingleOrDefaultAsync(
                application =>
                    application.Id == applicationId);
    }
    public async Task<UnderwritingApplication?>
    GetAssignedByIdForUpdateAsync(
        Guid applicationId,
        string underwriterIdentityUserId)
    {
        return await _dbContext
            .UnderwritingApplications
            .SingleOrDefaultAsync(
                application =>
                    application.Id ==
                        applicationId &&
                    application.AssignedUnderwriterIdentityUserId ==
                        underwriterIdentityUserId);
    }
    public async Task<int>
    GetActionableCountForUnderwriterAsync(
        string underwriterIdentityUserId)
    {
        return await _dbContext
            .UnderwritingApplications
            .AsNoTracking()
            .CountAsync(
                application =>
                    application.AssignedUnderwriterIdentityUserId ==
                        underwriterIdentityUserId &&
                    (
                        application.Status ==
                            ApplicationStatus.PendingReview ||
                        application.Status ==
                            ApplicationStatus.InReview
                    ));
    }
    public async Task<IReadOnlyList<UnderwritingApplication>>
    GetUnassignedAsync()
    {
        return await _dbContext
            .UnderwritingApplications
            .AsNoTracking()
            .Include(
                application =>
                    application.Snapshot)
            .Where(
                application =>
                    application.Status ==
                    ApplicationStatus.Unassigned)
            .OrderBy(
                application =>
                    application.CreatedAtUtc)
            .ToListAsync();
    }
    public async Task<UnderwritingApplication?>
        GetByIdAsync(
            Guid applicationId)
    {
        return await _dbContext
            .UnderwritingApplications
            .AsNoTracking()
            .AsSplitQuery()
            .Include(
                application =>
                    application.Snapshot)
                .ThenInclude(
                    snapshot =>
                        snapshot.Coverages)
            .Include(
                application =>
                    application.Snapshot)
                .ThenInclude(
                    snapshot =>
                        snapshot.AddOns)
            .Include(
                application =>
                    application.ReviewSections)
            .Include(
                application =>
                    application.Decision)
            .Include(
                application =>
                    application.CustomerOfferOutcome)
            .Include(
                application =>
                    application.StatusHistory)
            .SingleOrDefaultAsync(
                application =>
                    application.Id ==
                    applicationId);
    }

    public async Task AddAsync(
        UnderwritingApplication application)
    {
        await _dbContext
            .UnderwritingApplications
            .AddAsync(
                application);
    }

    public async Task SaveChangesAsync()
    {
        await _dbContext
            .SaveChangesAsync();
    }

    public async Task<IReadOnlyList<UnderwritingApplication>>
    GetAssignedToUnderwriterAsync(
        string underwriterIdentityUserId)
    {
        return await _dbContext
            .UnderwritingApplications
            .AsNoTracking()
            .Include(
                application =>
                    application.Snapshot)
            .Where(
                application =>
                    application.AssignedUnderwriterIdentityUserId ==
                        underwriterIdentityUserId &&
                    (
                        application.Status ==
                            ApplicationStatus.PendingReview ||
                        application.Status ==
                            ApplicationStatus.InReview
                    ))
            .OrderBy(
                application =>
                    application.AssignedAtUtc)
            .ToListAsync();
    }
    public async Task ExecuteInTransactionAsync(
        Func<Task> operation)
    {
        ArgumentNullException.ThrowIfNull(
            operation);

        await using var transaction =
            await _dbContext.Database
                .BeginTransactionAsync();

        try
        {
            await operation();

            await transaction
                .CommitAsync();
        }
        catch
        {
            await transaction
                .RollbackAsync();

            throw;
        }
    }
}
