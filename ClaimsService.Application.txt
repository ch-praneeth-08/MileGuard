using ClaimsService.Application.Dtos.Agent;
using ClaimsService.Application.Dtos.ClaimsOfficer;
using ClaimsService.Application.Dtos.Customer;
using ClaimsService.Application.Exceptions;
using ClaimsService.Application.Interfaces;
using ClaimsService.Application.Interfaces.Clients;
using ClaimsService.Application.Interfaces.Repositories;
using ClaimsService.Application.Dtos.Admin;
using ClaimsService.Domain.Entities;
using ClaimsService.Domain.Enums;
using ApplicationValidationException =
    ClaimsService.Application.Exceptions.ValidationException;

namespace ClaimsService.Application.Services;

public class ClaimService : IClaimService
{
    private readonly IClaimRepository
        _claimRepository;
    private readonly IIdentityClient
    _identityClient;
    private readonly IPolicyBillingClient
        _policyBillingClient;

    private readonly ICustomerVehicleClient
        _customerVehicleClient;

    public ClaimService(
    IClaimRepository claimRepository,
    IPolicyBillingClient policyBillingClient,
    ICustomerVehicleClient customerVehicleClient,
    IIdentityClient identityClient)
    {
        _claimRepository =
            claimRepository;

        _policyBillingClient =
            policyBillingClient;

        _customerVehicleClient =
            customerVehicleClient;

        _identityClient =
            identityClient;
    }

    public async Task<AgentClaimDetailDto>
        SubmitClaimAsync(
            Guid customerProfileId,
            string agentIdentityUserId,
            SubmitClaimRequest request,
            CancellationToken cancellationToken = default)
    {
        ArgumentNullException.ThrowIfNull(
            request);

        if (customerProfileId == Guid.Empty)
        {
            throw new NotFoundException(
                "Customer profile was not found.");
        }

        if (string.IsNullOrWhiteSpace(
                agentIdentityUserId))
        {
            throw new ForbiddenException(
                "Agent identity is required.");
        }

        ValidateSubmitRequest(
            request);

        var assigned =
            await _customerVehicleClient
                .IsCustomerAssignedToAgentAsync(
                    customerProfileId,
                    agentIdentityUserId,
                    cancellationToken);

        if (!assigned)
        {
            throw new ForbiddenException(
                "Customer is not assigned to the current Agent.");
        }

        var policy =
            await _policyBillingClient
                .GetAgentPolicyAsync(
                    request.PolicyId,
                    agentIdentityUserId,
                    cancellationToken);

        if (policy is null)
        {
            throw new NotFoundException(
                "Policy was not found.");
        }

        if (policy.CustomerProfileId !=
            customerProfileId)
        {
            throw new ForbiddenException(
                "The selected Policy does not belong to this Customer.");
        }

        ValidateIncidentAgainstPolicy(
            request.IncidentAtUtc,
            policy.EffectiveDate,
            policy.ExpirationDate,
            policy.CancellationEffectiveAtUtc);

        var claim =
            new Claim
            {
                Id =
                    Guid.NewGuid(),

                ClaimNumber =
                    await GenerateClaimNumberAsync(),

                PolicyId =
                    policy.PolicyId,

                CustomerProfileId =
                    policy.CustomerProfileId,

                VehicleId =
                    policy.VehicleId,

                ClaimType =
                    request.ClaimType,

                IncidentAtUtc =
                    request.IncidentAtUtc,

                IncidentLocation =
                    request
                        .IncidentLocation
                        .Trim(),

                IncidentDescription =
                    request
                        .IncidentDescription
                        .Trim(),

                EstimatedLossAmount =
                    request.EstimatedLossAmount,

                Status =
                    ClaimStatus.Submitted,

                SubmittedAtUtc =
                    DateTime.UtcNow
            };

        await _claimRepository.AddAsync(
            claim);

        await _claimRepository
            .SaveChangesAsync();

        return MapAgentDetail(
            claim);
    }

    public async Task<IReadOnlyList<AgentClaimListItemDto>>
        GetAgentCustomerClaimsAsync(
            Guid customerProfileId,
            string agentIdentityUserId,
            CancellationToken cancellationToken = default)
    {
        if (customerProfileId == Guid.Empty)
        {
            throw new NotFoundException(
                "Customer profile was not found.");
        }

        if (string.IsNullOrWhiteSpace(
                agentIdentityUserId))
        {
            throw new ForbiddenException(
                "Agent identity is required.");
        }

        var assigned =
            await _customerVehicleClient
                .IsCustomerAssignedToAgentAsync(
                    customerProfileId,
                    agentIdentityUserId,
                    cancellationToken);

        if (!assigned)
        {
            throw new ForbiddenException(
                "Customer is not assigned to the current Agent.");
        }

        var claims =
            await _claimRepository
                .GetByCustomerProfileIdAsync(
                    customerProfileId);

        return claims
            .OrderByDescending(
                claim =>
                    claim.SubmittedAtUtc)
            .Select(
                MapAgentListItem)
            .ToList();
    }

    public async Task<AgentClaimDetailDto>
        GetAgentClaimAsync(
            Guid claimId,
            string agentIdentityUserId,
            CancellationToken cancellationToken = default)
    {
        var claim =
            await GetRequiredClaimAsync(
                claimId);

        if (string.IsNullOrWhiteSpace(
                agentIdentityUserId))
        {
            throw new ForbiddenException(
                "Agent identity is required.");
        }

        var assigned =
            await _customerVehicleClient
                .IsCustomerAssignedToAgentAsync(
                    claim.CustomerProfileId,
                    agentIdentityUserId,
                    cancellationToken);

        if (!assigned)
        {
            throw new ForbiddenException(
                "Customer is not assigned to the current Agent.");
        }

        return MapAgentDetail(
            claim);
    }

    public async Task<IReadOnlyList<ClaimListItemDto>>
        GetCustomerClaimsAsync(
            string customerIdentityUserId,
            CancellationToken cancellationToken = default)
    {
        var customer =
            await GetRequiredCustomerAsync(
                customerIdentityUserId,
                cancellationToken);

        var claims =
            await _claimRepository
                .GetByCustomerProfileIdAsync(
                    customer.CustomerProfileId);

        return claims
            .OrderByDescending(
                claim =>
                    claim.SubmittedAtUtc)
            .Select(
                MapCustomerListItem)
            .ToList();
    }

    public async Task<CustomerClaimDetailDto>
        GetCustomerClaimAsync(
            Guid claimId,
            string customerIdentityUserId,
            CancellationToken cancellationToken = default)
    {
        var customer =
            await GetRequiredCustomerAsync(
                customerIdentityUserId,
                cancellationToken);

        var claim =
            await GetRequiredClaimAsync(
                claimId);

        if (claim.CustomerProfileId !=
            customer.CustomerProfileId)
        {
            throw new ForbiddenException(
                "This Claim does not belong to the current Customer.");
        }

        return MapCustomerDetail(
            claim);
    }

    public async Task<IReadOnlyList<ClaimsOfficerClaimListItemDto>>
    GetClaimsOfficerQueueAsync(
        string claimsOfficerIdentityUserId,
        CancellationToken cancellationToken = default)
    {
        ValidateClaimsOfficerIdentity(
            claimsOfficerIdentityUserId);

        var claims =
            await _claimRepository
                .GetByAssignedClaimsOfficerAsync(
                    claimsOfficerIdentityUserId);

                    claimsOfficer.ActionableClaimCount)

            .ThenBy(
                claimsOfficer =>

    public async Task<ClaimsOfficerClaimDetailDto>
        SettleAsync(
            Guid claimId,

            SettleClaimRequest request,
            CancellationToken cancellationToken = default)
    {
        ArgumentNullException.ThrowIfNull(

            claimsOfficerIdentityUserId);

        if (request.ApprovedAmount <= 0)
        {
            throw CreateValidationException(
                "approvedAmount",
                "Approved Amount must be greater than zero.");
        }

        var claim =
            await GetRequiredClaimAsync(
                claimId);

        EnsureReviewOwner(
            claim,
            claimsOfficerIdentityUserId);

        if (claim.Status !=
            ClaimStatus.Approved)
        {
            throw new ConflictException(
                "Only an approved Claim can be settled.");
        }

        claim.ApprovedAmount =
            request.ApprovedAmount;

        claim.SettledAtUtc =
            DateTime.UtcNow;

        claim.Status =
            ClaimStatus.Settled;

        await _claimRepository
            .SaveChangesAsync();

        return MapClaimsOfficerDetail(
            claim);
    }

            claimsOfficerIdentityUserId);

            .SaveChangesAsync();

        return MapClaimsOfficerDetail(
            claim);
    }

    private static void ValidateSubmitRequest(
        SubmitClaimRequest request)
    {
        var errors =
            new Dictionary<string, string[]>();

        if (request.PolicyId ==
            Guid.Empty)
        {
            errors["policyId"] =
            [
                "Policy ID is required."
            ];
        }

        if (!Enum.IsDefined(
                typeof(ClaimType),
                request.ClaimType))
        {
            errors["claimType"] =
            [
                "Claim Type is invalid."
            ];
        }

        if (request.IncidentAtUtc ==
            default)
        {
            errors["incidentAtUtc"] =
            [
                "Incident date and time are required."
            ];
        }
        else if (
            request.IncidentAtUtc >
            DateTime.UtcNow)
        {
            errors["incidentAtUtc"] =
            [
                "Incident date and time cannot be in the future."
            ];
        }

        if (string.IsNullOrWhiteSpace(
                request.IncidentLocation))
        {
            errors["incidentLocation"] =
            [
                "Incident Location is required."
            ];
        }
        else if (
            request.IncidentLocation
                .Trim()
                .Length > 250)
        {
            errors["incidentLocation"] =
            [
                "Incident Location cannot exceed 250 characters."
            ];
        }

        if (string.IsNullOrWhiteSpace(
                request.IncidentDescription))
        {
            errors["incidentDescription"] =
            [
                "Incident Description is required."
            ];
        }
        else if (
            request.IncidentDescription
                .Trim()
                .Length > 2000)
        {
            errors["incidentDescription"] =
            [
                "Incident Description cannot exceed 2000 characters."
            ];
        }

        if (request.EstimatedLossAmount <=
            0)
        {
            errors["estimatedLossAmount"] =
            [
                "Estimated Loss Amount must be greater than zero."
            ];
        }

        if (errors.Count > 0)
        {
            throw new ApplicationValidationException(
                errors);
        }
    }

    private static void
        ValidateIncidentAgainstPolicy(
            DateTime incidentAtUtc,
            DateOnly effectiveDate,
            DateOnly expirationDate,
            DateTime? cancellationEffectiveAtUtc)
    {
        var incidentDate =
            DateOnly.FromDateTime(
                incidentAtUtc);

        if (incidentDate <
            effectiveDate)
        {
            throw new ConflictException(
                "The incident occurred before the Policy became effective.");
        }

        if (incidentDate >=
            expirationDate)
        {
            throw new ConflictException(
                "The incident occurred after the Policy coverage period ended.");
        }

        if (
            cancellationEffectiveAtUtc.HasValue &&
            incidentAtUtc >=
                cancellationEffectiveAtUtc.Value)
        {
            throw new ConflictException(
                "The incident occurred after the Policy cancellation became effective.");
        }
    }

    /*
     * =====================================================
     * CLAIM NUMBER
     * =====================================================
     */

    private async Task<string>
        GenerateClaimNumberAsync()
    {
        for (
            var attempt = 0;
            attempt < 5;
            attempt++)
        {
            var candidate =
                $"CLM-{DateTime.UtcNow:yyyyMMdd}-{Guid.NewGuid():N}"
                    .ToUpperInvariant();

            candidate =
                candidate[..21];

            if (!await _claimRepository
                    .ClaimNumberExistsAsync(
                        candidate))
            {
                return candidate;
            }
        }

        throw new InvalidOperationException(
            "A unique Claim Number could not be generated.");
    }

    public async Task<AdminClaimDetailDto>
        AssignClaimsOfficerAsync(
            Guid claimId,
            string adminIdentityUserId,
            AssignClaimsOfficerRequest request,
            CancellationToken cancellationToken = default)
    {
        ArgumentNullException.ThrowIfNull(
            request);

        if (string.IsNullOrWhiteSpace(
                adminIdentityUserId))
        {
            throw new ForbiddenException(
                "Admin identity is required.");
        }

        var claimsOfficerIdentityUserId =
            request.ClaimsOfficerIdentityUserId
                ?.Trim();

        if (string.IsNullOrWhiteSpace(
                claimsOfficerIdentityUserId))
        {
            throw CreateValidationException(
                "claimsOfficerIdentityUserId",
                "Claims Officer Identity User ID is required.");
        }
        var eligibleClaimsOfficer =
    await _identityClient
        .GetEligibleClaimsOfficerAsync(
            claimsOfficerIdentityUserId,
            cancellationToken);

        if (eligibleClaimsOfficer is null)
        {
            throw new ConflictException(
                "The selected Claims Officer is not eligible for assignment.");
        }
        var claim =
            await GetRequiredClaimAsync(
                claimId);

        if (claim.Status !=
            ClaimStatus.Submitted)
        {
            throw new ConflictException(
                "Only a submitted Claim can be assigned.");
        }

        if (!string.IsNullOrWhiteSpace(
                claim.AssignedClaimsOfficerIdentityUserId))
        {
            throw new ConflictException(
                "This Claim is already assigned to a Claims Officer.");
        }

        claim.AssignedClaimsOfficerIdentityUserId =
            claimsOfficerIdentityUserId;

        claim.AssignedByAdminIdentityUserId =
            adminIdentityUserId;

        claim.AssignedAtUtc =
            DateTime.UtcNow;

        await _claimRepository
            .SaveChangesAsync();

        return MapAdminDetail(
            claim);
    }
