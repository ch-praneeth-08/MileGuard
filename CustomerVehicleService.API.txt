using System.Security.Claims;
using CustomerVehicleService.Application.Dtos.Assignments;
using CustomerVehicleService.Application.Interfaces;
using CustomerVehicleService.Domain.Constants;
using FluentValidation;
using Microsoft.AspNetCore.Authorization;
using Microsoft.AspNetCore.Mvc;

namespace CustomerVehicleService.API.Controllers;

[ApiController]
[Route("api/admin/customers")]
[Authorize(Roles = ApplicationRoles.Admin)]
public class AdminCustomersController : ControllerBase
{
    private readonly IAgentAssignmentService _assignmentService;

    private readonly IValidator<AssignAgentRequest>
        _assignAgentValidator;

    public AdminCustomersController(
        IAgentAssignmentService assignmentService,
        IValidator<AssignAgentRequest> assignAgentValidator)
    {
        _assignmentService = assignmentService;
        _assignAgentValidator = assignAgentValidator;
    }

    [HttpGet("unassigned")]
    public async Task<IActionResult> GetUnassignedCustomers()
    {
        var customers =
            await _assignmentService
                .GetUnassignedCustomersAsync();

        return Ok(customers);
    }

    [HttpGet("{id:guid}")]
    public async Task<IActionResult> GetCustomer(
        Guid id)
    {
        var customer =
            await _assignmentService
                .GetCustomerForAssignmentAsync(id);

        if (customer is null)
        {
            return NotFound(new
            {
                message =
                    "Customer profile was not found."
            });
        }

        return Ok(customer);
    }

    [HttpPost("{id:guid}/agent")]
    public async Task<IActionResult> AssignAgent(
        Guid id,
        AssignAgentRequest request)
    {
        var validation =
            await _assignAgentValidator
                .ValidateAsync(request);

        if (!validation.IsValid)
        {
            var errors =
                validation.Errors
                    .GroupBy(
                        error =>
                            error.PropertyName)
                    .ToDictionary(
                        group =>
                            group.Key,
                        group =>
                            group
                                .Select(
                                    error =>
                                        error.ErrorMessage)
                                .ToArray());

            return BadRequest(new
            {
                message =
                    "Validation failed.",
                errors
            });
        }

        var adminIdentityUserId =
            User.FindFirstValue(
                ClaimTypes.NameIdentifier);

        if (
            string.IsNullOrWhiteSpace(
                adminIdentityUserId)
        )
        {
            return Unauthorized();
        }

        var result =
            await _assignmentService
                .AssignAgentAsync(
                    id,
                    adminIdentityUserId,
                    request);

        if (result.CustomerNotFound)
        {
            return NotFound(new
            {
                message =
                    "Customer profile was not found."
            });
        }

        if (result.ProfileIncomplete)
        {
            return Conflict(new
            {
                message =
                    "Customer profile is not complete."
            });
        }

        if (result.AlreadyAssigned)
        {
            return Conflict(new
            {
                message =
                    "Customer already has a permanent Agent assignment."
            });
        }

        if (result.AgentIneligible)
        {
            return Conflict(new
            {
                message =
                    "The selected Agent is no longer eligible for assignment."
            });
        }

        if (result.IdentityUnavailable)
        {
            return StatusCode(
                StatusCodes.Status503ServiceUnavailable,
                new
                {
                    message =
                        "Identity service is currently unavailable."
                });
        }

        if (
            !result.Succeeded ||
            result.Assignment is null)
        {
            return BadRequest(new
            {
                message =
                    "Agent assignment could not be completed."
            });
        }

        return Ok(result.Assignment);
    }
}

[Authorize(Roles = ApplicationRoles.Agent)]
public class AgentVehiclesController : ControllerBase
{
    private readonly IVehicleService _vehicleService;

    private readonly IValidator<UpsertVehicleRequest>
        _validator;

    public AgentVehiclesController(
        IVehicleService vehicleService,
        IValidator<UpsertVehicleRequest> validator)
    {
        _vehicleService = vehicleService;
        _validator = validator;
    }

    [HttpGet]
    public async Task<IActionResult> GetVehicles(
        Guid customerId)
    {
        var agentIdentityUserId =
            GetCurrentUserId();

        if (
            string.IsNullOrWhiteSpace(
                agentIdentityUserId))
        {
            return Unauthorized();
        }

        var result =
            await _vehicleService
                .GetForAgentAsync(
                    agentIdentityUserId,
                    customerId);

        if (result.CustomerNotFound)
        {
            return NotFound(new
            {
                message =
                    "Customer profile was not found."
            });
        }

        if (result.NotAssignedToAgent)
        {
            return StatusCode(
                StatusCodes.Status403Forbidden,
                new
                {
                    message =
                        "Customer is not assigned to the current Agent."
                });
        }

        if (!result.Succeeded)
        {
            return BadRequest(new
            {
                message =
                    "Vehicles could not be loaded."
            });
        }

        return Ok(result.Vehicles);
    }

    [HttpPost]
    public async Task<IActionResult> CreateVehicle(
        Guid customerId,
        UpsertVehicleRequest request)
    {
        var validation =
            await _validator
                .ValidateAsync(request);

        if (!validation.IsValid)
        {
            return BadRequest(
                CreateValidationResponse(
                    validation));
        }

        var agentIdentityUserId =
            GetCurrentUserId();

        if (
            string.IsNullOrWhiteSpace(
                agentIdentityUserId))
        {
            return Unauthorized();
        }

        var result =
            await _vehicleService
                .CreateForAgentAsync(
                    agentIdentityUserId,
                    customerId,
                    request);

        if (result.CustomerNotFound)
        {
            return NotFound(new
            {
                message =
                    "Customer profile was not found."
            });
        }

        if (result.NotAssignedToAgent)
        {
            return StatusCode(
                StatusCodes.Status403Forbidden,
                new
                {
                    message =
                        "Customer is not assigned to the current Agent."
                });
        }

        if (result.DuplicateVin)
        {
            return Conflict(new
            {
                message =
                    "VIN is already in use."
            });
        }

        if (result.DuplicateRegistration)
        {
            return Conflict(new
            {
                message =
                    "Registration Number is already in use."
            });
        }

        if (
            !result.Succeeded ||
            result.Vehicle is null)
        {
            return BadRequest(new
            {
                message =
                    "Vehicle could not be created."
            });
        }

        return CreatedAtRoute(
            "GetVehicle",
            new
            {
                id = result.Vehicle.Id
            },
            result.Vehicle);
    }

    private string? GetCurrentUserId()
    {
        return User.FindFirstValue(
            ClaimTypes.NameIdentifier);
    }

        FluentValidation.Results.ValidationResult validation)
    {
        var errors =
            validation.Errors
                .GroupBy(
                    error =>
                        error.PropertyName)
                .ToDictionary(
                    group =>
                        group.Key,
                    group =>
                        group
                            .Select(
                                error =>
                                    error.ErrorMessage)
                            .ToArray());

        return new
        {
            message =
                "Validation failed.",
            errors
        };
    }
}

using System.Security.Claims;
using CustomerVehicleService.Application.Interfaces;
using CustomerVehicleService.Application.Interfaces.Repositories;
using CustomerVehicleService.Domain.Constants;
using Microsoft.AspNetCore.Authorization;
using Microsoft.AspNetCore.Mvc;

namespace CustomerVehicleService.API.Controllers;

[ApiController]
[Route("api")]
public class CustomerReadinessController
    : ControllerBase
{
    private readonly ICustomerReadinessService
        _customerReadinessService;

    private readonly ICustomerProfileRepository
        _customerProfileRepository;

    private readonly IAgentCustomerRepository
        _agentCustomerRepository;

    public CustomerReadinessController(
        ICustomerReadinessService customerReadinessService,
        ICustomerProfileRepository customerProfileRepository,
        IAgentCustomerRepository agentCustomerRepository)
    {
        _customerReadinessService =
            customerReadinessService;

        _customerProfileRepository =
            customerProfileRepository;

        _agentCustomerRepository =
            agentCustomerRepository;
    }

    /*
     * Customer:
     * retrieve readiness for the currently authenticated
     * CustomerProfile.
     */
    [HttpGet("customers/me/readiness")]
    [Authorize(
        Roles = ApplicationRoles.Customer)]
    public async Task<IActionResult>
        GetMyReadiness()
    {
        var identityUserId =
            GetCurrentUserId();

        if (
            string.IsNullOrWhiteSpace(
                identityUserId))
        {
            return Unauthorized();
        }

        var profile =
            await _customerProfileRepository
                .GetByIdentityUserIdAsync(
                    identityUserId);

        if (profile is null)
        {
            return NotFound(new
            {
                message =
                    "Customer profile was not found."
            });
        }

        var readiness =
            await _customerReadinessService
                .GetForCustomerProfileAsync(
                    profile.Id);

        if (readiness is null)
        {
            return NotFound(new
            {
                message =
                    "Customer profile was not found."
            });
        }

        return Ok(readiness);
    }

    /*
     * Agent:
     * retrieve readiness only for a Customer permanently
     * assigned to the currently authenticated Agent.
     */
    [HttpGet(
        "agent/customers/{customerProfileId:guid}/readiness")]
    [Authorize(
        Roles = ApplicationRoles.Agent)]
    public async Task<IActionResult>
        GetAgentCustomerReadiness(
            Guid customerProfileId)
    {
        var agentIdentityUserId =
            GetCurrentUserId();

        if (
            string.IsNullOrWhiteSpace(
                agentIdentityUserId))
        {
            return Unauthorized();
        }

        /*
         * Distinguish an unknown Customer from an
         * unauthorized assigned-customer request.
         */
        var profile =
            await _agentCustomerRepository
                .GetCustomerProfileAsync(
                    customerProfileId);

        if (profile is null)
        {
            return NotFound(new
            {
                message =
                    "Customer profile was not found."
            });
        }

        var assignment =
            await _agentCustomerRepository
                .GetAssignmentAsync(
                    customerProfileId);

        if (
            assignment is null ||
            assignment.AgentIdentityUserId !=
                agentIdentityUserId)
        {
            return StatusCode(
                StatusCodes.Status403Forbidden,
                new
                {
                    message =
                        "Customer is not assigned to the current Agent."
                });
        }

        var readiness =
            await _customerReadinessService
                .GetForCustomerProfileAsync(
                    customerProfileId);

        if (readiness is null)
        {
            return NotFound(new
            {
                message =
                    "Customer profile was not found."
            });
        }

        return Ok(readiness);
    }

    private string? GetCurrentUserId()
    {
        return User.FindFirstValue(
            ClaimTypes.NameIdentifier);
    }
}

    [HttpPost("{id:guid}/image")]

            });

        await using var stream =
            file.OpenReadStream();

        var result =
            await _vehicleImageService
                .UploadForAgentAsync(
                    agentIdentityUserId,
                    id,
                    new()
                    {
                        FileName =
                            file.FileName,

                        ContentType =
                            file.ContentType,

                        Length =
                            file.Length,

                        Content =
                            stream
                    });

        if (result.VehicleNotFound)
        {
            return NotFound(new
            {
                message =
                    "Vehicle was not found."
            });
        }

        if (result.NotAuthorized)
        {
            return StatusCode(
                StatusCodes.Status403Forbidden,
                new
                {
                    message =
                        "You are not authorized to modify this Vehicle."
                });
        }

        if (result.InvalidFile)
        {
            return BadRequest(new
            {
                message =
                    result.Error
            });
        }

        return Ok(new
        {
            message =
                "Vehicle image uploaded successfully."
        });
    }

    [HttpPut("{id:guid}")]
    [Authorize(Roles = ApplicationRoles.Agent)]
    public async Task<IActionResult> UpdateVehicle(
        Guid id,
        UpsertVehicleRequest request)
    {
        var validation =
            await _validator
                .ValidateAsync(request);

        if (!validation.IsValid)
        {
            var errors =
                validation.Errors
                    .GroupBy(
                        error =>
                            error.PropertyName)
                    .ToDictionary(
                        group =>
                            group.Key,
                        group =>
                            group
                                .Select(
                                    error =>
                                        error.ErrorMessage)
                            .ToArray());

            return BadRequest(new
            {
                message =
                    "Validation failed.",
                errors
            });
        }

        var agentIdentityUserId =
            GetCurrentUserId();

        if (
            string.IsNullOrWhiteSpace(
                agentIdentityUserId))
        {
            return Unauthorized();
        }

        var result =
            await _vehicleService
                .UpdateForAgentAsync(
                    agentIdentityUserId,
                    id,
                    request);

        if (result.VehicleNotFound)
        {
            return NotFound(new
            {
                message =
                    "Vehicle was not found."
            });
        }

        if (result.NotAssignedToAgent)
        {
            return StatusCode(
                StatusCodes.Status403Forbidden,
                new
                {
                    message =
                        "Customer is not assigned to the current Agent."
                });
        }

        if (result.DuplicateVin)
        {
            return Conflict(new
            {
                message =
                    "VIN is already in use."
            });
        }

        if (result.DuplicateRegistration)
        {
            return Conflict(new
            {
                message =
                    "Registration Number is already in use."
            });
        }

        if (
            !result.Succeeded ||
            result.Vehicle is null)
        {
            return BadRequest(new
            {
                message =
                    "Vehicle could not be updated."
            });
        }

        return Ok(result.Vehicle);
    }

    private string? GetCurrentUserId()
    {
        return User.FindFirstValue(
            ClaimTypes.NameIdentifier);
    }
}
