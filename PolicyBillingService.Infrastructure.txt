using System.Net;
using System.Net.Http.Json;
using PolicyBillingService.Application.Dtos.CustomerVehicle;
using PolicyBillingService.Application.Interfaces.Clients;

namespace PolicyBillingService.Infrastructure.Clients;

public class CustomerVehicleClient
    : ICustomerVehicleClient
{
    private readonly HttpClient _httpClient;

    public CustomerVehicleClient(
        HttpClient httpClient)
    {
        _httpClient = httpClient;
    }

    public async Task<AgentCustomerAssignmentResult>
    CheckAgentAssignmentAsync(
        Guid customerProfileId,
        string agentIdentityUserId)
    {
        if (customerProfileId == Guid.Empty)
        {
            return AgentCustomerAssignmentResult.Missing(
                "Customer Profile ID is required.");
        }

        if (string.IsNullOrWhiteSpace(
                agentIdentityUserId))
        {
            return AgentCustomerAssignmentResult.Missing(
                "Agent Identity User ID is required.");
        }

        try
        {
            var encodedAgentIdentityUserId =
                Uri.EscapeDataString(
                    agentIdentityUserId);

            var requestUrl =
                $"internal/customers/{customerProfileId}/assigned-agent/{encodedAgentIdentityUserId}";

            using var response =
                await _httpClient.GetAsync(
                    requestUrl);
            var responseBody =
    await response.Content.ReadAsStringAsync();

            Console.WriteLine(
                $"CustomerVehicle request: {_httpClient.BaseAddress}{requestUrl}");

            Console.WriteLine(
                $"CustomerVehicle status: {(int)response.StatusCode} {response.StatusCode}");

            Console.WriteLine(
                $"CustomerVehicle response: {responseBody}");

            if (response.StatusCode ==
                HttpStatusCode.OK)
            {
                var assignment =
    System.Text.Json.JsonSerializer
        .Deserialize<AgentCustomerAssignmentDto>(
            responseBody,
            new System.Text.Json.JsonSerializerOptions
            {
                PropertyNameCaseInsensitive = true
            });

                if (assignment is null)
                {
                    return AgentCustomerAssignmentResult.Unavailable(
                        "CustomerVehicleService returned an invalid Agent assignment response.");
                }

                if (assignment.CustomerProfileId !=
                    customerProfileId)
                {
                    return AgentCustomerAssignmentResult.Unavailable(
                        "CustomerVehicleService returned an unexpected Customer Profile ID.");
                }

                return AgentCustomerAssignmentResult.Success(
                    assignment.IsAssigned);
            }

            if (response.StatusCode ==
                HttpStatusCode.NotFound)
            {
                return AgentCustomerAssignmentResult.Missing(
                    "Customer profile was not found.");
            }

            return AgentCustomerAssignmentResult.Unavailable(
                "CustomerVehicleService returned an unexpected response.");
        }
        catch (HttpRequestException)
        {
            return AgentCustomerAssignmentResult.Unavailable(
                "CustomerVehicleService is temporarily unavailable.");
        }
        catch (TaskCanceledException)
        {
            return AgentCustomerAssignmentResult.Unavailable(
                "CustomerVehicleService request timed out.");
        }
    }
    public async Task<CustomerProfileReferenceResult>
        GetCustomerProfileAsync(
            string identityUserId)
    {
        if (string.IsNullOrWhiteSpace(identityUserId))
        {
            return CustomerProfileReferenceResult.Missing(
                "Customer Identity User ID is required.");
        }

        try
        {
            var encodedIdentityUserId =
                Uri.EscapeDataString(
                    identityUserId);

            using var response =
                await _httpClient.GetAsync(
                    $"internal/customers/by-identity/{encodedIdentityUserId}");

            if (response.StatusCode ==
                HttpStatusCode.OK)
            {
                var customer =
                    await response.Content
                        .ReadFromJsonAsync<
                            CustomerProfileReferenceDto>();

                if (customer is null ||
                    customer.CustomerProfileId == Guid.Empty ||
                    string.IsNullOrWhiteSpace(
                        customer.IdentityUserId))
                {
                    return CustomerProfileReferenceResult
                        .Unavailable(
                            "CustomerVehicleService returned an invalid Customer response.");
                }

                return CustomerProfileReferenceResult.Success(
                    customer);
            }

            if (response.StatusCode ==
                HttpStatusCode.NotFound)
            {
                return CustomerProfileReferenceResult.Missing(
                    "Customer profile was not found.");
            }

            return CustomerProfileReferenceResult.Unavailable(
                "CustomerVehicleService is temporarily unavailable.");
        }
        catch (HttpRequestException)
        {
            return CustomerProfileReferenceResult.Unavailable(
                "CustomerVehicleService is temporarily unavailable.");
        }
        catch (TaskCanceledException)
        {
            return CustomerProfileReferenceResult.Unavailable(
                "CustomerVehicleService request timed out.");
        }
    }

using System.Net;
using System.Net.Http.Json;
using PolicyBillingService.Application.Dtos.QuoteRating;
using PolicyBillingService.Application.Interfaces.Clients;

namespace PolicyBillingService.Infrastructure.Clients;

public class QuoteRatingClient : IQuoteRatingClient
{
    private readonly HttpClient _httpClient;

    public QuoteRatingClient(
        HttpClient httpClient)
    {
        _httpClient = httpClient;
    }

    public async Task<StartRenewalQuoteResult>
        StartRenewalQuoteAsync(
            StartRenewalQuoteRequest request)
    {
        ArgumentNullException.ThrowIfNull(request);

        try
        {
            using var response =
                await _httpClient.PostAsJsonAsync(
                    "internal/quotes/renewal",
                    request);

            if (response.StatusCode ==
                HttpStatusCode.OK)
            {
                var quote =
                    await response.Content
                        .ReadFromJsonAsync<
                            StartRenewalQuoteDto>();

                if (quote is null ||
                    quote.QuoteId == Guid.Empty ||
                    quote.RenewalId != request.RenewalId ||
                    quote.SourcePolicyId !=
                        request.SourcePolicyId ||
                    quote.CustomerProfileId !=
                        request.CustomerProfileId ||
                    quote.VehicleId !=
                        request.VehicleId)
                {
                    return StartRenewalQuoteResult.Unavailable(
                        "QuoteRatingService returned an invalid Renewal Quote response.");
                }

                return StartRenewalQuoteResult.Success(
                    quote);
            }

            if (response.StatusCode ==
                HttpStatusCode.Conflict)
            {
                return StartRenewalQuoteResult.ConflictResult(
                    "Renewal Quote could not be started.");
            }

            return StartRenewalQuoteResult.Unavailable(
                "QuoteRatingService is temporarily unavailable.");
        }
        catch (HttpRequestException)
        {
            return StartRenewalQuoteResult.Unavailable(
                "QuoteRatingService is temporarily unavailable.");
        }
        catch (TaskCanceledException)
        {
            return StartRenewalQuoteResult.Unavailable(
                "QuoteRatingService request timed out.");
        }
    }
}

using System.Net;
using System.Net.Http.Json;
using PolicyBillingService.Application.Dtos.Underwriting;
using PolicyBillingService.Application.Interfaces.Clients;

namespace PolicyBillingService.Infrastructure.Clients;

public class UnderwritingClient : IUnderwritingClient
{
    private readonly HttpClient _httpClient;

    public UnderwritingClient(
        HttpClient httpClient)
    {
        _httpClient = httpClient;
    }

    public async Task<ApprovedOfferResult>
        GetApprovedOfferAsync(
            Guid underwritingApplicationId)
    {
        if (underwritingApplicationId == Guid.Empty)
        {
            return ApprovedOfferResult.Missing(
                "Underwriting application was not found.");
        }

        try
        {
            using var response =
                await _httpClient.GetAsync(
                    $"internal/underwriting/applications/{underwritingApplicationId}/approved-offer");

            if (response.StatusCode ==
                HttpStatusCode.OK)
            {
                var offer =
                    await response.Content
                        .ReadFromJsonAsync<ApprovedOfferDto>();

                if (offer is null ||
                    offer.UnderwritingApplicationId ==
                        Guid.Empty ||
                    offer.CustomerProfileId ==
                        Guid.Empty ||
                    offer.VehicleId ==
                        Guid.Empty ||
                    offer.AnnualPremium <= 0)
                {
                    return ApprovedOfferResult.Unavailable(
                        "UnderwritingService returned an invalid approved-offer response.");
                }

                return ApprovedOfferResult.Success(
                    offer);
            }

            if (response.StatusCode ==
                HttpStatusCode.NotFound)
            {
                return ApprovedOfferResult.Missing(
                    "Underwriting application was not found.");
            }

            if (response.StatusCode ==
                HttpStatusCode.Conflict)
            {
                return ApprovedOfferResult.Unusable(
                    "Underwriting offer is not currently available for purchase.");
            }

            return ApprovedOfferResult.Unavailable(
                "UnderwritingService is temporarily unavailable.");
        }
        catch (HttpRequestException)
        {
            return ApprovedOfferResult.Unavailable(
                "UnderwritingService is temporarily unavailable.");
        }
        catch (TaskCanceledException)
        {
            return ApprovedOfferResult.Unavailable(
                "UnderwritingService request timed out.");
        }
    }
}

using PolicyBillingService.Application.Interfaces.Documents;
using PolicyBillingService.Domain.Entities;
using QuestPDF.Fluent;
using QuestPDF.Helpers;
using QuestPDF.Infrastructure;

namespace PolicyBillingService.Infrastructure.Documents;

public class PolicyPdfGenerator
    : IPolicyPdfGenerator
{
    public byte[] Generate(
        Policy policy)
    {
        ArgumentNullException.ThrowIfNull(policy);

        if (policy.Snapshot is null)
        {
            throw new InvalidOperationException(
                "Policy does not contain its immutable snapshot.");
        }

        var snapshot =
            policy.Snapshot;

        var document =
            Document.Create(container =>
            {
                container.Page(page =>
                {
                    page.Size(PageSizes.A4);

                    page.Margin(40);

                    page.DefaultTextStyle(
                        style =>
                            style.FontSize(10));

                    page.Header()
                        .Column(column =>
                        {
                            column.Item()
                                .Text("AUTO INSURANCE POLICY")
                                .Bold()
                                .FontSize(20)
                                .FontColor(
                                    Colors.Blue.Darken2);

                            column.Item()
                                .PaddingTop(5)
                                .Text(
                                    $"Policy Number: {policy.PolicyNumber}")
                                .Bold();
                        });

                    page.Content()
                        .PaddingVertical(20)
                        .Column(column =>
                        {
                            column.Spacing(18);

                            column.Item()
                                .Element(
                                    section =>
                                        AddPolicySummary(
                                            section,
                                            policy,
                                            snapshot));

                            column.Item()
                                .Element(
                                    section =>
                                        AddVehicleDetails(
                                            section,
                                            snapshot));

                            column.Item()
                                .Element(
                                    section =>
                                        AddDriverDetails(
                                            section,
                                            snapshot));

                            column.Item()
                                .Element(
                                    section =>
                                        AddCoverageDetails(
                                            section,
                                            snapshot));

                            column.Item()
                                .Element(
                                    section =>
                                        AddAddOnDetails(
                                            section,
                                            snapshot));
                        });

                    page.Footer()
                        .AlignCenter()
                        .Text(text =>
                        {
                            text.Span(
                                "Generated from the immutable Policy contract. ");

                            text.Span(
                                $"Issued {policy.IssuedAtUtc:yyyy-MM-dd HH:mm} UTC");
                        });
                });
            });

        return document.GeneratePdf();
    }

    private static void AddCoverageDetails(
        IContainer container,
        PolicySnapshot snapshot)
    {
        container.Column(column =>
        {
            AddSectionHeading(
                column,
                "Coverages");

            column.Item()
                .Table(table =>
                {
                    table.ColumnsDefinition(
                        columns =>
                        {
                            columns.RelativeColumn(2);

                            columns.RelativeColumn();

                            columns.RelativeColumn();
                        });

                    table.Header(header =>
                    {
                        header.Cell()
                            .Element(HeaderCell)
                            .Text("Coverage");

                        header.Cell()
                            .Element(HeaderCell)
                            .Text("Limit");

                        header.Cell()
                            .Element(HeaderCell)
                            .Text("Deductible");
                    });

                    foreach (var coverage
                             in snapshot.Coverages)
                    {
                        table.Cell()

                        table.Cell()
                            .Element(BodyCell)
                            .Text(
                                coverage.ResolvedLimitAmount
                                    .ToString("0.00"));

    private static void AddAddOnDetails(
        IContainer container,
        PolicySnapshot snapshot)
    {
        container.Column(column =>
        {
            AddSectionHeading(
                column,
                "Add-ons");

            if (snapshot.AddOns.Count == 0)
            {
                column.Item()
                    .Text(
                        "No add-ons selected.");

                return;
            }

            column.Item()
                .Table(table =>
                {
                    table.ColumnsDefinition(
                        columns =>
                        {
                            columns.RelativeColumn(2);

                            columns.RelativeColumn();
                        });

                    table.Header(header =>
                    {
                        header.Cell()
                            .Element(HeaderCell)
                            .Text("Add-on");

                        header.Cell()
                            .Element(HeaderCell)
                            .Text("Annual Price");
                    });

                    foreach (var addOn
                             in snapshot.AddOns)
                    {
                        table.Cell()
                            .Element(BodyCell)
                            .Text(
                                addOn.AddOnName);

                        table.Cell()
                            .Element(BodyCell)
                            .Text(
                                addOn.AnnualPrice
                                    .ToString("0.00"));
                    }
                });
        });
    }

    private static void AddSectionHeading(
        ColumnDescriptor column,
        string title)
    {
        column.Item()
            .PaddingBottom(6)
            .Text(title)
            .Bold()
            .FontSize(14)
            .FontColor(
                Colors.Blue.Darken2);
    }

    private static void AddRow(
        TableDescriptor table,
        string label,
        string value)
    {
        table.Cell()
            .Element(BodyCell)
            .Text(label)
            .SemiBold();

        table.Cell()
            .Element(BodyCell)
            .Text(value);
    }

    private static IContainer HeaderCell(
        IContainer container)
    {
        return container
            .Background(
                Colors.Grey.Lighten3)
            .BorderBottom(1)
            .BorderColor(
                Colors.Grey.Lighten1)
            .Padding(6);
    }

    private static IContainer BodyCell(
        IContainer container)
    {
        return container
            .BorderBottom(1)
            .BorderColor(
                Colors.Grey.Lighten2)
            .Padding(6);
    }
}

    : IEntityTypeConfiguration<Installment>
{
    public void Configure(
        EntityTypeBuilder<Installment> builder)
    {
        builder.ToTable(
            "Installments",
            table =>
            {
                table.HasCheckConstraint(
                    "CK_Installments_Number_Positive",
                    "[InstallmentNumber] > 0");

                table.HasCheckConstraint(
                    "CK_Installments_Amount_Positive",
                    "[Amount] > 0");
            });

        builder.HasKey(
            installment => installment.Id);

        builder.HasIndex(
                installment =>
                    new
                    {
                        installment.PaymentScheduleId,
                        installment.InstallmentNumber
                    })
            .IsUnique();

        builder.Property(
                installment => installment.Amount)
            .HasPrecision(18, 2)
            .IsRequired();

        builder.Property(
                installment => installment.ContractStatus)
            .IsRequired();

        builder.HasOne(
                installment => installment.PaymentSchedule)
            .WithMany(
                schedule => schedule.Installments)
            .HasForeignKey(
                installment => installment.PaymentScheduleId)
            .OnDelete(
                DeleteBehavior.Cascade);
    }

            Guid installmentId)

        return await _dbContext
            .Installments
            .Include(
                installment =>
                    installment.PaymentSchedule)
                .ThenInclude(
                    schedule =>

            .SingleOrDefaultAsync(
                installment =>

                    installmentId);

        return await _dbContext
            .Installments
            .AsNoTracking()

                installment =>
                    installment.PaymentScheduleId ==
                    paymentScheduleId)
            .OrderBy(
                installment =>
                    installment.InstallmentNumber)
            .ToListAsync();
    }

        await _dbContext

                transaction);
    }

    {
        await _dbContext.SaveChangesAsync();
    }

        Func<Task> operation)

            operation);

        try
        {
            await operation();

            await transaction.CommitAsync();
        }
        catch
        {
            await transaction.RollbackAsync();

            throw;
        }
    }

namespace PolicyBillingService.Infrastructure.Repositories;

public class PolicyRepository : IPolicyRepository
{
    private readonly PolicyBillingDbContext _dbContext;

    public PolicyRepository(
        PolicyBillingDbContext dbContext)
    {
        _dbContext = dbContext;
    }
    public async Task<Policy?>
    GetByUnderwritingApplicationIdAsync(
        Guid underwritingApplicationId)
    {
        return await _dbContext
            .Policies
            .AsNoTracking()
            .SingleOrDefaultAsync(
                policy =>
                    policy.OriginatingUnderwritingApplicationId ==
                        underwritingApplicationId);
    }
    public async Task<IReadOnlyList<Policy>>
    GetByCustomerProfileIdAsync(
        Guid customerProfileId)
    {
        return await _dbContext
            .Policies
            .AsNoTracking()
            .Include(
                policy =>
                    policy.Snapshot)
            .Where(
                policy =>
                    policy.CustomerProfileId ==
                        customerProfileId)
            .OrderByDescending(
                policy =>
                    policy.IssuedAtUtc)
            .ToListAsync();
    }
    public async Task<Policy?> GetByIdAsync(
    Guid policyId)
    {
        return await _dbContext
            .Policies
            .AsNoTracking()
            .AsSplitQuery()
            .Include(
                policy =>
                    policy.Snapshot)
                .ThenInclude(
                    snapshot =>
                        snapshot.Coverages)
            .Include(
                policy =>
                    policy.Snapshot)
                .ThenInclude(
                    snapshot =>
                        snapshot.AddOns)
            .Include(
                policy =>
                    policy.PaymentSchedule)
                .ThenInclude(
                    schedule =>
                        schedule.Installments)
                    .ThenInclude(
                        installment =>
                            installment.Transactions)
            .Include(
                policy =>
                    policy.Document)
            .Include(
                policy =>
                    policy.StatusHistory)
            .SingleOrDefaultAsync(
                policy =>
                    policy.Id ==
                        policyId);
    }
    public async Task<bool> HasOverlappingPolicyAsync(
        Guid vehicleId,
        DateOnly effectiveDate,
        DateOnly expirationDate)
    {
        return await _dbContext
            .Policies
            .AsNoTracking()
            .AnyAsync(
                policy =>
                    policy.VehicleId == vehicleId &&
                    effectiveDate < policy.ExpirationDate &&
                    expirationDate > policy.EffectiveDate &&
                    (
                        policy.CancellationEffectiveAtUtc == null ||
                        DateOnly.FromDateTime(
                            policy.CancellationEffectiveAtUtc.Value)
                            > effectiveDate
                    ));
    }
}
