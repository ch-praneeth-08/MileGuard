using IdentityService.Application.Dtos.Admin;
using IdentityService.Application.Interfaces;
using IdentityService.Application.Interfaces.Repositories;
using IdentityService.Domain.Constants;
using IdentityService.Domain.Enums;

namespace IdentityService.Application.Services;

public class AdminIdentityService : IAdminIdentityService
{
    private readonly IUserRepository _userRepository;

    public AdminIdentityService(
        IUserRepository userRepository)
    {
        _userRepository = userRepository;
    }

    public async Task<IReadOnlyList<PendingInternalRegistrationDto>>
        GetPendingRegistrationsAsync()
    {
        var users =
            await _userRepository
                .GetPendingInternalUsersAsync();

        return users
            .Select(user =>
                new PendingInternalRegistrationDto
                {
                    Id = user.Id,
                    FirstName = user.FirstName,
                    LastName = user.LastName,
                    Email = user.Email ?? string.Empty,
                    RegisteredAtUtc = user.CreatedAtUtc
                })
            .ToList();
    }

    public async Task<InternalRegistrationDetailDto?>
        GetPendingRegistrationAsync(
            string id)
    {
        var user =
            await _userRepository.GetByIdAsync(id);

        if (user is null ||
            user.AccountType != AccountType.Internal ||
            user.AccountStatus != AccountStatus.Pending)
        {
            return null;
        }

        return new InternalRegistrationDetailDto
        {
            Id = user.Id,
            FirstName = user.FirstName,
            LastName = user.LastName,
            Email = user.Email ?? string.Empty,
            Status = user.AccountStatus.ToString(),
            RegisteredAtUtc = user.CreatedAtUtc
        };
    }

    public async Task<InternalRegistrationActionResult>
        ApproveAsync(
            string id,
            ApproveInternalRegistrationRequest request,
            string adminUserId)
    {
        var user =
            await _userRepository.GetByIdAsync(id);

        if (user is null)
        {
            return InternalRegistrationActionResult
                .RegistrationNotFound();
        }

        if (user.AccountType != AccountType.Internal ||
            user.AccountStatus != AccountStatus.Pending)
        {
            return InternalRegistrationActionResult
                .InvalidRegistrationState();
        }

        if (!IsPermittedOperationalRole(request.Role))
        {
            return InternalRegistrationActionResult.Failed(
                ["The selected role is not permitted."]);
        }

        var result =
    await _userRepository.ApproveInternalUserAsync(
        user,
        request.Role,
        adminUserId,
        DateTime.UtcNow);

        if (!result.Succeeded)
        {
            return InternalRegistrationActionResult.Failed(
                result.Errors);
        }

        return InternalRegistrationActionResult.Success();
    }

    public async Task<InternalRegistrationActionResult>
        RejectAsync(
            string id,
            string adminUserId)
    {
        var user =
            await _userRepository.GetByIdAsync(id);

        if (user is null)
        {
            return InternalRegistrationActionResult
                .RegistrationNotFound();
        }

        if (user.AccountType != AccountType.Internal ||
            user.AccountStatus != AccountStatus.Pending)
        {
            return InternalRegistrationActionResult
                .InvalidRegistrationState();
        }

        user.AccountStatus = AccountStatus.Rejected;
        user.RejectedAtUtc = DateTime.UtcNow;
        user.RejectedByUserId = adminUserId;

        var updateResult =
            await _userRepository.UpdateAsync(user);

        if (!updateResult.Succeeded)
        {
            return InternalRegistrationActionResult.Failed(
                updateResult.Errors);
        }

        return InternalRegistrationActionResult.Success();
    }

    private static bool IsPermittedOperationalRole(
        string role)
    {
        return role == ApplicationRoles.Agent ||
               role == ApplicationRoles.Underwriter ||
               role == ApplicationRoles.ClaimsAdjuster;
    }
}

using IdentityService.Application.Dtos.Admin;
using IdentityService.Application.Interfaces;
using IdentityService.Application.Interfaces.Repositories;
using IdentityService.Domain.Enums;

namespace IdentityService.Application.Services;

public class AdminUserService : IAdminUserService
{
    private readonly IUserRepository _userRepository;

    public AdminUserService(
        IUserRepository userRepository)
    {
        _userRepository = userRepository;
    }

    public async Task<IReadOnlyList<InternalUserListDto>>
        GetUsersAsync()
    {
        var users =
            await _userRepository
                .GetEstablishedInternalUsersAsync();

        var result = new List<InternalUserListDto>();

        foreach (var user in users)
        {
            var roles =
                await _userRepository.GetRolesAsync(user);

            var role =
                roles.FirstOrDefault();

            if (string.IsNullOrWhiteSpace(role))
            {
                continue;
            }

            result.Add(
                new InternalUserListDto
                {
                    Id = user.Id,
                    FirstName = user.FirstName,
                    LastName = user.LastName,
                    Email = user.Email ?? string.Empty,
                    Role = role,
                    Status = user.AccountStatus.ToString(),
                    TwoFactorStatus =
                        GetTwoFactorStatus(user)
                });
        }

        return result;
    }

    public async Task<InternalUserDetailDto?>
        GetUserAsync(string id)
    {
        var user =
            await _userRepository.GetByIdAsync(id);

        if (user is null ||
            user.AccountType != AccountType.Internal ||
            user.AccountStatus == AccountStatus.Pending ||
            user.AccountStatus == AccountStatus.Rejected)
        {
            return null;
        }

        var roles =
            await _userRepository.GetRolesAsync(user);

        var role =
            roles.FirstOrDefault();

        if (string.IsNullOrWhiteSpace(role))
        {
            return null;
        }

        return new InternalUserDetailDto
        {
            Id = user.Id,
            FirstName = user.FirstName,
            LastName = user.LastName,
            Email = user.Email ?? string.Empty,
            Role = role,
            Status = user.AccountStatus.ToString(),
            TwoFactorStatus =
                GetTwoFactorStatus(user),
            CreatedAtUtc = user.CreatedAtUtc,
            ApprovedAtUtc = user.ApprovedAtUtc,
            DeactivatedAtUtc = user.DeactivatedAtUtc
        };
    }

    private static string GetTwoFactorStatus(
        IdentityService.Domain.Entities.ApplicationUser user)
    {
        return user.TwoFactorEnabled
            ? "Enabled"
            : "Pending enrollment";
    }
}

    }

    public async Task<LoginResult> LoginAsync(
        LoginRequest request)
    {
        var email = request.Email.Trim();

        var user = await _userRepository.GetByEmailAsync(email);

        if (user is null)
        {
            return LoginResult.InvalidCredentials();
        }

        var passwordValid =
            await _userRepository.CheckPasswordAsync(
                user,
                request.Password);

        if (!passwordValid)
        {
            return LoginResult.InvalidCredentials();
        }

        if (user.AccountType == AccountType.Customer)
        {
            if (user.AccountStatus != AccountStatus.Active)
            {
                return LoginResult.InvalidCredentials();
            }

            var roles_ =
                await _userRepository.GetRolesAsync(user);

            var role = roles_.FirstOrDefault();

            if (string.IsNullOrWhiteSpace(role))
            {
                return LoginResult.InvalidCredentials();
            }

            var tokens =
                await _authSessionService.CreateSessionAsync(
                    user,
                    role);

            return new LoginResult
            {
                Status = LoginStatus.CustomerAuthenticated,
                UserId = user.Id,
                Email = user.Email,
                Role = role,
                Tokens = tokens
            };
        }

        if (user.AccountStatus == AccountStatus.Pending)
        {
            return new LoginResult
            {
                Status = LoginStatus.PendingApproval
            };
        }

        if (user.AccountStatus == AccountStatus.Rejected)
        {
            return new LoginResult
            {
                Status = LoginStatus.Rejected
            };
        }

        if (user.AccountStatus == AccountStatus.Inactive)
        {
            return new LoginResult
            {
                Status = LoginStatus.Inactive
            };
        }

        if (user.AccountStatus != AccountStatus.Active)
        {
            return LoginResult.InvalidCredentials();
        }

        var roles =
            await _userRepository.GetRolesAsync(user);

        var internalRole = roles.FirstOrDefault();

        if (string.IsNullOrWhiteSpace(internalRole))
        {
            return LoginResult.InvalidCredentials();
        }

        var twoFactorEnabled =
            await _userRepository
                .GetTwoFactorEnabledAsync(user);

        var preAuth =
            _preAuthTokenService.Create(user);

        return new LoginResult
        {
            Status = twoFactorEnabled
                ? LoginStatus.TwoFactorRequired
                : LoginStatus.TwoFactorSetupRequired,

            UserId = user.Id,
            Email = user.Email,
            Role = internalRole,
            PreAuth = preAuth
        };
    }
}

using IdentityService.Application.Dtos.Authentication;
using IdentityService.Application.Interfaces;
using IdentityService.Application.Interfaces.Repositories;
using IdentityService.Domain.Entities;
using IdentityService.Domain.Enums;

namespace IdentityService.Application.Services;

public class RefreshTokenService : IRefreshTokenService
{
    private readonly IRefreshSessionRepository
        _refreshSessionRepository;

    private readonly IUserRepository _userRepository;
    private readonly ITokenHashService _tokenHashService;
    private readonly ITokenService _tokenService;

    public RefreshTokenService(
        IRefreshSessionRepository refreshSessionRepository,
        IUserRepository userRepository,
        ITokenHashService tokenHashService,
        ITokenService tokenService)
    {
        _refreshSessionRepository =
            refreshSessionRepository;

        _userRepository = userRepository;
        _tokenHashService = tokenHashService;
        _tokenService = tokenService;
    }

    public async Task<RefreshTokenResult> RefreshAsync(
        string refreshToken)
    {
        if (string.IsNullOrWhiteSpace(refreshToken))
        {
            return RefreshTokenResult.Failed();
        }

        var tokenHash =
            _tokenHashService.Hash(refreshToken);

        var session =
            await _refreshSessionRepository
                .GetByTokenHashAsync(tokenHash);

        if (session is null)
        {
            return RefreshTokenResult.Failed();
        }

        if (session.RevokedAtUtc is not null)
        {
            return RefreshTokenResult.Failed();
        }

        if (session.AbsoluteExpiresAtUtc <= DateTime.UtcNow)
        {
            await _refreshSessionRepository.RevokeAsync(
                session,
                DateTime.UtcNow);

            return RefreshTokenResult.Failed();
        }

        var user =
            await _userRepository.GetByIdAsync(
                session.UserId);

        if (user is null ||
            user.AccountStatus != AccountStatus.Active)
        {
            await _refreshSessionRepository.RevokeAsync(
                session,
                DateTime.UtcNow);

            return RefreshTokenResult.Failed();
        }

        var roles =
            await _userRepository.GetRolesAsync(user);

        var role = roles.FirstOrDefault();

        if (string.IsNullOrWhiteSpace(role))
        {
            await _refreshSessionRepository.RevokeAsync(
                session,
                DateTime.UtcNow);

            return RefreshTokenResult.Failed();
        }

        if (user.AccountType == AccountType.Internal)
        {
            var twoFactorEnabled =
                await _userRepository
                    .GetTwoFactorEnabledAsync(user);

            if (!twoFactorEnabled)
            {
                await _refreshSessionRepository.RevokeAsync(
                    session,
                    DateTime.UtcNow);

                return RefreshTokenResult.Failed();
            }
        }

        var tokens =
            _tokenService.CreateRefreshedToken(
                user,
                role,
                session.AbsoluteExpiresAtUtc);

        var replacementSession =
            new RefreshSession
            {
                Id = Guid.NewGuid(),
                UserId = user.Id,

                TokenHash =
                    _tokenHashService.Hash(
                        tokens.RefreshToken),

                SessionStartedAtUtc =
                    session.SessionStartedAtUtc,

                AbsoluteExpiresAtUtc =
                    session.AbsoluteExpiresAtUtc
            };

        await _refreshSessionRepository.RotateAsync(
            session,
            replacementSession);

        return new RefreshTokenResult
        {
            Succeeded = true,

            AccessToken =
                tokens.AccessToken,

            RefreshToken =
                tokens.RefreshToken,

            AccessTokenExpiresAtUtc =
                tokens.AccessTokenExpiresAtUtc,

            RefreshSessionExpiresAtUtc =
                session.AbsoluteExpiresAtUtc
        };
    }
}

using IdentityService.Application.Dtos.Authentication;
using IdentityService.Application.Interfaces;
using IdentityService.Application.Interfaces.Repositories;
using IdentityService.Domain.Enums;

namespace IdentityService.Application.Services;

public class TwoFactorService : ITwoFactorService
{
    private readonly IUserRepository _userRepository;
    private readonly IAuthSessionService _authSessionService;

    public TwoFactorService(
        IUserRepository userRepository,
        IAuthSessionService authSessionService)
    {
        _userRepository = userRepository;
        _authSessionService = authSessionService;
    }

    public async Task<TwoFactorEnrollResult?>
        BeginEnrollmentAsync(string userId)
    {
        var user =
            await _userRepository.GetByIdAsync(userId);

        if (user is null ||
            user.AccountType != AccountType.Internal ||
            user.AccountStatus != AccountStatus.Active)
        {
            return null;
        }

        var twoFactorEnabled =
            await _userRepository
                .GetTwoFactorEnabledAsync(user);

        if (twoFactorEnabled)
        {
            return null;
        }

        var key =
            await _userRepository
                .GetAuthenticatorKeyAsync(user);

        if (string.IsNullOrWhiteSpace(key))
        {
            await _userRepository
                .ResetAuthenticatorKeyAsync(user);

            key =
                await _userRepository
                    .GetAuthenticatorKeyAsync(user);
        }

        if (string.IsNullOrWhiteSpace(key))
        {
            return null;
        }

        var email = user.Email ?? string.Empty;

        var authenticatorUri = GenerateAuthenticatorUri(
            email,
            key);

        return new TwoFactorEnrollResult
        {
            SharedKey = FormatKey(key),
            AuthenticatorUri = authenticatorUri
        };
    }

    public async Task<TwoFactorVerificationResult>
        VerifyAsync(
            string userId,
            string code)
    {
        var user =
            await _userRepository.GetByIdAsync(userId);

        if (user is null ||
            user.AccountType != AccountType.Internal ||
            user.AccountStatus != AccountStatus.Active)
        {
            return new TwoFactorVerificationResult
            {
                Error = "Two-factor verification is unavailable."
            };
        }

        var roles =
            await _userRepository.GetRolesAsync(user);

        var role = roles.FirstOrDefault();

        if (string.IsNullOrWhiteSpace(role))
        {
            return new TwoFactorVerificationResult
            {
                Error = "Two-factor verification is unavailable."
            };
        }

        var normalizedCode =
            code.Replace(" ", string.Empty)
                .Replace("-", string.Empty);

        var valid =
            await _userRepository
                .VerifyAuthenticatorCodeAsync(
                    user,
                    normalizedCode);

        if (!valid)
        {
            return new TwoFactorVerificationResult
            {
                Error = "Invalid verification code."
            };
        }

        var twoFactorEnabled =
            await _userRepository
                .GetTwoFactorEnabledAsync(user);

        if (!twoFactorEnabled)
        {
            var enableResult =
                await _userRepository
                    .SetTwoFactorEnabledAsync(
                        user,
                        true);

            if (!enableResult.Succeeded)
            {
                return new TwoFactorVerificationResult
                {
                    Error =
                        "Unable to enable two-factor authentication."
                };
            }
        }

        var tokens =
            await _authSessionService.CreateSessionAsync(
                user,
                role);

        return new TwoFactorVerificationResult
        {
            Succeeded = true,
            Email = user.Email,
            Role = role,
            Tokens = tokens
        };
    }

    private static string GenerateAuthenticatorUri(
        string email,
        string unformattedKey)
    {
        const string issuer = "AutoInsurance";

        return
            $"otpauth://totp/" +
            $"{Uri.EscapeDataString(issuer)}:" +
            $"{Uri.EscapeDataString(email)}" +
            $"?secret={unformattedKey}" +
            $"&issuer={Uri.EscapeDataString(issuer)}" +
            $"&digits=6";
    }

    private static string FormatKey(string unformattedKey)
    {
        var result = new System.Text.StringBuilder();

        var currentPosition = 0;

        while (currentPosition + 4 < unformattedKey.Length)
        {
            result
                .Append(
                    unformattedKey.AsSpan(
                        currentPosition,
                        4))
                .Append(' ');

            currentPosition += 4;
        }

        if (currentPosition < unformattedKey.Length)
        {
            result.Append(
                unformattedKey.AsSpan(currentPosition));
        }

        return result.ToString().ToLowerInvariant();
    }
}
