        _userRepositoryMock;

    private readonly AdminIdentityService _service;

            new Mock<IUserRepository>();

                _userRepositoryMock.Object);

    [Fact]
    public async Task ApproveAsync_AdminRole_IsRejected()
    {
        var user = CreatePendingInternalUser();

        _userRepositoryMock

            .ReturnsAsync(user);

            new ApproveInternalRegistrationRequest
            {
                Role = ApplicationRoles.Admin

            repository =>
                repository.ApproveInternalUserAsync(
                    It.IsAny<ApplicationUser>(),
                    It.IsAny<string>(),
                    It.IsAny<string>(),
                    It.IsAny<DateTime>()),
            Times.Never);
    }

    [Fact]
    public async Task ApproveAsync_AgentRole_ApprovesPendingUser()
    {
        var user = CreatePendingInternalUser();

        _userRepositoryMock
            .Setup(repository =>
                repository.GetByIdAsync(user.Id))
            .ReturnsAsync(user);

        _userRepositoryMock
            .Setup(repository =>
                repository.ApproveInternalUserAsync(
                    user,
                    ApplicationRoles.Agent,
                    "admin-user-id",
                    It.IsAny<DateTime>()))
            .ReturnsAsync(
                UserOperationResult.Success());

        var request =
            new ApproveInternalRegistrationRequest
            {
                Role = ApplicationRoles.Agent
            };

        var result =
            await _service.ApproveAsync(
                user.Id,
                request,
                "admin-user-id");

        Assert.True(result.Succeeded);

        _userRepositoryMock.Verify(
            repository =>
                repository.ApproveInternalUserAsync(
                    user,
                    ApplicationRoles.Agent,
                    "admin-user-id",
                    It.IsAny<DateTime>()),
            Times.Once);
    }

    [Fact]
    public async Task ApproveAsync_NonPendingUser_IsRejected()
    {
        var user = CreatePendingInternalUser();

        user.AccountStatus = AccountStatus.Active;

        _userRepositoryMock
            .Setup(repository =>
                repository.GetByIdAsync(user.Id))
            .ReturnsAsync(user);

        var request =
            new ApproveInternalRegistrationRequest
            {
                Role = ApplicationRoles.Agent
            };

using IdentityService.Application.Dtos.Authentication;
using IdentityService.Application.Interfaces;
using IdentityService.Application.Interfaces.Repositories;
using IdentityService.Application.Services;
using IdentityService.Domain.Entities;
using Moq;

namespace IdentityService.Tests.Unit.Services;

public class AuthSessionServiceTests
{
    private readonly Mock<IRefreshSessionRepository>
        _refreshRepositoryMock;

    private readonly Mock<ITokenService>
        _tokenServiceMock;

    private readonly Mock<ITokenHashService>
        _tokenHashServiceMock;

    private readonly AuthSessionService _service;

    public AuthSessionServiceTests()
    {
        _refreshRepositoryMock =
            new Mock<IRefreshSessionRepository>();

        _tokenServiceMock =
            new Mock<ITokenService>();

        _tokenHashServiceMock =
            new Mock<ITokenHashService>();

        _service = new AuthSessionService(
            _refreshRepositoryMock.Object,
            _tokenServiceMock.Object,
            _tokenHashServiceMock.Object);
    }

    [Fact]
    public async Task CreateSessionAsync_RevokesExistingActiveSession()
    {
        var user = new ApplicationUser
        {
            Id = "user-id"
        };

        var existingSession =
            new RefreshSession
            {
                Id = Guid.NewGuid(),
                UserId = user.Id,
                TokenHash = "old-hash",
                SessionStartedAtUtc =
                    DateTime.UtcNow.AddHours(-1),
                AbsoluteExpiresAtUtc =
                    DateTime.UtcNow.AddDays(6)
            };

        var absoluteExpiry =
            DateTime.UtcNow.AddDays(7);

        _refreshRepositoryMock
            .Setup(repository =>
                repository.GetActiveByUserIdAsync(
                    user.Id))
            .ReturnsAsync(existingSession);

        _tokenServiceMock
            .Setup(service =>
                service.CreateToken(
                    user,
                    "Customer"))
            .Returns(
                new TokenResult
                {
                    AccessToken = "access-token",
                    RefreshToken = "refresh-token",
                    AccessTokenExpiresAtUtc =
                        DateTime.UtcNow.AddMinutes(15),
                    RefreshSessionExpiresAtUtc =
                        absoluteExpiry
                });

        _tokenHashServiceMock
            .Setup(service =>
                service.Hash("refresh-token"))
            .Returns("new-hash");

        await _service.CreateSessionAsync(
            user,
            "Customer");

        Assert.NotNull(
            existingSession.RevokedAtUtc);

        _refreshRepositoryMock.Verify(
            repository =>
                repository.AddAsync(
                    It.Is<RefreshSession>(session =>
                        session.UserId == user.Id &&
                        session.TokenHash ==
                            "new-hash")),
            Times.Once);

        _refreshRepositoryMock.Verify(
            repository =>
                repository.SaveChangesAsync(),
            Times.Once);
    }
}

{

    {

    }

    [Fact]
    public async Task RefreshAsync_ValidToken_RotatesAndPreservesAbsoluteExpiry()
    {
        var sessionStartedAt =
            DateTime.UtcNow.AddDays(-2);

        var absoluteExpiry =
            DateTime.UtcNow.AddDays(5);

        var session = new RefreshSession
        {
            Id = Guid.NewGuid(),
            UserId = "user-id",
            TokenHash = "old-hash",
            SessionStartedAtUtc = sessionStartedAt,
            AbsoluteExpiresAtUtc = absoluteExpiry
        };

        var user = CreateActiveCustomer();

        _tokenHashServiceMock
            .Setup(service =>
                service.Hash("old-refresh-token"))
            .Returns("old-hash");

        _refreshRepositoryMock
            .Setup(repository =>
                repository.GetByTokenHashAsync("old-hash"))
            .ReturnsAsync(session);

        _userRepositoryMock
            .Setup(repository =>
                repository.GetByIdAsync(session.UserId))
            .ReturnsAsync(user);

        _userRepositoryMock
            .Setup(repository =>
                repository.GetRolesAsync(user))
            .ReturnsAsync(["Customer"]);

        _tokenServiceMock
            .Setup(service =>
                service.CreateRefreshedToken(
                    user,
                    "Customer",
                    absoluteExpiry))
            .Returns(
                new TokenResult
                {
                    AccessToken = "new-access-token",
                    RefreshToken = "new-refresh-token",
                    AccessTokenExpiresAtUtc =
                        DateTime.UtcNow.AddMinutes(15),
                    RefreshSessionExpiresAtUtc =
                        absoluteExpiry
                });

        _tokenHashServiceMock
            .Setup(service =>
                service.Hash("new-refresh-token"))
            .Returns("new-hash");

        var result =
            await _service.RefreshAsync(
                "old-refresh-token");

        Assert.True(result.Succeeded);

        Assert.Equal(
            "new-access-token",
            result.AccessToken);

        Assert.Equal(
            "new-refresh-token",
            result.RefreshToken);

        Assert.Equal(
            absoluteExpiry,
            result.RefreshSessionExpiresAtUtc);

        _refreshRepositoryMock.Verify(
            repository =>
                repository.RotateAsync(
                    session,
                    It.Is<RefreshSession>(
                        replacement =>
                            replacement.UserId == user.Id &&
                            replacement.TokenHash == "new-hash" &&
                            replacement.SessionStartedAtUtc ==
                                sessionStartedAt &&
                            replacement.AbsoluteExpiresAtUtc ==
                                absoluteExpiry)),
            Times.Once);

    [Fact]
    public async Task RegisterCustomerAsync_CreatesActiveCustomerWithCustomerRole()
    {
        var request = new RegisterCustomerRequest
        {
            FirstName = "Test",
            LastName = "Customer",
            Email = "customer@example.com",
            Password = "Customer@123",
            ConfirmPassword = "Customer@123"
        };

        _userRepositoryMock
            .Setup(repository =>
                repository.GetByEmailAsync(request.Email))
            .ReturnsAsync((ApplicationUser?)null);

        _userRepositoryMock
            .Setup(repository =>
                repository.CreateAsync(
                    It.IsAny<ApplicationUser>(),
                    request.Password))
            .ReturnsAsync(UserOperationResult.Success());

        _userRepositoryMock
            .Setup(repository =>
                repository.AddToRoleAsync(
                    It.IsAny<ApplicationUser>(),
                    ApplicationRoles.Customer))
            .ReturnsAsync(UserOperationResult.Success());

        var result =
            await _service.RegisterCustomerAsync(request);

        Assert.True(result.Succeeded);

        _userRepositoryMock.Verify(
            repository =>
                repository.CreateAsync(
                    It.Is<ApplicationUser>(user =>
                        user.AccountType ==
                            AccountType.Customer &&
                        user.AccountStatus ==
                            AccountStatus.Active &&
                        user.FirstName == "Test" &&
                        user.LastName == "Customer"),
                    request.Password),
            Times.Once);

        _userRepositoryMock.Verify(
            repository =>
                repository.AddToRoleAsync(
                    It.IsAny<ApplicationUser>(),
                    ApplicationRoles.Customer),
            Times.Once);
    }
