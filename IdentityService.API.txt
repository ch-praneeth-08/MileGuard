using System.Security.Claims;
using FluentValidation;
using IdentityService.Application.Dtos.Admin;
using IdentityService.Application.Interfaces;
using IdentityService.Domain.Constants;
using Microsoft.AspNetCore.Authorization;
using Microsoft.AspNetCore.Mvc;

namespace IdentityService.API.Controllers;

[ApiController]
[Route("api/admin/internal-registrations")]
[Authorize(Roles = ApplicationRoles.Admin)]
public class AdminInternalRegistrationsController
    : ControllerBase
{
    private readonly IAdminIdentityService
        _adminIdentityService;

    private readonly IValidator<ApproveInternalRegistrationRequest>
        _approveValidator;

    public AdminInternalRegistrationsController(
        IAdminIdentityService adminIdentityService,
        IValidator<ApproveInternalRegistrationRequest>
            approveValidator)
    {
        _adminIdentityService = adminIdentityService;
        _approveValidator = approveValidator;
    }

    [HttpGet]
    public async Task<IActionResult>
        GetPendingRegistrations()
    {
        var registrations =
            await _adminIdentityService
                .GetPendingRegistrationsAsync();

        return Ok(registrations);
    }

    [HttpGet("{id}")]
    public async Task<IActionResult>
        GetPendingRegistration(string id)
    {
        var registration =
            await _adminIdentityService
                .GetPendingRegistrationAsync(id);

        if (registration is null)
        {
            return NotFound(new
            {
                message =
                    "Pending registration was not found."
            });
        }

        return Ok(registration);
    }

    [HttpPost("{id}/approve")]
    public async Task<IActionResult>
        Approve(
            string id,
            ApproveInternalRegistrationRequest request)
    {
        var validationResult =
            await _approveValidator
                .ValidateAsync(request);

        if (!validationResult.IsValid)
        {
            var errors =
                validationResult.Errors
                    .GroupBy(error => error.PropertyName)
                    .ToDictionary(
                        group => group.Key,
                        group => group
                            .Select(error => error.ErrorMessage)
                            .ToArray());

            return BadRequest(new
            {
                message = "Validation failed.",
                errors
            });
        }

        var adminUserId = GetCurrentUserId();

        if (adminUserId is null)
        {
            return Unauthorized();
        }

        var result =
            await _adminIdentityService.ApproveAsync(
                id,
                request,
                adminUserId);

        if (result.NotFound)
        {
            return NotFound(new
            {
                message =
                    "Registration was not found."
            });
        }

        if (result.InvalidState)
        {
            return Conflict(new
            {
                message =
                    "Registration is no longer pending."
            });
        }

        if (!result.Succeeded)
        {
            return BadRequest(new
            {
                message =
                    "Registration could not be approved.",
                errors = result.Errors
            });
        }

        return Ok(new
        {
            message =
                "Registration approved and activated."
        });
    }

    [HttpPost("{id}/reject")]
    public async Task<IActionResult>
        Reject(string id)
    {
        var adminUserId = GetCurrentUserId();

        if (adminUserId is null)
        {
            return Unauthorized();
        }

        var result =
            await _adminIdentityService.RejectAsync(
                id,
                adminUserId);

        if (result.NotFound)
        {
            return NotFound(new
            {
                message =
                    "Registration was not found."
            });
        }

        if (result.InvalidState)
        {
            return Conflict(new
            {
                message =
                    "Registration is no longer pending."
            });
        }

        if (!result.Succeeded)
        {
            return BadRequest(new
            {
                message =
                    "Registration could not be rejected.",
                errors = result.Errors
            });
        }

        return Ok(new
        {
            message = "Registration rejected."
        });
    }

    private string? GetCurrentUserId()
    {
        return User.FindFirstValue(
            ClaimTypes.NameIdentifier);
    }
}

                message = "Registration failed.",
                errors = result.Errors
            });
        }

        return Ok(new
        {
            message = "Registration successful."
        });
    }

    [HttpPost("login")]
    public async Task<IActionResult> Login(
    LoginRequest request)
    {
        var validationResult =
            await _loginValidator.ValidateAsync(request);

        if (!validationResult.IsValid)
        {
            return BadRequest(new
            {
                message = "Validation failed.",
                errors = GetValidationErrors(validationResult)
            });
        }

        var result =
            await _authenticationService.LoginAsync(request);

        if (result.Status == LoginStatus.InvalidCredentials)
        {
            return Unauthorized(new
            {
                message = "Invalid email or password."
            });
        }

        if (result.Status == LoginStatus.PendingApproval)
        {
            return Ok(new
            {
                status = "Pending",
                message = "Account awaiting approval."
            });
        }

        if (result.Status == LoginStatus.Rejected)
        {
            return Ok(new
            {
                status = "Rejected",
                message = "Internal access not approved."
            });
        }

        if (result.Status == LoginStatus.Inactive)
        {
            return Ok(new
            {
                status = "Inactive",
                message =
                    "Your internal account is currently inactive."
            });
        }

        if (result.Status == LoginStatus.TwoFactorSetupRequired &&
    result.PreAuth is not null)
        {
            return Ok(new
            {
                status = "TwoFactorSetupRequired",
                preAuthToken = result.PreAuth.Token,
                preAuthTokenExpiresAtUtc =
                    result.PreAuth.ExpiresAtUtc,
                result.Email,
                result.Role
            });
        }

        if (result.Status == LoginStatus.TwoFactorRequired &&
    result.PreAuth is not null)
        {
            return Ok(new
            {
                status = "TwoFactorRequired",
                preAuthToken = result.PreAuth.Token,
                preAuthTokenExpiresAtUtc =
                    result.PreAuth.ExpiresAtUtc,
                result.Email,
                result.Role
            });
        }

        if (result.Status == LoginStatus.CustomerAuthenticated &&
            result.Tokens is not null)
        {
            SetRefreshCookie(
                result.Tokens.RefreshToken,
                result.Tokens.RefreshSessionExpiresAtUtc);

            return Ok(new
            {
                status = "Authenticated",
                accessToken = result.Tokens.AccessToken,
                accessTokenExpiresAtUtc =
                    result.Tokens.AccessTokenExpiresAtUtc,
                result.Email,
                result.Role
            });
        }

        return Unauthorized();
    }
    [HttpPost("2fa/enroll")]
    public async Task<IActionResult> EnrollTwoFactor(
    [FromHeader(Name = "X-Pre-Auth-Token")]
    string preAuthToken)
    {
        var userId = ValidatePreAuthToken(preAuthToken);

        if (userId is null)
        {
            return Unauthorized(new
            {
                message =
                    "Invalid or expired pre-authentication token."
            });
        }

        var result =
            await _twoFactorService.BeginEnrollmentAsync(
                userId);

        if (result is null)
        {
            return BadRequest(new
            {
                message =
                    "Two-factor enrollment is not available."
            });
        }

        return Ok(result);
    }

    [HttpPost("refresh")]
    public async Task<IActionResult> Refresh()
    {
        if (!Request.Cookies.TryGetValue(
                "refreshToken",
                out var refreshToken))
        {
            ClearRefreshCookie();

            return Unauthorized(new
            {
                message =
                    "Refresh session is invalid or expired."
            });
        }

        var result =
            await _refreshTokenService
                .RefreshAsync(refreshToken);

        if (!result.Succeeded ||
            string.IsNullOrWhiteSpace(result.RefreshToken) ||
            string.IsNullOrWhiteSpace(result.AccessToken) ||
            result.RefreshSessionExpiresAtUtc is null)
        {
            ClearRefreshCookie();

            return Unauthorized(new
            {
                message =
                    "Refresh session is invalid or expired."
            });
        }

        SetRefreshCookie(
            result.RefreshToken,
            result.RefreshSessionExpiresAtUtc.Value);

        return Ok(new
        {
            accessToken = result.AccessToken,
            accessTokenExpiresAtUtc =
                result.AccessTokenExpiresAtUtc
        });
    }
    [HttpPost("logout")]
    public async Task<IActionResult> Logout()
    {
        if (Request.Cookies.TryGetValue(
                "refreshToken",
                out var refreshToken))
        {
            await _logoutService.LogoutAsync(
                refreshToken);
        }

        ClearRefreshCookie();

        return Ok(new
        {
            message = "Signed out successfully."
        });
    }

    }
    [HttpPost("2fa/verify")]
    public async Task<IActionResult> VerifyTwoFactor(
    TwoFactorVerifyRequest request,
    [FromHeader(Name = "X-Pre-Auth-Token")]
    string preAuthToken)
    {
        var validationResult =
            await _twoFactorVerifyValidator
                .ValidateAsync(request);

        if (!validationResult.IsValid)
        {
            return BadRequest(new
            {
                message = "Validation failed.",
                errors =
                    GetValidationErrors(validationResult)
            });
        }

        var userId =
            ValidatePreAuthToken(preAuthToken);

        if (userId is null)
        {
            return Unauthorized(new
            {
                message =
                    "Invalid or expired pre-authentication token."
            });
        }

        var result =
            await _twoFactorService.VerifyAsync(
                userId,
                request.Code);

        if (!result.Succeeded ||
            result.Tokens is null)
        {
            return BadRequest(new
            {
                message =
                    result.Error ??
                    "Two-factor verification failed."
            });
        }

        SetRefreshCookie(
            result.Tokens.RefreshToken,
            result.Tokens.RefreshSessionExpiresAtUtc);

        return Ok(new
        {
            status = "Authenticated",
            accessToken =
                result.Tokens.AccessToken,

            accessTokenExpiresAtUtc =
                result.Tokens.AccessTokenExpiresAtUtc,

            result.Email,
            result.Role
        });
    }
    private string? ValidatePreAuthToken(
    string preAuthToken)
    {
        if (string.IsNullOrWhiteSpace(preAuthToken))
        {
            return null;
        }

        var tokenHandler =
            new JwtSecurityTokenHandler();

        var key = Encoding.UTF8.GetBytes(
            _jwtOptions.Key);

        try
        {
            var principal =
                tokenHandler.ValidateToken(
                    preAuthToken,
                    new TokenValidationParameters
                    {
                        ValidateIssuer = true,
                        ValidIssuer = _jwtOptions.Issuer,

                        ValidateAudience = true,
                        ValidAudience = _jwtOptions.Audience,

                        ValidateIssuerSigningKey = true,
                        IssuerSigningKey =
                            new SymmetricSecurityKey(key),

                        ValidateLifetime = true,
                        ClockSkew = TimeSpan.Zero
                    },
                    out _);

            var purpose =
                principal.FindFirst(
                    PreAuthTokenService.PurposeClaim)
                ?.Value;

            if (purpose !=
                PreAuthTokenService.PurposeValue)
            {
                return null;
            }

            return principal.FindFirst(
            JwtRegisteredClaimNames.Sub)?.Value
        ?? principal.FindFirst(
            ClaimTypes.NameIdentifier)?.Value;
        }
        catch
        {
            return null;
        }
    }
    private static Dictionary<string, string[]> GetValidationErrors(
        FluentValidation.Results.ValidationResult validationResult)
    {
        return validationResult.Errors
            .GroupBy(error => error.PropertyName)
            .ToDictionary(
                group => group.Key,
                group => group
                    .Select(error => error.ErrorMessage)
                    .ToArray());
    }
    private void ClearRefreshCookie()
    {
        Response.Cookies.Delete(
            "refreshToken",
            new CookieOptions
            {
                HttpOnly = true,
                Secure = true,
                SameSite = SameSiteMode.None,
                Path = "/api/auth",
                IsEssential = true
            });
    }
    private void SetRefreshCookie(
    string refreshToken,
    DateTime expiresAtUtc)
    {
        Response.Cookies.Append(
            "refreshToken",
            refreshToken,
            new CookieOptions
            {
                HttpOnly = true,
                Secure = true,
                SameSite = SameSiteMode.None,
                Path = "/api/auth",
                Expires =
                    new DateTimeOffset(
                        expiresAtUtc),
                IsEssential = true
            });
    }
}

namespace IdentityService.API.Middleware;

public class CorrelationIdMiddleware
{
    private const string HeaderName = "X-Correlation-Id";

    private readonly RequestDelegate _next;

    public CorrelationIdMiddleware(
        RequestDelegate next)
    {
        _next = next;
    }

    public async Task InvokeAsync(
        HttpContext context)
    {
        var correlationId =
            context.Request.Headers.TryGetValue(
                HeaderName,
                out var existingCorrelationId)
                && !string.IsNullOrWhiteSpace(
                    existingCorrelationId)
            ? existingCorrelationId.ToString()
            : Guid.NewGuid().ToString();

        context.TraceIdentifier = correlationId;

        context.Response.Headers[HeaderName] =
            correlationId;

        await _next(context);
    }
}

namespace IdentityService.API.Middleware;

public class ExceptionHandlingMiddleware
{
    private readonly RequestDelegate _next;

    private readonly ILogger<ExceptionHandlingMiddleware>
        _logger;

    public ExceptionHandlingMiddleware(
        RequestDelegate next,
        ILogger<ExceptionHandlingMiddleware> logger)
    {
        _next = next;
        _logger = logger;
    }

    public async Task InvokeAsync(
        HttpContext context)
    {
        try
        {
            await _next(context);
        }
        catch (Exception exception)
        {
            _logger.LogError(
                exception,
                "Unhandled exception. CorrelationId: {CorrelationId}",
                context.TraceIdentifier);

            await WriteErrorResponseAsync(context);
        }
    }

    private static async Task WriteErrorResponseAsync(
        HttpContext context)
    {
        context.Response.StatusCode =
            (int)HttpStatusCode.InternalServerError;

        context.Response.ContentType =
            "application/json";

        var response = new
        {
            message =
                "An unexpected error occurred.",
            correlationId =
                context.TraceIdentifier
        };

        var json =
            JsonSerializer.Serialize(response);

        await context.Response.WriteAsync(json);
    }
}
