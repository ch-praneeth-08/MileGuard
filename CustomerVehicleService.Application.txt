    public AgentAssignmentService(
        IAgentAssignmentRepository assignmentRepository,

    {
        _assignmentRepository =
            assignmentRepository;

        _identityServiceClient =
            identityServiceClient;
    }

    public async Task<IReadOnlyList<UnassignedCustomerDto>>
        GetUnassignedCustomersAsync()

        var profiles =

                    IdentityUserId =

                            .ToString(),

                    CompletedAtUtc =

                        ?? profile.CreatedAtUtc

    }
    public async Task<AssignmentReviewCustomerDto?>
     GetCustomerForAssignmentAsync(
         Guid customerProfileId)
    {
        var profile =
            await _assignmentRepository
                .GetCustomerProfileAsync(
                    customerProfileId);

        if (profile is null)
        {
            return null;
        }

        var isAssigned =
            await _assignmentRepository
                .AssignmentExistsAsync(
                    customerProfileId);

        return new AssignmentReviewCustomerDto
        {
            CustomerProfileId =
                profile.Id,

            IdentityUserId =
                profile.IdentityUserId,

            DateOfBirth =
                profile.DateOfBirth,

            Phone =
                profile.Phone,

            City =
                profile.City,

            State =
                profile.State,

            PostalCode =
                profile.PostalCode,

            AreaType =

            IsAssigned =
                isAssigned
        };

    public async Task<EligibleAgentsResult>
        GetEligibleAgentsAsync()
    {
        return await _identityServiceClient
            .GetEligibleAgentsAsync();
    }

    public async Task<AssignAgentResult>
        AssignAgentAsync(
            Guid customerProfileId,
            string adminIdentityUserId,
            AssignAgentRequest request)
    {
        var profile =
            await _assignmentRepository
                .GetCustomerProfileAsync(
                    customerProfileId);

        if (profile is null)
        {
            return AssignAgentResult
                .CustomerMissing();
        }

        if (
            profile.ProfileStatus !=
            ProfileStatus.Complete
        )
        {
            return AssignAgentResult
                .Incomplete();
        }

        var alreadyAssigned =
            await _assignmentRepository
                .AssignmentExistsAsync(
                    customerProfileId);

        if (alreadyAssigned)
        {
            return AssignAgentResult
                .Duplicate();
        }

        /*
         * Revalidate eligibility at assignment time.
         * Do not trust the Agent list previously rendered
         * to the Admin.
         */
        var agentResult =
            await _identityServiceClient
                .GetEligibleAgentAsync(
                    request.AgentIdentityUserId);

        if (
            !agentResult.Succeeded &&
            agentResult.NotEligible
        )
        {
            return AssignAgentResult
                .Ineligible();
        }

        if (!agentResult.Succeeded)
        {
            return AssignAgentResult
                .DependencyUnavailable();
        }

        var assignment =
            new AgentAssignment
            {
                Id = Guid.NewGuid(),

                CustomerProfileId =
                    customerProfileId,

                AgentIdentityUserId =
                    request.AgentIdentityUserId,

                AssignedByIdentityUserId =
                    adminIdentityUserId,

                AssignedAtUtc =
                    DateTime.UtcNow
            };

        await _assignmentRepository
            .AddAsync(assignment);

        try
        {
            await _assignmentRepository
                .SaveChangesAsync();
        }
        catch (Exception)
        {
            /*
             * CustomerProfileId has a unique index.
             *
             * If two Admin requests race, one succeeds
             * and the other is reported as a conflict.
             */
            return AssignAgentResult
                .Duplicate();
        }

        return AssignAgentResult.Success(
            new AgentAssignmentDto
            {
                Id =
                    assignment.Id,

                CustomerProfileId =
                    assignment.CustomerProfileId,

                AgentIdentityUserId =
                    assignment.AgentIdentityUserId,

                AssignedByIdentityUserId =
                    assignment
                        .AssignedByIdentityUserId,

                AssignedAtUtc =
                    assignment.AssignedAtUtc
            });
    }
}

        var identitySummary =

    }

    public async Task<CustomerProfileDto?>
        GetAsync(
            string identityUserId)
    {
        var profile =
            await _profileRepository
                .GetByIdentityUserIdAsync(
                    identityUserId);

        return profile is null
            ? null
            : ToDto(profile);
    }

    public async Task<ProfileOperationResult>
        CreateAsync(
            string identityUserId,
            CreateCustomerProfileRequest request)
    {
        var existing =
            await _profileRepository
                .GetByIdentityUserIdAsync(
                    identityUserId);

        if (existing is not null)
        {
            return ProfileOperationResult.Duplicate();
        }

        var now = DateTime.UtcNow;

        var profile =
            new CustomerProfile
            {
                Id = Guid.NewGuid(),

                IdentityUserId =
                    identityUserId,

                DateOfBirth =
                    request.DateOfBirth,

                Phone =
                    NormalizeRequired(request.Phone),

                AddressLine1 =
                    NormalizeRequired(
                        request.AddressLine1),

                AddressLine2 =
                    NormalizeOptional(
                        request.AddressLine2),

                City =
                    NormalizeRequired(request.City),

                State =
                    NormalizeRequired(request.State),

                PostalCode =
                    NormalizeRequired(
                        request.PostalCode),

                AreaType =
                    request.AreaType,

                CreatedAtUtc = now
            };

        /*
         * Status is never taken from the request.
         * It is derived from data owned by this service.
         */
        profile.ProfileStatus =
            IsComplete(profile)
                ? ProfileStatus.Complete
                : ProfileStatus.Incomplete;

        if (
            profile.ProfileStatus ==
            ProfileStatus.Complete
        )
        {
            profile.CompletedAtUtc = now;
        }

        await _profileRepository.AddAsync(
            profile);

        await _profileRepository.SaveChangesAsync();

        return ProfileOperationResult.Success(
            ToDto(profile));
    }

    private static bool IsComplete(
        CustomerProfile profile)
    {
        return
            profile.DateOfBirth is not null &&
            !string.IsNullOrWhiteSpace(
                profile.Phone) &&
            !string.IsNullOrWhiteSpace(
                profile.AddressLine1) &&
            !string.IsNullOrWhiteSpace(
                profile.City) &&
            !string.IsNullOrWhiteSpace(
                profile.State) &&
            !string.IsNullOrWhiteSpace(
                profile.PostalCode) &&
            profile.AreaType is not null;
    }

    private static string NormalizeRequired(
        string value)
    {
        return value.Trim();
    }

    private static string? NormalizeOptional(
        string? value)
    {
        return string.IsNullOrWhiteSpace(value)
            ? null
            : value.Trim();
    }

        profile.State =
            NormalizeRequired(
                request.State);

        profile.PostalCode =
            NormalizeRequired(
                request.PostalCode);

            .SaveChangesAsync();

using System;
using System.Collections.Generic;
using System.Linq;
using System.Threading.Tasks;
using CustomerVehicleService.Application.Dtos.Readiness;
using CustomerVehicleService.Application.Interfaces;
using CustomerVehicleService.Application.Interfaces.Repositories;
using CustomerVehicleService.Domain.Entities;

namespace CustomerVehicleService.Application.Services;

public class CustomerReadinessService
    : ICustomerReadinessService
{
    private readonly IAgentCustomerRepository
        _agentCustomerRepository;

    private readonly IDriverRepository
        _driverRepository;

    private readonly IVehicleRepository
        _vehicleRepository;

    public CustomerReadinessService(
        IAgentCustomerRepository agentCustomerRepository,
        IDriverRepository driverRepository,
        IVehicleRepository vehicleRepository)
    {
        _agentCustomerRepository =
            agentCustomerRepository;

        _driverRepository =
            driverRepository;

        _vehicleRepository =
            vehicleRepository;
    }

    public async Task<CustomerReadinessDto?>
        GetForCustomerProfileAsync(
            Guid customerProfileId)
    {
        /*
         * -------------------------------------------------
         * PROFILE
         * -------------------------------------------------
         *
         * IAgentCustomerRepository already exposes the
         * CustomerProfile lookup we need.
         */
        var profile =
            await _agentCustomerRepository
                .GetCustomerProfileAsync(
                    customerProfileId);

        if (profile is null)
        {
            return null;
        }

        /*
         * -------------------------------------------------
         * AGENT ASSIGNMENT
         * -------------------------------------------------
         */
        var assignment =
            await _agentCustomerRepository
                .GetAssignmentAsync(
                    customerProfileId);

        /*
         * -------------------------------------------------
         * DRIVER
         * -------------------------------------------------
         */
        var driver =
            await _driverRepository
                .GetByCustomerProfileIdAsync(
                    customerProfileId);

        /*
         * -------------------------------------------------
         * VEHICLES
         * -------------------------------------------------
         */
        var vehicles =
            await _vehicleRepository
                .GetByCustomerProfileIdAsync(
                    customerProfileId);

        /*
         * -------------------------------------------------
         * EVALUATE EACH READINESS STAGE
         * -------------------------------------------------
         */
        var profileStage =
            EvaluateProfile(
                profile);

        var agentStage =
            EvaluateAgent(
                assignment);

        var driverStage =
            EvaluateDriver(
                profile,
                driver);

        var vehicleStage =
            EvaluateVehicles(
                vehicles);

        /*
         * -------------------------------------------------
         * AGGREGATE STRUCTURED ISSUES
         * -------------------------------------------------
         */
        var readinessIssues =
            new List<string>();

        readinessIssues.AddRange(
            profileStage.Issues);

        readinessIssues.AddRange(
            agentStage.Issues);

        readinessIssues.AddRange(
            driverStage.Issues);

        readinessIssues.AddRange(
            vehicleStage.Issues);

        /*
         * Overall CustomerVehicle readiness requires every
         * stage to be complete.
         */
        var readyForQuote =
            profileStage.Complete &&
            agentStage.Complete &&
            driverStage.Complete &&
            vehicleStage.Complete;

        return new CustomerReadinessDto
        {
            ReadyForQuote =
                readyForQuote,

            Profile =
                profileStage,

            Agent =
                agentStage,

            Driver =
                driverStage,

            Vehicle =
                vehicleStage,

            ReadinessIssues =
                readinessIssues
        };
    }

    /*
     * =====================================================
     * PROFILE READINESS
     * =====================================================
     */

    private static ReadinessStageDto
        EvaluateProfile(
            CustomerProfile profile)
    {
        var issues =
            new List<string>();

        /*
         * ProfileStatus is already calculated by the
         * Customer Profile workflow.
         *
         * P14 should not duplicate all Profile completion
         * validation rules here.
         */
        var profileComplete =
            string.Equals(
                profile.ProfileStatus.ToString(),
                "Complete",
                StringComparison.OrdinalIgnoreCase);

        if (!profileComplete)
        {
            issues.Add(
                "Customer profile is not complete.");
        }

        return CreateStage(
            issues);
    }

    /*
     * =====================================================
     * AGENT READINESS
     * =====================================================
     */

    private static ReadinessStageDto
        EvaluateAgent(
            AgentAssignment? assignment)
    {
        var issues =
            new List<string>();

        /*
         * Agent stage is complete only when a permanent
         * AgentAssignment exists.
         */
        if (assignment is null)
        {
            issues.Add(
                "Customer does not have a permanent Agent assignment.");
        }

        return CreateStage(
            issues);
    }

    /*
     * =====================================================
     * DRIVER READINESS
     * =====================================================
     */

    private static ReadinessStageDto
        EvaluateDriver(
            CustomerProfile profile,
            DriverProfile? driver)
    {
        var issues =
            new List<string>();

        /*
         * A Primary Driver must exist.
         */
        if (driver is null)
        {
            issues.Add(
                "Primary Driver has not been configured.");

            return CreateStage(
                issues);
        }

        var today =
            DateOnly.FromDateTime(
                DateTime.UtcNow);

        /*
         * -------------------------------------------------
         * LICENCE ISSUE DATE
         * -------------------------------------------------
         */
        if (
            driver.LicenceIssueDate >
            today)
        {
            issues.Add(
                "Primary Driver licence issue date cannot be in the future.");
        }

        /*
         * -------------------------------------------------
         * LICENCE EXPIRY DATE
         * -------------------------------------------------
         */
        if (
            driver.LicenceExpiryDate <=
            driver.LicenceIssueDate)
        {
            issues.Add(
                "Primary Driver licence expiry date must be later than the issue date.");
        }

        /*
         * An expired licence may exist in the database,
         * but it does not satisfy Driver readiness.
         */
        if (
            driver.LicenceExpiryDate <
            today)
        {
            issues.Add(
                "Primary Driver licence is expired.");
        }

        /*
         * -------------------------------------------------
         * DRIVER AGE
         * -------------------------------------------------
         *
         * CustomerProfile.DateOfBirth is nullable in the
         * Profile model exposed throughout this service,
         * therefore pattern matching safely handles both
         * the missing and populated cases.
         */
        if (
            profile.DateOfBirth is DateOnly
                dateOfBirth)
        {
            var age =
                CalculateCompletedYears(
                    dateOfBirth,
                    today);

            /*
             * An under-18 Driver may remain persisted,
             * but Driver readiness is false.
             */
            if (age < 18)
            {
                issues.Add(
                    "Primary Driver must be at least 18 years old for readiness.");
            }
        }
        else
        {
            issues.Add(
                "Primary Driver age cannot be calculated because Date of Birth is unavailable.");
        }

        /*
         * -------------------------------------------------
         * DRIVING EXPERIENCE
         * -------------------------------------------------
         *
         * Experience must be derivable from the licence
         * issue date.
         */
        if (
            driver.LicenceIssueDate <=
            today)
        {
            var drivingExperienceYears =
                CalculateCompletedYears(
                    driver.LicenceIssueDate,
                    today);

            /*
             * This is intentionally defensive.
             *
             * Zero completed years is valid. A negative
             * result would indicate invalid persisted data.
             */
            if (
                drivingExperienceYears < 0)
            {
                issues.Add(
                    "Primary Driver driving experience cannot be calculated.");
            }
        }

        return CreateStage(
            issues);
    }

    /*
     * =====================================================
     * VEHICLE READINESS
     * =====================================================
     */

    private static ReadinessStageDto
        EvaluateVehicles(
            IReadOnlyCollection<Vehicle>
                vehicles)
    {
        var issues =
            new List<string>();

        /*
         * The Customer needs at least one Vehicle.
         */
        if (vehicles.Count == 0)
        {
            issues.Add(
                "At least one Vehicle is required.");

            return CreateStage(
                issues);
        }

        var currentYear =
            DateTime.UtcNow.Year;

        /*
         * IMPORTANT:
         *
         * The Vehicle stage does NOT require every Vehicle
         * to be ready.
         *
         * At least one ready Vehicle is sufficient.
         */
        var hasReadyVehicle =
            vehicles.Any(
                vehicle =>
                    IsVehicleReady(
                        vehicle,
                        currentYear));

        if (!hasReadyVehicle)
        {
            issues.Add(
                "At least one Vehicle must meet the Vehicle readiness requirements.");
        }

        return CreateStage(
            issues);
    }

    /*
     * =====================================================
     * INDIVIDUAL VEHICLE READINESS
     * =====================================================
     */

    private static bool IsVehicleReady(
        Vehicle vehicle,
        int currentYear)
    {
        /*
         * ManufacturingYear cannot be future through
         * normal Vehicle validation.
         *
         * Math.Max keeps the derived age defensive if
         * unexpected persisted data exists.
         */
        var vehicleAge =
            Math.Max(
                currentYear -
                vehicle.ManufacturingYear,
                0);

        /*
         * Required identification information.
         */
        var identificationComplete =
            !string.IsNullOrWhiteSpace(
                vehicle.Vin) &&
            !string.IsNullOrWhiteSpace(
                vehicle.RegistrationNumber) &&
            !string.IsNullOrWhiteSpace(
                vehicle.Make) &&
            !string.IsNullOrWhiteSpace(
                vehicle.Model);

        /*
         * Required financial/usage information.
         */
        var riskDataComplete =
            vehicle.VehicleValue > 0 &&
            vehicle.AnnualMileage > 0;

        /*
         * Vehicle age greater than 20 is allowed to exist,
         * but cannot satisfy Vehicle readiness.
         */
        var ageEligible =
            vehicleAge <= 20;

        /*
         * Vehicle image is intentionally NOT checked.
         *
         * Image upload is optional and never affects
         * insurance readiness.
         */
        return
            identificationComplete &&
            riskDataComplete &&
            ageEligible;
    }

    /*
     * =====================================================
     * COMPLETED-YEAR CALCULATION
     * =====================================================
     *
     * Used for:
     *
     * - Customer/Driver age
     * - completed driving experience years
     *
     * This calculates completed calendar years rather than
     * simply subtracting the years.
     */

    private static int CalculateCompletedYears(
        DateOnly startDate,
        DateOnly endDate)
    {
        if (startDate > endDate)
        {
            return -1;
        }

        var completedYears =
            endDate.Year -
            startDate.Year;

        /*
         * Example:
         *
         * Start: 20 December 2020
         * Today:  03 October 2026
         *
         * 6 calendar years have not yet been completed,
         * therefore the result must be 5.
         */
        if (
            startDate >
            endDate.AddYears(
                -completedYears))
        {
            completedYears--;
        }

        return completedYears;
    }

    /*
     * =====================================================
     * STAGE BUILDER
     * =====================================================
     */

    private static ReadinessStageDto
        CreateStage(
            List<string> issues)
    {
        return new ReadinessStageDto
        {
            Complete =
                issues.Count == 0,

            Issues =
                issues
        };
    }
}

        var driver =

            AssignedAgentIdentityUserId =

                        profile.AreaType?
                            .ToString()
                        ?? string.Empty,

                    ProfileStatus =
                        profile.ProfileStatus
                            .ToString()

                vehicleRisks,

            ReadinessIssues =

                    .ToList()

                : GetCompletedYears(
                    dateOfBirth.Value,
                    today);

        var drivingExperienceYears =

                today);

        drivingExperienceYears =
            Math.Max(
                drivingExperienceYears,
                0);

        var licenceValid =
            driver.LicenceExpiryDate >=
            today;

        var ready =
            dateOfBirth is not null &&
            age >= 18 &&
            licenceValid;

        return new DriverRiskDto
        {
            DriverId =
                driver.Id,

            Age =
                age,

            DrivingExperienceYears =
                drivingExperienceYears,

            LicenceIssueDate =
                driver.LicenceIssueDate,

            LicenceExpiryDate =
                driver.LicenceExpiryDate,

            LicenceValid =
                licenceValid,

            Ready =
                ready
        };
    }

            AnnualMileage =
                vehicle.AnnualMileage,

            OwnershipType =
                vehicle.OwnershipType
                    .ToString(),

                vehicle.PurchaseDate,

                ready

    }

    public async Task<DriverOperationResult>

            await _agentCustomerRepository
                .GetCustomerProfileAsync(

        if (profile is null)

            return DriverOperationResult

        var assignment =

        if (
            assignment is null ||

        var driver =

                    CreatedAtUtc =

                .AddAsync(driver);
        }
        else
        {
            driver.UpdatedAtUtc =
                now;
        }

        driver.LicenceNumber =
            licenceNumber;

        driver.LicenceState =
            request.LicenceState

            await _driverRepository
                .SaveChangesAsync();

    private static DriverDto ToDto(
        DriverProfile driver,
        DateOnly? dateOfBirth)

        var today =
            DateOnly.FromDateTime(
                DateTime.UtcNow);

        var age =
            dateOfBirth is null
                ? 0
                : GetCompletedYears(
                    dateOfBirth.Value,
                    today);

        var experience =
            GetCompletedYears(
                driver.LicenceIssueDate,
                today);

        /*
         * Future issue dates are rejected by validation,
         * but guard against negative values anyway.
         */
        experience =
            Math.Max(
                experience,
                0);

        var licenceValid =
            driver.LicenceExpiryDate >=
            today;

        var issues =
            new List<string>();

        if (dateOfBirth is null)
        {
            issues.Add(
                "Customer date of birth is unavailable.");
        }
        else if (age < 18)
        {
            issues.Add(
                "Driver must be at least 18 years old for insurance readiness.");
        }

        if (!licenceValid)
        {
            issues.Add(
                "Driver licence has expired.");
        }

        var ready =
            dateOfBirth is not null &&
            age >= 18 &&
            licenceValid;

        return new DriverDto
        {
            Id =
                driver.Id,

            CustomerProfileId =
                driver.CustomerProfileId,

            LicenceNumber =
                driver.LicenceNumber,

            LicenceState =
                driver.LicenceState,

            LicenceIssueDate =
                driver.LicenceIssueDate,

            LicenceExpiryDate =
                driver.LicenceExpiryDate,

            Age =
                age,

            DrivingExperienceYears =
                experience,

            LicenceValid =
                licenceValid,

            Ready =
                ready,

            ReadinessIssues =
                issues,

            CreatedAtUtc =
                driver.CreatedAtUtc,

            UpdatedAtUtc =
                driver.UpdatedAtUtc
        };
    }

using CustomerVehicleService.Application.Dtos.VehicleImages;
using CustomerVehicleService.Application.Interfaces;
using CustomerVehicleService.Application.Interfaces.Repositories;
using CustomerVehicleService.Application.Interfaces.Storage;

namespace CustomerVehicleService.Application.Services;

public class VehicleImageService
    : IVehicleImageService
{
    private const long MaxFileSize =
        5L * 1024L * 1024L;

    private static readonly Dictionary<
        string,
        string> AllowedContentTypes =
            new(
                StringComparer.OrdinalIgnoreCase)
            {
                ["image/jpeg"] = ".jpg",
                ["image/png"] = ".png"
            };

    private readonly IVehicleRepository
        _vehicleRepository;

    private readonly IAgentCustomerRepository
        _agentCustomerRepository;

    private readonly ICustomerProfileRepository
        _customerProfileRepository;

    private readonly IVehicleImageStorage
        _imageStorage;

    public VehicleImageService(
        IVehicleRepository vehicleRepository,
        IAgentCustomerRepository agentCustomerRepository,
        ICustomerProfileRepository customerProfileRepository,
        IVehicleImageStorage imageStorage)
    {
        _vehicleRepository =
            vehicleRepository;

        _agentCustomerRepository =
            agentCustomerRepository;

        _customerProfileRepository =
            customerProfileRepository;

        _imageStorage =
            imageStorage;
    }

    public async Task<VehicleImageResult>
        UploadForAgentAsync(
            string agentIdentityUserId,
            Guid vehicleId,
            VehicleImageUpload upload)
    {
        var vehicle =
            await _vehicleRepository
                .GetByIdAsync(
                    vehicleId);

        if (vehicle is null)
        {
            return VehicleImageResult
                .MissingVehicle();
        }

        var assignment =
            await _agentCustomerRepository
                .GetAssignmentAsync(
                    vehicle.CustomerProfileId);

        if (
            assignment is null ||
            assignment.AgentIdentityUserId !=
                agentIdentityUserId)
        {
            return VehicleImageResult
                .Forbidden();
        }

        var validationError =
            ValidateUpload(upload);

        if (validationError is not null)
        {
            return VehicleImageResult
                .Invalid(
                    validationError);
        }

        var extension =
            AllowedContentTypes[
                upload.ContentType];

        string newStoredFileName;

        try
        {
            newStoredFileName =
                await _imageStorage.SaveAsync(
                    upload.Content,
                    extension);
        }
        catch
        {
            return VehicleImageResult.Invalid(
                "Vehicle image could not be stored.");
        }

        var previousStoredFileName =
            vehicle.ImageStoredFileName;

        vehicle.ImageFileName =
            Path.GetFileName(
                upload.FileName);

        vehicle.ImageStoredFileName =
            newStoredFileName;

        vehicle.ImageContentType =
            upload.ContentType;

        vehicle.ImageSizeBytes =
            upload.Length;

        vehicle.ImageUpdatedAtUtc =
            DateTime.UtcNow;

        vehicle.UpdatedAtUtc =
            DateTime.UtcNow;

        try
        {
            await _vehicleRepository
                .SaveChangesAsync();
        }
        catch
        {
            await _imageStorage.DeleteAsync(
                newStoredFileName);

            throw;
        }

        if (
            !string.IsNullOrWhiteSpace(
                previousStoredFileName))
        {
            await _imageStorage.DeleteAsync(
                previousStoredFileName);
        }

        return VehicleImageResult.Success();
    }

    public async Task<VehicleImageResult>
        GetForAgentAsync(
            string agentIdentityUserId,
            Guid vehicleId)
    {
        var vehicle =
            await _vehicleRepository
                .GetByIdAsync(
                    vehicleId);

        if (vehicle is null)
        {
            return VehicleImageResult
                .MissingVehicle();
        }

        var assignment =
            await _agentCustomerRepository
                .GetAssignmentAsync(
                    vehicle.CustomerProfileId);

        if (
            assignment is null ||
            assignment.AgentIdentityUserId !=
                agentIdentityUserId)
        {
            return VehicleImageResult
                .Forbidden();
        }

        return await ReadImageAsync(
            vehicle);
    }

    public async Task<VehicleImageResult>
        GetForCustomerAsync(
            string customerIdentityUserId,
            Guid vehicleId)
    {
        var profile =
            await _customerProfileRepository
                .GetByIdentityUserIdAsync(
                    customerIdentityUserId);

        if (profile is null)
        {
            return VehicleImageResult
                .Forbidden();
        }

        var vehicle =
            await _vehicleRepository
                .GetByIdAsync(
                    vehicleId);

        if (vehicle is null)
        {
            return VehicleImageResult
                .MissingVehicle();
        }

        if (
            vehicle.CustomerProfileId !=
            profile.Id)
        {
            return VehicleImageResult
                .Forbidden();
        }

        return await ReadImageAsync(
            vehicle);
    }

    private async Task<VehicleImageResult>
        ReadImageAsync(
            Domain.Entities.Vehicle vehicle)
    {
        if (
            string.IsNullOrWhiteSpace(
                vehicle.ImageStoredFileName) ||
            string.IsNullOrWhiteSpace(
                vehicle.ImageContentType))
        {
            return VehicleImageResult
                .MissingImage();
        }

        var content =
            await _imageStorage.ReadAsync(
                vehicle.ImageStoredFileName);

        if (content is null)
        {
            return VehicleImageResult
                .MissingImage();
        }

        return VehicleImageResult.Success(
            vehicle.ImageFileName
                ?? "vehicle-image",
            vehicle.ImageContentType,
            content);
    }

    private static string? ValidateUpload(
        VehicleImageUpload upload)
    {
        if (upload.Length <= 0)
        {
            return "Vehicle image is empty.";
        }

        if (upload.Length > MaxFileSize)
        {
            return
                "Vehicle image cannot exceed 5 MB.";
        }

        if (
            !AllowedContentTypes.ContainsKey(
                upload.ContentType))
        {
            return
                "Vehicle image must be JPG, JPEG, or PNG.";
        }

        return null;
    }
}

            await _agentCustomerRepository
                .GetCustomerProfileAsync(
                    customerProfileId);

        if (profile is null)
        {
            return VehicleOperationResult
                .MissingCustomer();
        }

        if (
            !await IsAssignedAgentAsync(
                agentIdentityUserId,
                customerProfileId))
        {
            return VehicleOperationResult
                .Forbidden();
        }

        var vin =
            NormalizeVin(
                request.Vin);

        var registration =
            NormalizeRegistration(
                request.RegistrationNumber);

        var vinExists =
            await _vehicleRepository
                .VinExistsForAnotherVehicleAsync(
                    vin);

        if (vinExists)
        {
            return VehicleOperationResult
                .VinConflict();
        }

        var registrationExists =
            await _vehicleRepository
                .RegistrationExistsForAnotherVehicleAsync(
                    registration);

        if (registrationExists)
        {
            return VehicleOperationResult
                .RegistrationConflict();
        }

        var vehicle =
            new Vehicle
            {
                Id =
                    Guid.NewGuid(),

                CustomerProfileId =
                    customerProfileId,

                CreatedAtUtc =
                    DateTime.UtcNow
            };

        ApplyRequest(
            vehicle,
            request,
            vin,
            registration);

        await _vehicleRepository
            .AddAsync(vehicle);

        try
        {
            await _vehicleRepository
                .SaveChangesAsync();
        }
        catch (Exception)
        {
            return await ResolveConflictAsync(
                vin,
                registration,
                vehicle.Id);
        }

        return VehicleOperationResult.Success(
            ToDto(vehicle));
    }

    public async Task<VehicleOperationResult>
        GetVehicleForAgentAsync(

    }

    public async Task<VehicleOperationResult>
        UpdateForAgentAsync(
            string agentIdentityUserId,
            Guid vehicleId,
            UpsertVehicleRequest request)
    {
        var vehicle =
            await _vehicleRepository
                .GetByIdAsync(
                    vehicleId);

        if (vehicle is null)
        {
            return VehicleOperationResult
                .MissingVehicle();
        }

        if (
            !await IsAssignedAgentAsync(
                agentIdentityUserId,

            return VehicleOperationResult
                .Forbidden();
        }

        var vin =
            NormalizeVin(

    public async Task<IReadOnlyList<VehicleDto>?>
        GetForCustomerAsync(
            string customerIdentityUserId)
    {
        var profile =
            await _customerProfileRepository
                .GetByIdentityUserIdAsync(
                    customerIdentityUserId);

        if (profile is null)
        {
            return null;
        }

        var vehicles =
            await _vehicleRepository
                .GetByCustomerProfileIdAsync(
                    profile.Id);

        return vehicles
            .Select(ToDto)
            .ToList();
    }

    public async Task<VehicleOperationResult>
        GetVehicleForCustomerAsync(
            string customerIdentityUserId,
            Guid vehicleId)
    {
        var profile =
            await _customerProfileRepository
                .GetByIdentityUserIdAsync(
                    customerIdentityUserId);

        if (profile is null)
        {
            return VehicleOperationResult
                .MissingCustomer();
        }

        var vehicle =
            await _vehicleRepository
                .GetByIdAsync(
                    vehicleId);

        if (vehicle is null)
        {
            return VehicleOperationResult
                .MissingVehicle();
        }

        if (
            vehicle.CustomerProfileId !=
            profile.Id)
        {
            return VehicleOperationResult
                .Forbidden();
        }

        return VehicleOperationResult.Success(
            ToDto(vehicle));
    }

    private async Task<bool>
        IsAssignedAgentAsync(
            string agentIdentityUserId,
            Guid customerProfileId)
    {
        var assignment =
            await _agentCustomerRepository
                .GetAssignmentAsync(
                    customerProfileId);

        return
            assignment is not null &&
            assignment.AgentIdentityUserId ==
                agentIdentityUserId;
    }

    private static void ApplyRequest(
        Vehicle vehicle,
        UpsertVehicleRequest request,
        string vin,
        string registration)
    {
        vehicle.Vin =
            vin;

        vehicle.RegistrationNumber =
            registration;

        vehicle.Make =
            request.Make.Trim();

        vehicle.Model =
            request.Model.Trim();

        vehicle.ManufacturingYear =
            request.ManufacturingYear;

        vehicle.VehicleType =
            request.VehicleType;

        vehicle.FuelType =
            request.FuelType;

        vehicle.VehicleValue =
            request.VehicleValue;

        vehicle.CurrentOdometer =
            request.CurrentOdometer;

        vehicle.AnnualMileage =
            request.AnnualMileage;

        vehicle.OwnershipType =
            request.OwnershipType;

        vehicle.PurchaseDate =
            request.PurchaseDate!.Value;
    }

    private static string NormalizeVin(
        string vin)
    {
        return vin
            .Trim()
            .ToUpperInvariant();
    }

    private static string NormalizeRegistration(
        string registrationNumber)
    {
        return registrationNumber
            .Trim()
            .ToUpperInvariant();
    }

    private static VehicleDto ToDto(
        Vehicle vehicle)
    {
        var currentYear =
            DateTime.UtcNow.Year;

        var vehicleAge =
            Math.Max(
                currentYear -
                vehicle.ManufacturingYear,
                0);

        var issues =
            new List<string>();

        if (vehicleAge > 20)
        {
            issues.Add(
                "Vehicle is more than 20 years old.");
        }

        if (vehicle.VehicleValue <= 0)
        {
            issues.Add(
                "Vehicle Value is required.");
        }

        if (vehicle.AnnualMileage <= 0)
        {
            issues.Add(
                "Annual Mileage is required.");
        }

        var ready =
            vehicleAge <= 20 &&
            vehicle.VehicleValue > 0 &&
            vehicle.AnnualMileage > 0;

        return new VehicleDto
        {
            Id =
                vehicle.Id,

            CustomerProfileId =
                vehicle.CustomerProfileId,

            Vin =
                vehicle.Vin,

            RegistrationNumber =
                vehicle.RegistrationNumber,

            Make =
                vehicle.Make,

            Model =
                vehicle.Model,

            ManufacturingYear =
                vehicle.ManufacturingYear,

            VehicleType =
                vehicle.VehicleType.ToString(),

            FuelType =
                vehicle.FuelType.ToString(),

            VehicleValue =
                vehicle.VehicleValue,

            CurrentOdometer =
                vehicle.CurrentOdometer,

            AnnualMileage =
                vehicle.AnnualMileage,

            OwnershipType =
                vehicle.OwnershipType.ToString(),

            PurchaseDate =
                vehicle.PurchaseDate,

            VehicleAge =
                vehicleAge,

            Ready =
                ready,

            ReadinessIssues =
                issues,

            /*
             * Image metadata.
             *
             * We expose whether an image exists and safe
             * display metadata, but never expose the
             * internally generated storage filename or
             * physical filesystem path.
             */
            HasImage =
                !string.IsNullOrWhiteSpace(
                    vehicle.ImageStoredFileName),

            ImageFileName =
                vehicle.ImageFileName,

            ImageContentType =
                vehicle.ImageContentType,

            ImageSizeBytes =
                vehicle.ImageSizeBytes,

            ImageUpdatedAtUtc =
                vehicle.ImageUpdatedAtUtc,

            CreatedAtUtc =
                vehicle.CreatedAtUtc,

            UpdatedAtUtc =
                vehicle.UpdatedAtUtc
        };
    }

    private async Task<VehicleOperationResult>
        ResolveConflictAsync(
            string vin,
            string registrationNumber,
            Guid vehicleId)
    {
        var vinExists =
            await _vehicleRepository
                .VinExistsForAnotherVehicleAsync(
                    vin,
                    vehicleId);

        if (vinExists)
        {
            return VehicleOperationResult
                .VinConflict();
        }

        var registrationExists =
            await _vehicleRepository
                .RegistrationExistsForAnotherVehicleAsync(
                    registrationNumber,
                    vehicleId);

        if (registrationExists)
        {
            return VehicleOperationResult
                .RegistrationConflict();
        }

        throw new InvalidOperationException(
            "Vehicle could not be saved because of a database update error.");
    }
}

    : AbstractValidator<UpsertDriverRequest>

                request =>
                    request.LicenceNumber)
            .NotEmpty()
            .WithMessage(
                "Licence number is required.")
            .MaximumLength(50)
            .WithMessage(
                "Licence number cannot exceed 50 characters.");

        RuleFor(
                request =>
                    request.LicenceState)
            .NotEmpty()
            .WithMessage(
                "Licence state is required.")
            .MaximumLength(100)
            .WithMessage(
                "Licence state cannot exceed 100 characters.");

        RuleFor(
                request =>
                    request.LicenceIssueDate)
            .NotNull()
            .WithMessage(
                "Licence issue date is required.")
            .Must(
                issueDate =>
                    issueDate is null ||
                    issueDate.Value <=
                    DateOnly.FromDateTime(
                        DateTime.UtcNow))
            .WithMessage(
                "Licence issue date cannot be in the future.");

        RuleFor(
                request =>
                    request.LicenceExpiryDate)
            .NotNull()
            .WithMessage(
                "Licence expiry date is required.");

        RuleFor(request => request)
            .Must(request =>
            {
                if (
                    request.LicenceIssueDate is null ||
                    request.LicenceExpiryDate is null)
                {
                    return true;
                }

                return
                    request.LicenceExpiryDate.Value >
                    request.LicenceIssueDate.Value;
            })
            .WithMessage(
                "Licence expiry date must be later than the issue date.");
    }
}
