    public async Task<AssignUnderwriterResult> AssignAsync(
    Guid applicationId,
    AssignUnderwriterRequest request,
    string adminIdentityUserId)
    {
        ArgumentNullException.ThrowIfNull(request);

        if (applicationId == Guid.Empty)
        {
            return AssignUnderwriterResult.Missing(
                "Underwriting application was not found.");
        }

        if (string.IsNullOrWhiteSpace(
                adminIdentityUserId))
        {
            throw new ArgumentException(
                "Admin Identity User ID is required.",
                nameof(adminIdentityUserId));
        }

        if (string.IsNullOrWhiteSpace(
                request.UnderwriterIdentityUserId))
        {
            return AssignUnderwriterResult.Ineligible(
                "Underwriter Identity User ID is required.");
        }

        var application =
            await _repository.GetByIdForUpdateAsync(
                applicationId);

        if (application is null)
        {
            return AssignUnderwriterResult.Missing(
                "Underwriting application was not found.");
        }

        if (application.Status !=
                ApplicationStatus.Unassigned ||
            !string.IsNullOrWhiteSpace(
                application.AssignedUnderwriterIdentityUserId))
        {
            return AssignUnderwriterResult.AlreadyAssigned(
                "Underwriting application has already been assigned.");
        }

        var underwriterResult =
            await _identityClient
                .GetEligibleUnderwriterAsync(
                    request.UnderwriterIdentityUserId);

        if (underwriterResult.DependencyUnavailable)
        {
            return AssignUnderwriterResult.Unavailable(
                underwriterResult.Message ??
                "Underwriter eligibility could not be verified.");
        }

        if (!underwriterResult.Succeeded ||
            underwriterResult.Underwriter is null)
        {
            return AssignUnderwriterResult.Ineligible(
                underwriterResult.Message ??
                "Selected Underwriter is not currently eligible.");
        }

        var nowUtc =
            DateTime.UtcNow;

        await _repository.ExecuteInTransactionAsync(
    async () =>
    {
        application.AssignedUnderwriterIdentityUserId =
            underwriterResult.Underwriter.Id;

        application.AssignedByIdentityUserId =
            adminIdentityUserId;

        application.AssignedAtUtc =
            nowUtc;

        application.Status =
            ApplicationStatus.PendingReview;

        var history =
            new ApplicationStatusHistory
            {
                Id =
                    Guid.NewGuid(),

                UnderwritingApplicationId =
                    application.Id,

                FromStatus =
                    ApplicationStatus.Unassigned,

                ToStatus =
                    ApplicationStatus.PendingReview,

                ChangedByIdentityUserId =
                    adminIdentityUserId,

                ChangedAtUtc =
                    nowUtc
            };

        await _repository.AddStatusHistoryAsync(
            history);

        await _repository.SaveChangesAsync();
    });

        var assignedApplication =
            await _repository.GetByIdAsync(
                application.Id);

        if (assignedApplication is null)
        {
            throw new InvalidOperationException(
                "Assigned Underwriting application could not be reloaded.");
        }

        return AssignUnderwriterResult.Success(
            MapAdminApplicationDetail(
                assignedApplication));
    }

    public async Task<EligibleUnderwritersAdminResult>
        GetEligibleUnderwritersAsync()
    {
        var identityResult =
            await _identityClient
                .GetEligibleUnderwritersAsync();

        if (!identityResult.Succeeded)
        {
            return EligibleUnderwritersAdminResult.Unavailable(
                identityResult.Message ??
                "Eligible Underwriters are temporarily unavailable.");
        }

        var result =
            new List<EligibleUnderwriterWithWorkloadDto>();

        foreach (var underwriter in identityResult.Underwriters)
        {
            var actionableCount =
                await _repository
                    .GetActionableCountForUnderwriterAsync(
                        underwriter.Id);

            result.Add(
                new EligibleUnderwriterWithWorkloadDto
                {
                    IdentityUserId =
                        underwriter.Id,

                    FirstName =
                        underwriter.FirstName,

                    LastName =
                        underwriter.LastName,

                    Email =
                        underwriter.Email,

                    ActionableApplicationCount =
                        actionableCount
                });
        }

        return EligibleUnderwritersAdminResult.Success(
            result);
    }

    public async Task<UpdateReviewSectionResult>
    UpdateReviewSectionAsync(
        Guid applicationId,
        ReviewSectionType sectionType,
        UpdateReviewSectionRequest request,
        string underwriterIdentityUserId)
    {
        ArgumentNullException.ThrowIfNull(request);

        if (applicationId == Guid.Empty)
        {
            return UpdateReviewSectionResult.Missing(
                "Underwriting application was not found.");
        }

        if (string.IsNullOrWhiteSpace(
                underwriterIdentityUserId))
        {
            throw new ArgumentException(
                "Underwriter Identity User ID is required.",
                nameof(underwriterIdentityUserId));
        }

        if (!Enum.IsDefined(
                typeof(ReviewSectionType),
                sectionType))
        {
            return UpdateReviewSectionResult.Invalid(
                "Review section type is invalid.");
        }

        if (request.Notes is not null &&
            request.Notes.Length > 2000)
        {
            return UpdateReviewSectionResult.Invalid(
                "Review notes cannot exceed 2000 characters.");
        }

        var application =
            await _repository
                .GetAssignedByIdForUpdateAsync(
                    applicationId,
                    underwriterIdentityUserId);

        if (application is null)
        {
            var existingApplication =
                await _repository.GetByIdAsync(
                    applicationId);

            if (existingApplication is null)
            {
                return UpdateReviewSectionResult.Missing(
                    "Underwriting application was not found.");
            }

            return UpdateReviewSectionResult.AccessDenied(
                "Underwriting application is not assigned to the current Underwriter.");
        }

        if (application.Status !=
            ApplicationStatus.InReview)
        {
            return UpdateReviewSectionResult.Invalid(
                "Review sections can only be updated while the application is InReview.");
        }

        var section =
            await _repository
                .GetReviewSectionForUpdateAsync(
                    applicationId,
                    sectionType);

        if (section is null)
        {
            return UpdateReviewSectionResult.Missing(
                "Review section was not found.");
        }

        var normalizedNotes =
            string.IsNullOrWhiteSpace(
                request.Notes)
                ? null
                : request.Notes.Trim();

        section.Notes =
            normalizedNotes;

        if (request.Reviewed)
        {
            section.Reviewed =
                true;

            section.ReviewedByIdentityUserId =
                underwriterIdentityUserId;

            section.ReviewedAtUtc =
                DateTime.UtcNow;
        }
        else
        {
            section.Reviewed =
                false;

            section.ReviewedByIdentityUserId =
                null;

            section.ReviewedAtUtc =
                null;
        }

        await _repository.SaveChangesAsync();

        return UpdateReviewSectionResult.Success(
            new UnderwriterReviewSectionDto
            {
                SectionId =
                    section.Id,

                SectionType =
                    section.SectionType.ToString(),

                Reviewed =
                    section.Reviewed,

                ReviewedAtUtc =
                    section.ReviewedAtUtc,

                Notes =
                    section.Notes
            });
    }

    public async Task<MakeUnderwritingDecisionResult>
    MakeDecisionAsync(
        Guid applicationId,
        MakeUnderwritingDecisionRequest request,
        string underwriterIdentityUserId)
    {
        ArgumentNullException.ThrowIfNull(request);

        if (applicationId == Guid.Empty)
        {
            return MakeUnderwritingDecisionResult.Missing(
                "Underwriting application was not found.");
        }

        if (string.IsNullOrWhiteSpace(
                underwriterIdentityUserId))
        {
            throw new ArgumentException(
                "Underwriter Identity User ID is required.",
                nameof(underwriterIdentityUserId));
        }

        if (!Enum.TryParse<UnderwritingDecisionType>(
                request.Decision,
                ignoreCase: true,
                out var decisionType) ||
            !Enum.IsDefined(
                typeof(UnderwritingDecisionType),
                decisionType))
        {
            return MakeUnderwritingDecisionResult.Invalid(
                "Decision must be Approved or Rejected.");
        }

        var rejectionReason =
            string.IsNullOrWhiteSpace(
                request.RejectionReason)
                ? null
                : request.RejectionReason.Trim();

        if (decisionType ==
                UnderwritingDecisionType.Rejected &&
            string.IsNullOrWhiteSpace(
                rejectionReason))
        {
            return MakeUnderwritingDecisionResult.Invalid(
                "A rejection reason is required when rejecting an application.");
        }

        if (decisionType ==
                UnderwritingDecisionType.Approved &&
            rejectionReason is not null)
        {
            return MakeUnderwritingDecisionResult.Invalid(
                "A rejection reason must not be provided when approving an application.");
        }

        if (rejectionReason is not null &&
            rejectionReason.Length > 2000)
        {
            return MakeUnderwritingDecisionResult.Invalid(
                "Rejection reason cannot exceed 2000 characters.");
        }

        var application =
            await _repository
                .GetAssignedByIdForUpdateAsync(
                    applicationId,
                    underwriterIdentityUserId);

        if (application is null)
        {
            var existingApplication =
                await _repository.GetByIdAsync(
                    applicationId);

            if (existingApplication is null)
            {
                return MakeUnderwritingDecisionResult.Missing(
                    "Underwriting application was not found.");
            }

            return MakeUnderwritingDecisionResult.AccessDenied(
                "Underwriting application is not assigned to the current Underwriter.");
        }

        if (application.Status !=
            ApplicationStatus.InReview)
        {
            return MakeUnderwritingDecisionResult.Invalid(
                "Only an application that is InReview can receive an Underwriting decision.");
        }

        var decisionExists =
            await _repository.DecisionExistsAsync(
                applicationId);

        if (decisionExists)
        {
            return MakeUnderwritingDecisionResult.Invalid(
                "An Underwriting decision has already been recorded.");
        }

        var sections =
            await _repository.GetReviewSectionsAsync(
                applicationId);

        var requiredSectionTypes =
            Enum.GetValues<ReviewSectionType>();

        if (sections.Count !=
            requiredSectionTypes.Length)
        {
            return MakeUnderwritingDecisionResult
                .ReviewIncomplete(
                    "All four review sections must exist before a decision can be made.");
        }

        var allReviewed =
            requiredSectionTypes.All(
                requiredType =>
                    sections.Any(
                        section =>
                            section.SectionType ==
                                requiredType &&
                            section.Reviewed));

        if (!allReviewed)
        {
            return MakeUnderwritingDecisionResult
                .ReviewIncomplete(
                    "All four review sections must be reviewed before a decision can be made.");
        }

        var nowUtc =
            DateTime.UtcNow;

        var newStatus =
            decisionType ==
            UnderwritingDecisionType.Approved
                ? ApplicationStatus.Approved
                : ApplicationStatus.Rejected;

        var decision =
            new UnderwritingDecision
            {
                Id =
                    Guid.NewGuid(),

                UnderwritingApplicationId =
                    application.Id,

                DecisionType =
                    decisionType,

                DecidedByIdentityUserId =
                    underwriterIdentityUserId,

                DecidedAtUtc =
                    nowUtc,

                RejectionReason =
                    rejectionReason
            };

        await _repository.ExecuteInTransactionAsync(
            async () =>
            {
                await _repository.AddDecisionAsync(
                    decision);

                application.Status =
                    newStatus;

                application.DecisionAtUtc =
                    nowUtc;

                if (decisionType ==
                    UnderwritingDecisionType.Approved)
                {
                    application.ApprovedAtUtc =
                        nowUtc;

                    application.OfferValidUntilUtc =
                        nowUtc.AddDays(7);
                }

                var history =
                    new ApplicationStatusHistory
                    {
                        Id =
                            Guid.NewGuid(),

                        UnderwritingApplicationId =
                            application.Id,

                        FromStatus =
                            ApplicationStatus.InReview,

                        ToStatus =
                            newStatus,

                        ChangedByIdentityUserId =
                            underwriterIdentityUserId,

                        ChangedAtUtc =
                            nowUtc
                    };

                await _repository
                    .AddStatusHistoryAsync(
                        history);

                await _repository
                    .SaveChangesAsync();
            });

        return MakeUnderwritingDecisionResult.Success(
            new UnderwritingDecisionDto
            {
                ApplicationId =
                    application.Id,

                Status =
                    application.Status.ToString(),

                Decision =
                    decision.DecisionType.ToString(),

                DecidedAtUtc =
                    decision.DecidedAtUtc,

                RejectionReason =
                    decision.RejectionReason,

                OfferValidUntilUtc =
                    application.OfferValidUntilUtc
            });
    }

    public async Task<UnderwriterReviewResult>
        GetReviewAsync(
            Guid applicationId,
            string underwriterIdentityUserId)
    {
        if (applicationId == Guid.Empty)
        {
            return UnderwriterReviewResult.Missing(
                "Underwriting application was not found.");
        }

        if (string.IsNullOrWhiteSpace(
                underwriterIdentityUserId))
        {
            throw new ArgumentException(
                "Underwriter Identity User ID is required.",
                nameof(underwriterIdentityUserId));
        }

        var application =
            await _repository
                .GetAssignedByIdForUpdateAsync(
                    applicationId,
                    underwriterIdentityUserId);

        if (application is null)
        {
            var existingApplication =
                await _repository.GetByIdAsync(
                    applicationId);

            if (existingApplication is null)
            {
                return UnderwriterReviewResult.Missing(
                    "Underwriting application was not found.");
            }

            return UnderwriterReviewResult.AccessDenied(
                "Underwriting application is not assigned to the current Underwriter.");
        }

        if (application.Status ==
            ApplicationStatus.PendingReview)
        {
            var nowUtc =
                DateTime.UtcNow;

            await _repository.ExecuteInTransactionAsync(
                async () =>
                {
                    application.Status =
                        ApplicationStatus.InReview;

                    application.ReviewStartedAtUtc =
                        nowUtc;

                    var history =
                        new ApplicationStatusHistory
                        {
                            Id = Guid.NewGuid(),

                            UnderwritingApplicationId =
                                application.Id,

                            FromStatus =
                                ApplicationStatus.PendingReview,

                            ToStatus =
                                ApplicationStatus.InReview,

                            ChangedByIdentityUserId =
                                underwriterIdentityUserId,

                            ChangedAtUtc =
                                nowUtc
                        };

                    await _repository
                        .AddStatusHistoryAsync(
                            history);

                    await _repository
                        .SaveChangesAsync();
                });
        }
        else if (application.Status !=
                 ApplicationStatus.InReview)
        {
            return UnderwriterReviewResult.Invalid(
                "Underwriting application is not available for review.");
        }

        var reviewApplication =
            await _repository.GetByIdAsync(
                application.Id);

        if (reviewApplication is null)
        {
            throw new InvalidOperationException(
                "Underwriting application could not be reloaded after opening the review.");
        }

        return UnderwriterReviewResult.Success(
            MapReview(
                reviewApplication));
    }

                    Vin =
                        snapshot.VehicleVin,

                    Make =
                        snapshot.VehicleMake,

                    Model =
                        snapshot.VehicleModel,

                    ManufacturingYear =
                        snapshot.VehicleManufacturingYear,

                    VehicleAge =
                        snapshot.VehicleAge,

                    VehicleType =
                        snapshot.VehicleType,

                    FuelType =
                        snapshot.FuelType,

                    VehicleValue =
                        snapshot.VehicleValue,

                    CurrentOdometer =
                        snapshot.CurrentOdometer,

                    AnnualMileage =
                        snapshot.AnnualMileage,

                    OwnershipType =
                        snapshot.OwnershipType,

                    PurchaseDate =
                        snapshot.VehiclePurchaseDate
                },

            Quote =
                new UnderwriterQuoteReviewDto
                {
                    PlanId =
                        snapshot.PlanId,

                    PlanName =
                        snapshot.PlanName,

                    AnnualPremium =
                        snapshot.AnnualPremium,

                    OverallRiskRating =
                        snapshot.OverallRiskRating
                            .ToString(),

                    QuoteFinalizedAtUtc =
                        snapshot.QuoteFinalizedAtUtc,

                    QuoteValidUntilUtc =
                        snapshot.QuoteValidUntilUtc,

                    Coverages =
                        snapshot.Coverages
                            .Select(
                                coverage =>
                                    new UnderwriterCoverageDto
                                    {
                                        CoverageName =
                                            coverage.CoverageName,

                                        ResolvedLimitAmount =
                                            coverage.ResolvedLimitAmount,

                                        DeductibleAmount =
                                            coverage.DeductibleAmount
                                    })
                            .ToList(),

                    AddOns =
                        snapshot.AddOns
                            .Select(
                                addOn =>
                                    new UnderwriterAddOnDto
                                    {
                                        AddOnName =
                                            addOn.AddOnName,

                                        AnnualPrice =
                                            addOn.AnnualPrice
                                    })
                            .ToList()
                },

            ReviewSections =
                application.ReviewSections
                    .OrderBy(
                        section =>
                            section.SectionType)
                    .Select(
                        section =>
                            new UnderwriterReviewSectionDto
                            {
                                SectionId =
                                    section.Id,

                                SectionType =
                                    section.SectionType
                                        .ToString(),

                                Reviewed =
                                    section.Reviewed,

                                ReviewedAtUtc =
                                    section.ReviewedAtUtc,

                                Notes =
                                    section.Notes
                            })
                    .ToList()
        };
    }

using UnderwritingService.Application.Dtos.Applications;
using UnderwritingService.Application.Interfaces;
using UnderwritingService.Application.Interfaces.Repositories;
using UnderwritingService.Domain.Entities;
using UnderwritingService.Domain.Enums;

namespace UnderwritingService.Application.Services;

public class UnderwritingApplicationService : IUnderwritingApplicationService
{
    private readonly IUnderwritingApplicationRepository _repository;

    public UnderwritingApplicationService(
        IUnderwritingApplicationRepository repository)
    {
        _repository = repository;
    }

    public async Task<CreateUnderwritingApplicationResult> CreateAsync(
        CreateUnderwritingApplicationRequest request)
    {
        ArgumentNullException.ThrowIfNull(request);

        var validationMessage = ValidateRequest(request);

        if (validationMessage is not null)
        {
            return CreateUnderwritingApplicationResult.Failed(
                validationMessage);
        }

        var existingApplication =
            await _repository.GetByOriginatingQuoteIdAsync(
                request.OriginatingQuoteId);

        if (existingApplication is not null)
        {
            return CreateUnderwritingApplicationResult.Success(
                MapResponse(
                    existingApplication,
                    existingApplication: true),
                existingApplication: true);
        }

        var application = new UnderwritingApplication
        {
            Id = Guid.NewGuid(),
            OriginatingQuoteId = request.OriginatingQuoteId,
            CustomerProfileId = request.CustomerProfileId,
            VehicleId = request.VehicleId,
            AgentIdentityUserId =
                request.AgentIdentityUserId.Trim(),
            Status = ApplicationStatus.Unassigned,
            CreatedAtUtc = DateTime.UtcNow
        };

        var snapshot = new ApplicationSnapshot
        {
            Id = Guid.NewGuid(),
            UnderwritingApplicationId = application.Id,

            QuoteFinalizedAtUtc =
                request.QuoteFinalizedAtUtc,

            QuoteValidUntilUtc =
                request.QuoteValidUntilUtc,

            CustomerDateOfBirth =
                request.CustomerDateOfBirth,

            Geography =
                request.Geography.Trim(),

            DriverId =
                request.DriverId,

            DriverAge =
                request.DriverAge,

            DrivingExperienceYears =
                request.DrivingExperienceYears,

            LicenceIssueDate =
                request.LicenceIssueDate,

            LicenceExpiryDate =
                request.LicenceExpiryDate,

            LicenceValid =
                request.LicenceValid,

            VehicleVin =
                request.VehicleVin.Trim(),

            VehicleMake =
                request.VehicleMake.Trim(),

            VehicleModel =
                request.VehicleModel.Trim(),

            VehicleManufacturingYear =
                request.VehicleManufacturingYear,

            VehicleAge =
                request.VehicleAge,

            VehicleType =
                request.VehicleType.Trim(),

            FuelType =
                request.FuelType.Trim(),

            VehicleValue =
                request.VehicleValue,

            CurrentOdometer =
                request.CurrentOdometer,

            AnnualMileage =
                request.AnnualMileage,

            OwnershipType =
                request.OwnershipType.Trim(),

            VehiclePurchaseDate =
                request.VehiclePurchaseDate,

            PlanId =
                request.PlanId,

            PlanName =
                request.PlanName.Trim(),

            AnnualPremium =
                request.AnnualPremium,

            QualifyingClaimsCount =
                request.QualifyingClaimsCount,

            OverallRiskRating =
                request.OverallRiskRating
        };

        foreach (var coverage in request.Coverages)
        {
            snapshot.Coverages.Add(
                new ApplicationCoverageSnapshot
                {
                    Id = Guid.NewGuid(),
                    ApplicationSnapshotId = snapshot.Id,
                    CoverageName =
                        coverage.CoverageName.Trim(),
                    ResolvedLimitAmount =
                        coverage.ResolvedLimitAmount,
                    DeductibleAmount =
                        coverage.DeductibleAmount
                });
        }

        foreach (var addOn in request.AddOns)
        {
            snapshot.AddOns.Add(
                new ApplicationAddOnSnapshot
                {
                    Id = Guid.NewGuid(),
                    ApplicationSnapshotId = snapshot.Id,
                    AddOnName =
                        addOn.AddOnName.Trim(),
                    AnnualPrice =
                        addOn.AnnualPrice
                });
        }

        application.Snapshot = snapshot;

        foreach (var sectionType in
                 Enum.GetValues<ReviewSectionType>())
        {
            application.ReviewSections.Add(
                new ApplicationReviewSection
                {
                    Id = Guid.NewGuid(),
                    UnderwritingApplicationId =
                        application.Id,
                    SectionType = sectionType,
                    Reviewed = false
                });
        }

        try
        {
            await _repository.ExecuteInTransactionAsync(
                async () =>
                {
                    await _repository.AddAsync(
                        application);

                    await _repository.SaveChangesAsync();
                });
        }
        catch
        {
            var concurrentApplication =
                await _repository
                    .GetByOriginatingQuoteIdAsync(
                        request.OriginatingQuoteId);

            if (concurrentApplication is not null)
            {
                return CreateUnderwritingApplicationResult.Success(
                    MapResponse(
                        concurrentApplication,
                        existingApplication: true),
                    existingApplication: true);
            }

            throw;
        }

        return CreateUnderwritingApplicationResult.Success(
            MapResponse(
                application,
                existingApplication: false),
            existingApplication: false);
    }

    private static string? ValidateRequest(
        CreateUnderwritingApplicationRequest request)
    {
        if (request.OriginatingQuoteId == Guid.Empty)
        {
            return "Originating Quote ID is required.";
        }

        if (request.CustomerProfileId == Guid.Empty)
        {
            return "Customer Profile ID is required.";
        }

        if (request.VehicleId == Guid.Empty)
        {
            return "Vehicle ID is required.";
        }

        if (string.IsNullOrWhiteSpace(
                request.AgentIdentityUserId))
        {
            return "Agent Identity User ID is required.";
        }

        if (request.QuoteFinalizedAtUtc == default)
        {
            return "Quote finalized timestamp is required.";
        }

        if (request.QuoteValidUntilUtc == default)
        {
            return "Quote validity timestamp is required.";
        }

        if (request.QuoteValidUntilUtc <=
            request.QuoteFinalizedAtUtc)
        {
            return
                "Quote validity timestamp must be after the finalized timestamp.";
        }

        if (request.CustomerDateOfBirth == default)
        {
            return "Customer Date of Birth is required.";
        }

        if (string.IsNullOrWhiteSpace(
                request.Geography))
        {
            return "Geography is required.";
        }

        if (request.DriverId == Guid.Empty)
        {
            return "Driver ID is required.";
        }

        if (request.DriverAge < 0)
        {
            return "Driver Age cannot be negative.";
        }

        if (request.DrivingExperienceYears < 0)
        {
            return
                "Driving Experience cannot be negative.";
        }

        if (request.LicenceIssueDate == default)
        {
            return "Licence Issue Date is required.";
        }

        if (request.LicenceExpiryDate == default)
        {
            return "Licence Expiry Date is required.";
        }

        if (request.LicenceExpiryDate <=
            request.LicenceIssueDate)
        {
            return
                "Licence Expiry Date must be after Licence Issue Date.";
        }

        if (string.IsNullOrWhiteSpace(
                request.VehicleVin))
        {
            return "Vehicle VIN is required.";
        }

        if (string.IsNullOrWhiteSpace(
                request.VehicleMake))
        {
            return "Vehicle Make is required.";
        }

        if (string.IsNullOrWhiteSpace(
                request.VehicleModel))
        {
            return "Vehicle Model is required.";
        }

        if (request.VehicleManufacturingYear <= 0)
        {
            return
                "Vehicle Manufacturing Year is invalid.";
        }

        if (request.VehicleAge < 0)
        {
            return "Vehicle Age cannot be negative.";
        }

        if (string.IsNullOrWhiteSpace(
                request.VehicleType))
        {
            return "Vehicle Type is required.";
        }

        if (string.IsNullOrWhiteSpace(
                request.FuelType))
        {
            return "Fuel Type is required.";
        }

        if (request.VehicleValue <= 0)
        {
            return
                "Vehicle Value must be greater than zero.";
        }

        if (request.CurrentOdometer < 0)
        {
            return
                "Current Odometer cannot be negative.";
        }

        if (request.AnnualMileage <= 0)
        {
            return
                "Annual Mileage must be greater than zero.";
        }

        if (string.IsNullOrWhiteSpace(
                request.OwnershipType))
        {
            return "Ownership Type is required.";
        }

        if (request.VehiclePurchaseDate == default)
        {
            return "Vehicle Purchase Date is required.";
        }

        if (request.PlanId == Guid.Empty)
        {
            return "Plan ID is required.";
        }

        if (string.IsNullOrWhiteSpace(
                request.PlanName))
        {
            return "Plan Name is required.";
        }

        if (request.AnnualPremium < 0)
        {
            return
                "Annual Premium cannot be negative.";
        }

        if (request.QualifyingClaimsCount < 0)
        {
            return
                "Qualifying Claims Count cannot be negative.";
        }

        if (!Enum.IsDefined(
                typeof(OverallRiskRating),
                request.OverallRiskRating))
        {
            return
                "Overall Risk Rating is invalid.";
        }

        if (request.Coverages is null ||
            request.Coverages.Count == 0)
        {
            return
                "At least one Coverage is required.";
        }

        foreach (var coverage in request.Coverages)
        {
            if (string.IsNullOrWhiteSpace(
                    coverage.CoverageName))
            {
                return
                    "Coverage Name is required.";
            }

            if (coverage.ResolvedLimitAmount <= 0)
            {
                return
                    $"Coverage '{coverage.CoverageName}' has an invalid limit.";
            }

            if (coverage.DeductibleAmount.HasValue &&
                coverage.DeductibleAmount.Value <= 0)
            {
                return
                    $"Coverage '{coverage.CoverageName}' has an invalid deductible.";
            }
        }

        if (request.AddOns is not null)
        {
            foreach (var addOn in request.AddOns)
            {
                if (string.IsNullOrWhiteSpace(
                        addOn.AddOnName))
                {
                    return
                        "Add-on Name is required.";
                }

                if (addOn.AnnualPrice < 0)
                {
                    return
                        $"Add-on '{addOn.AddOnName}' has an invalid price.";
                }
            }
        }

        return null;
    }

    private static CreateUnderwritingApplicationResponse MapResponse(
        UnderwritingApplication application,
        bool existingApplication)
    {
        return new CreateUnderwritingApplicationResponse
        {
            ApplicationId =
                application.Id,

            OriginatingQuoteId =
                application.OriginatingQuoteId,

            Status =
                application.Status.ToString(),

            ExistingApplication =
                existingApplication,

            CreatedAtUtc =
                application.CreatedAtUtc
        };
    }
}
